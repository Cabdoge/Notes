\chapter{同伦}

\section{同伦}


\begin{definition}{同伦}
    设$X,Y$是拓扑空间，$f,g:X\to Y$是两个连续映射，如果存在一个连续映射$H:X\times[0,1]\to Y$，使得$H(x,0)=f(x),H(x,1)=g(x)$，则称$f$和$g$是\textbf{同伦的}，记作$f\simeq g$，映射$H$称为从$f$到$g$的\textbf{同伦}，记作$H:f\simeq g$.
\end{definition}

\begin{figure}[h]
    \centering
    \includegraphics[width=0.6\textwidth]{fig/10.1.png}
    \caption{同伦示意图}
\end{figure}
如果固定$t$，可以定义函数$\setjot[-6]\begin{aligned}[t]
    h_t:X&\to Y \\
    x&\mapsto H(x,t)
\end{aligned} $，如果固定$x$，则$\setjot[-6]\begin{aligned}[t]
    a_x:[0,1]&\to Y \\
    t&\mapsto H(x,t)
\end{aligned}$是一条道路.

\begin{instance}
    设$X$是一个拓扑空间，$Y$是$\mb{E}^n$的凸子空间，任取$f,g:X\to Y$，则$f\simeq g$.\par
    可以构造$\setjot[-6]\begin{aligned}[t]
        H:X\times [0,1]&\to Y \\
        (x,t) &\mapsto (1-t)f(x)+tg(x)
     \end{aligned}$.则$H(x,0)=f(x),H(x,1)=g(x)$.\par
     这种同伦称为\bluebf{直线同伦}.
\end{instance}

\begin{instance}
    设$f,g:X\to S^n$，且$\forall\, x\in X$，$f(x)+g(x)\neq 0$，则$f\simeq g$.\par
    构造$\setjot[-6]\begin{aligned}[t]
        H:X\times [0,1]&\to S^n \\
        (x,t) &\mapsto \frac{(1-t)f(x)+tg(x)}{\|(1-t)f(x)+tg(x)\|}
     \end{aligned}$.则$H(x,0)=f(x),H(x,1)=g(x)$.\par
    上述同伦实际上是$\mb{E}^{n+1}\setminus \{0\}$中的直线同伦投影到$S^n$上\par
\end{instance}


\begin{proposition}
    设$X,Y$是拓扑空间，同伦是$X$到$Y$的连续映射之间的等价关系.
\end{proposition}

\begin{proof}
    (1) 自反性：$f\simeq f$. 取$\setjot[-6]\begin{aligned}[t]
        H:X\times [0,1]&\to Y \\
        (x,t) &\mapsto f(x)
    \end{aligned}$.\par
    (2) 对称性：$f\simeq g \Rightarrow g\simeq f$.假设$f\simeq g$通过$H:X\times [0,1]\to Y$.\par
    令$\setjot[-6]\begin{aligned}[t]
        \overline{H}:X\times [0,1]&\to Y \\
        (x,t) &\mapsto H(x,1-t)
    \end{aligned}$.
    则$\setjot[-6]\begin{gathered}[t]
        \overline{H}(x,0)=H(x,1)=g(x) \\
        \overline{H}(x,1)=H(x,0)=f(x)
    \end{gathered}$. 所以$\overline{H}:g\simeq f$.
    
    (3) 传递性：$f\simeq g, g\simeq h \Rightarrow f\simeq h$.
    假设$H_1:f\simeq g, H_2:g\simeq h$.
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.6\textwidth]{fig/10.2-1.png}
        \includegraphics[width=0.3\textwidth]{fig/10.2-2.png}
        \captionof{figure}{同伦的传递性}
    \end{figure}
    
    定义$\setjot[-2]\begin{aligned}[t]
        H:X\times [0,1]&\to Y \\
        (x,t)&\mapsto \begin{cases}
            H_1(x,2t) & 0\leq t\leq \frac{1}{2} \\
            H_2(x,2t-1) & \frac{1}{2}\leq t\leq 1
        \end{cases}
    \end{aligned}$\par
    由粘合引理，$H$是连续的，且$H(x,0)=f(x), H(x,1)=h(x)$. 所以$f\simeq h$.
\end{proof}

\begin{definition}{零同伦}
    如果$f:X\to Y$同伦于常值映射，则称$f$是\textbf{零伦的}.
\end{definition}

\begin{instance}
    $f:X\to \mathbb{E}^n$是零伦的. 事实上，任何映入$\mathbb{E}^n$的凸子空间的映射都是零同伦的.
\end{instance}

\begin{proposition}
    设$f_0\simeq f_1:X\to Y, g_0\simeq g_1:Y\to Z$.
    则$g_0\circ f_0 \simeq g_1\circ f_1:X\to Z$.
\end{proposition}

\begin{proof}
    假设$H_1:f_0\simeq f_1, H_1:X\times [0,1]\to Y$.$H_2:g_0\simeq g_1, H_2:Y\times [0,1]\to Z$.\par
    令$\setjot[-2]\begin{aligned}[t]
        \widetilde{H_1}:X\times [0,1]&\to Y\times [0,1] \\
        (x,t)&\mapsto (H_1(x,t),t)
    \end{aligned}$，则$\widetilde{H_1}$是连续的.\par
    令$H=H_2\circ \widetilde{H_1}:X\times [0,1]\to Z$.
    那么就有$X\times [0,1] \xrightarrow{\widetilde{H_1}} Y\times [0,1] \xrightarrow{H_2} Z$.\par
    $H(x,0)=H_2\circ\widetilde{H_1}(x,0)=H_2(H_1(x,0),0)=H_2(f_0(x),0)=g_0(f_0(x))=g_0\circ f_0(x)$.\par
    $H(x,1)=H_2\circ\widetilde{H_1}(x,1)=H_2(H_1(x,1),1)=H_2(f_1(x),1)=g_1(f_1(x))=g_1\circ f_1(x)$.\par
    所以$g_0\circ f_0 \simeq g_1\circ f_1$.
\end{proof}

\section{相对同伦}

\begin{definition}{相对同伦}
    设$f,g:X\to Y, A\subset X$. 若$f,g$通过$H:X\times [0,1]\to Y$同伦，且$\forall\, a\in A, H(a,t)=f(a)=g(a)$，则称$f$和$g$是\textbf{相对于$A$同伦的}，记作$f\simeq g \text{ rel } A$.
\end{definition}
从定义中可以看出，相对同伦是对同伦概念的加强，它不仅要求普通的同伦，还要求映射在子集$A$上保持不变，即在同伦变换过程中，$A$上的点始终映射到相同的位置.特别地，如果$A=\varnothing$，则相对同伦就是同伦.\par

\begin{remark}
    $\forall\,x\in X, a_x=\setjot[-6]\begin{aligned}[t]
        H(x,\cdot):[0,1] &\to Y \\
        t &\mapsto H(x,t)
    \end{aligned}$是$Y$中的一条道路.\par
    $\forall\,p\in A, a_p=\setjot[-6]\begin{aligned}[t]
        H(p,\cdot):[0,1] &\to Y \\
        t &\mapsto H(p,t)=f(p)=g(p)
    \end{aligned}$是$Y$中的常值道路.
\end{remark}

\begin{instance}
    设$f,g:X\to \mathbb{E}^n, A=\{x\in X\mid f(x)=g(x)\}$.
    构造$\setjot[-2]\begin{aligned}[t]
        H:X\times [0,1]&\to \mathbb{E}^n \\
        (x,t)&\mapsto (1-t)f(x)+tg(x)
    \end{aligned}$\par
    则$\forall\,  a\in A, H(a,t)=(1-t)f(a)+tg(a)=f(a)=g(a)$.
    所以$H:f\simeq g \text{ rel } A$.
\end{instance}

\section{道路同伦}

\begin{definition}{道路同伦}
    设$X$是拓扑空间，$a,b:[0,1]\to X$是两条道路.
    如果$a\simeq b \text{ rel } \{0,1\}$，则称$a$和$b$是\textbf{道路同伦的(定端同伦的)}，记作$a\simeqdot b$.
\end{definition}

具体来说，存在$H:[0,1]\times [0,1]\to X$满足：$H(s,0)=a(s), H(s,1)=b(s)$，$H(0,t)=a(0)=b(0), H(1,t)=a(1)=b(1)$.\par

\begin{definition}{道路类}
    道路$a$的\textbf{道路类}定义为$[a]=\{b\mid b\simeqdot a\}$. 如果$a$是一条闭路，则$[a]$称为\textbf{闭路类}，且$a(0)$(或$a(1)$)称为该闭路类的\textbf{基点}.
\end{definition}

\begin{lemma}
    $a\simeqdot b \implies \overline{a}\simeqdot \overline{b}$.
\end{lemma}

\begin{proof}
    假设$H:[0,1]\times [0,1]\to X$实现了$a\simeqdot b$.\par
    令$\setjot[-6]\begin{aligned}[t]
        \overline{H}:[0,1]\times [0,1]&\to X \\
        (s,t)&\mapsto H(1-s,t)
    \end{aligned}$，则$\overline{H}:\overline{a}\simeqdot \overline{b}$.
\end{proof}

\begin{definition}{逆道路类}
    定义道路类$[a]$的\textbf{逆}为$[a]^{-1}:=[ \overline{a} ]$.
\end{definition}

\begin{lemma}
    设$a\simeqdot c, b\simeqdot d$，且$a(1)=b(0)$(从而$c(1)=d(0)$)，则$ab\simeqdot cd$.
\end{lemma}

\begin{proof}
    \begin{figure}[H]
        \centering
        \vspace{-2.8 em}
        \includegraphics[width=0.5\textwidth]{fig/10.8.png}
        \captionof{figure}{}
    \end{figure}
    \vspace{-1 em}
    设$H_1:a\simeqdot c, H_2:b\simeqdot d$.
    定义$\setjot[-2]\begin{aligned}[t]
        H:[0,1]\times [0,1]&\to X \\
        (s,t)&\mapsto \begin{cases}
            H_1(2s,t), & 0\leq s\leq \frac{1}{2} \\
            H_2(2s-1,t), & \frac{1}{2}\leq s\leq 1
        \end{cases}
    \end{aligned}$\par
    验证边界条件：
    $H(s,0)=\begin{cases}
        H_1(2s,0)=a(2s) &0\leq s\leq\frac{1}{2} \\
        H_2(2s-1,0)=b(2s-1) &\frac{1}{2}\leq s\leq 1
    \end{cases}=(ab)(s)$\par
    类似的，$H(s,1)=(cd)(s)$，且$H(0,t)=a(0)=c(0)$，$H(1,t)=b(1)=d(1)$.\par
    由粘合引理，$H$是连续的. 所以$ab\simeqdot cd$.
\end{proof}

\begin{definition}{道路类的乘积}
    设$[a],[b]$是两个道路类，且$a(1)=b(0)$，定义它们的\textbf{乘积}为$[a]\cdot [b]:=[ab]$.
\end{definition}

\begin{lemma}[运算的结合律]
    设$a,b,c:[0,1]\to X$是道路，且$a(1)=b(0), b(1)=c(0)$，则$\left([a]\cdot [b]\right)\cdot [c] = [a]\cdot \left([b]\cdot [c]\right)$.
    特别地，$\pi_1(X,x_0)$中的乘法满足\textbf{结合律}.
\end{lemma}

\begin{proof}
    $([a]\cdot [b])\cdot [c] = [ab]\cdot [c] = [(ab)c]$.
    $[a]\cdot ([b]\cdot [c]) = [a]\cdot [bc] = [a(bc)]$.
    只需证明$(ab)c \simeqdot a(bc)$.
    则\[((ab)c)(s)=\begin{cases}

            b(4s-1), & \frac{1}{4}\leq s\leq \frac{1}{2} \\[5pt]
            c(2s-1), & \frac{1}{2}\leq s\leq 1
        \end{cases}\quad\quad
        (a(bc))(s) = \begin{cases}
            a(2s), & 0\leq s\leq \frac{1}{2} \\[5pt]
            b(4s-2), & \frac{1}{2}\leq s\leq \frac{3}{4} \\[5pt]
            c(4s-3), & \frac{3}{4}\leq s\leq 1
        \end{cases}\]\par

        \captionof{figure}{}
    \end{figure}

        \includegraphics[width=0.28\textwidth]{fig/10.11.png}

        \vspace{-1 em}
    \end{wrapfigure}
    \vspace{-1 em}
    令$\setjot[-2]\begin{aligned}[t]

\begin{theorem}
    $\pi_1(X,x_0)$是一个群，称为$X$的以$x_0$为基点的\textbf{基本群}.

    $\pi_1(\mathbb{E}^n, x_0) = \{1\}$，平凡群.
    因为对于任意闭路$a$，都有$a\simeqdot c_{x_0}$(通过直线同伦).
\end{instance}

    如果$X$是$\mathbb{E}^n$的凸子空间，则$\pi_1(X,x_0)=\{1\}$.
\end{instance}

\section{诱导同态}

\begin{definition}{同态}
    设$G_1, G_2$是两个群，若$\varphi:G_1\to G_2$满足$\varphi(gh)=\varphi(g)\varphi(h)$，则称$\varphi$是\textbf{同态}.
\end{definition}
因此，同态保持了群的运算结构.


\begin{instance}
    $\setjot[-8]\begin{aligned}[t]
        \varphi:\mathbb{Z}&\to \mathbb{Z} \\
        z &\mapsto 2z
    \end{aligned}$，$\setjot[-8]\begin{aligned}[t]
        \varphi:\mathbb{Z}&\to \mathbb{Z} \\
        z &\mapsto z+1
    \end{aligned}$，$\setjot[-8]\begin{aligned}[t]
        \varphi_2:\mathbb{Z}&\to \mathbb{Z} \\
        z&\mapsto 0
    \end{aligned}$，上述三个均为同态，其中第三个称为\bluebf{平凡同态}
\end{instance}

设$X,Y$是两个空间，$f:X\to Y$是连续映射.
$a:[0,1]\to X$是$X$中的道路，则$f\circ a:[0,1]\to Y$是$Y$中的道路.

\begin{lemma}
    若在$X$中$a\simeqdot b$，则在$Y$中$f\circ a \simeqdot f\circ b$.
\end{lemma}

\begin{proof}
    假设$H:a\simeqdot b$.
    $H:[0,1]\times [0,1]\to X$.
    \begin{figure}[H]
        \vspace{-0.75 em}
        \centering
        \includegraphics[width=0.8\textwidth]{fig/10.16.png}
        \captionof{figure}{}
    \end{figure}
    \vspace{-0.5 em}
    $f\circ H:[0,1]\times [0,1]\to Y$.则$f\circ H: f\circ a \simeqdot f\circ b$.
\end{proof}

\begin{definition}
    $f:X\to Y$诱导了一个映射$\setjot[-2]\begin{aligned}[t]
            f_*:\pi_1(X,x_0) &\to \pi_1(Y,f(x_0)) \\
            [a] &\mapsto [f\circ a]
        \end{aligned}$.
\end{definition}

\begin{lemma}
    $f_*$是一个同态.
\end{lemma}

\begin{proof}
    设$[a],[b]\in \pi_1(X,x_0)$.$\setjot[-2]\begin{aligned}[t]
        f_*([a]\cdot [b]) &= f_*([a\cdot b]) = [f\circ (a\cdot b)]=[(f\circ a)\cdot (f\circ b)] \\
        & = [f\circ a]\cdot [f\circ b] = f_*[a]\cdot f_*[b].
    \end{aligned}$
    \begin{figure}[H]
        \vspace{-0.75 em}
        \centering
        \includegraphics[width=0.58\textwidth]{fig/10.17.png}
        \captionof{figure}{证明图例}
    \end{figure}
\end{proof}

\begin{instance}
    $\setjot[-8]\begin{aligned}[t]
        f:X&\to Y \\
        x &\mapsto y_0
    \end{aligned}$是常值映射，则$\setjot[-4]\begin{aligned}[t]
            f_*:\pi_1(X,x_0) &\to \pi_1(Y,y_0) \\
            [a] &\mapsto [f\circ a] = [c_{y_0}]
        \end{aligned}$是平凡同态.
\end{instance}

\begin{instance}
    $\setjot[-8]\begin{aligned}[t]
        \mathrm{id}:X&\to X \\
        x &\mapsto x
    \end{aligned}$，则$\setjot[-4]\begin{aligned}[t]

            [a] &\mapsto [a]
        \end{aligned}$是恒等同构.
\end{instance}

\begin{lemma}
    设$f:X\to Y, g:Y\to Z, x_0\in X$.
    则$(g\circ f)_* = g_*\circ f_*: \pi_1(X,x_0)\to \pi_1(Z,g(f(x_0)))$.
\end{lemma}

\begin{proof}
    $\forall\,[a]\in \pi_1(X,x_0)$，$(g\circ f)_*([a]) = [g\circ f\circ a]= g_*[f\circ a] = g_*(f_*[a]) = g_*\circ f_*[a]$.
\end{proof}

\begin{theorem}
    设$f:X\to Y$是同胚，$x_0\in X$.
    则$f_*:\pi_1(X,x_0)\to \pi_1(Y,f(x_0))$是一个同构.
\end{theorem}

\begin{proof}
    设$f^{-1}:Y\to X$是$f$的逆映射.\par
    则$f^{-1}\circ f = \mathrm{id}_X: X\to X$. $f\circ f^{-1} = \mathrm{id}_Y: Y\to Y$，那么
    \[
        (f^{-1})_*\circ f_* = (f^{-1}\circ f)_* = (\mathrm{id}_X)_*: \pi_1(X,x_0)\to \pi_1(X,x_0)
    \]

        \end{tikzcd}
    \end{figure}
    复合映射是恒等映射，则$f_*$是单射，$(f^{-1})_*$是满射.
    \[

\section{圆周的基本群 $\pi_1(S^1)$}
这里的圆周定义为复平面上的单位圆周$S^1 := \{z \in \mathbb{C} \mid |z|=1\}$.\par
其中的元素可以写成$z = e^{i2\pi t} = \cos 2\pi t + i \sin 2\pi t,0 \leq t \leq 1$.\par
\begin{wrapfigure}{r}{0.25\textwidth}
    \vspace{-6 em}

    \vspace{-4 em}

它可以被视作是一维欧式空间的商空间，其商映射为：
{
    \setjot[-6]
    \begin{align*}
    p: \mathbb{E}^1 &\to S^1 \\
    t &\mapsto e^{i2\pi t}
    \end{align*}
}\par
(i) $p^{-1}(1) = \mathbb{Z}$\par
(ii)$p|_{\left(-\frac{1}{2}, \frac{1}{2}\right)}: \left(-\frac{1}{2}, \frac{1}{2}\right) \to S^1 - \{-1\}$ 是双射，连续且开的，所以它是一个同胚.\par
令 $\psi = \left(p|_{\left(-\frac{1}{2}, \frac{1}{2}\right)}\right)^{-1}: S^1 - \{-1\} \to \left(-\frac{1}{2}, \frac{1}{2}\right)$.

\begin{lemma}[道路提升引理]
    如果 $a$ 是 $S^1$ 中以 $1 \in S^1$ 为起点的道路，则存在 $\mathbb{E}^1$ 中\uline{唯一的}道路 $\tilde{a}$ 以 $0 \in \mathbb{E}^1$ 为起点，使得 $p \circ \tilde{a} = a$.
\end{lemma}

    \caption{两个引理的交换图}

\begin{lemma}[同伦提升引理]
    如果 $b$ 也是 $[0,1]\to S^1$ 中的道路，使得 $H: a \simeq b$，则存在\uline{唯一的} $\tilde{H}: [0,1] \times [0,1] \to \mathbb{E}^1$，使得 $p \circ \tilde{H} = H$，且 $\tilde{H}: \tilde{a} \simeq \tilde{b}$，其中 $\tilde{b}$ 是 $b$ 的提升.
\end{lemma}

\begin{proof}
    设 $X$ 是 $[0,1]$ 或 $[0,1] \times [0,1]$.设 $f: X \to S^1$ 是 $a: [0,1] \to S^1$ 或 $H: [0,1] \times [0,1] \to S^1$.\par
    设 $0 \in X$ 是 $0 \in [0,1]$ 或 $(0,0) \in [0,1] \times [0,1]$.\par
    \begin{claim}{$\exists\,\lambda > 0$，使得对于$\forall\, x, x' \in X$满足$|x - x'| < \lambda \implies f(x) + f(x') \neq 0$，.}
        令$U_1 = \{x+iy \in S^1 \mid x > 0\}$，$U_2 = \{x+iy \in S^1 \mid x < 0\}$，$U_3 = \{x+iy \in S^1 \mid y > 0\}$，$U_4 = \{x+iy \in S^1 \mid y < 0\}$.\par
        则$\{U_1, U_2, U_3, U_4\}$ 是 $S^1$ 的开覆盖，故$\{f^{-1}(U_1), f^{-1}(U_2), f^{-1}(U_3), f^{-1}(U_4)\}$ 是 $X$ 的开覆盖.\par
        $X$ 是紧致度量空间，由Lebesgue引理，存在 $\lambda > 0$，使得对于任意 $x \in X$，存在某个 $i=1,2,3,4$，使得 $B(x, \lambda) \subset f^{-1}(U_i)$.\par
        所以 $|x - x'| < \lambda \implies$ 存在某个 $f^{-1}(U_i)$ 包含 $x$ 和 $x'\implies$ 存在某个 $U_i$ 同时包含 $f(x)$ 和 $f(x')$.\par
        因为 $U_i$ 中任意两点都不是对径点，所以 $f(x) + f(x') \neq 0$.
    \end{claim}\par
    由断言1可知$f(x) \neq -f(x')\implies \frac{f(x)}{f(x')}\neq -1 \in S^1$，因此$\psi\left(\frac{f(x)}{f(x')}\right) \in \left(-\frac{1}{2}, \frac{1}{2}\right)$.\par
    我们可以找到足够大的 $N$，使得对于所有 $x \in X$，有 $|x| = |x-0| < N \cdot \lambda$.\par
    定义$\setjot[0]\begin{aligned}[t]
        \tilde{f}: X &\to \mathbb{E}^1 \\
        x&\mapsto\psi\left(\vcenter{\hbox{$\dfrac{f(x)}{f\left(\dfrac{N-1}{N}x\right)}$}}\right) + \psi\left(\vcenter{\hbox{$\dfrac{f\left(\dfrac{N-1}{N}x\right)}{f\left(\dfrac{N-2}{N}x\right)}$}}\right) + \dots + \psi\left(\vcenter{\hbox{$\dfrac{f\left(\dfrac{x}{N}\right)}{f(0)}$}}\right) \\
    \end{aligned}$\par
    则$\begin{aligned}[t]
        p \circ \tilde{f}(x) &= p \circ \psi\left(\vcenter{\hbox{$\dfrac{f(x)}{f\left(\dfrac{N-1}{N}x\right)}$}}\right) \cdot p \circ \psi\left(\vcenter{\hbox{$\dfrac{f\left(\dfrac{N-1}{N}x\right)}{f\left(\dfrac{N-2}{N}x\right)}$}}\right) \cdot \dots \cdot p \circ \psi\left(\vcenter{\hbox{$\dfrac{f\left(\dfrac{x}{N}\right)}{f(0)}$}}\right) \\
        &= \dfrac{f(x)}{f\left(\dfrac{N-1}{N}x\right)} \cdot \dfrac{f\left(\dfrac{N-1}{N}x\right)}{f\left(\dfrac{N-2}{N}x\right)} \cdot \dots \cdot \dfrac{f\left(\dfrac{x}{N}\right)}{f(0)} \\
        &= f(x) \quad (\text{注意：} f(0)=1)
    \end{aligned}$\par
    存在性得证.\par
    如果还存在$\tilde{f}_1: X \to \mathbb{E}^1$，使得$\tilde{f}_1(0) = 0 \in \mathbb{E}^1$，且 $p \circ \tilde{f}_1 = f$.\par
    那么 $p\left(\tilde{f}(x) - \tilde{f}_1(x)\right) = \frac{p\left(\tilde{f}(x)\right)}{p\left(\tilde{f}_1(x)\right)} =\frac{f(x)}{f(x)} = 1 \in S^1$.所以 $\tilde{f}(x) - \tilde{f}_1(x) \in \mathbb{Z}$.\par
    因为 $X$ 是连通的，所以$\tilde{f} - \tilde{f}_1$的像连通，故只能是常数.\par
    所以 $\tilde{f}(x) - \tilde{f}_1(x) = \tilde{f}(0) - \tilde{f}_1(0) = 0 \in \mathbb{E}^1$.因此 $\tilde{f}_1(x) = \tilde{f}(x)$，$\forall x \in X$.\par
    唯一性得证.
    
    在 $X = [0,1] \times [0,1]$ 的情况下，$f=H$，$\tilde{f}=\tilde{H}$.$\tilde{H}(0,0) = 0 \in \mathbb{E}^1$，且 $p \circ \tilde{H} = H: [0,1]^2 \to S^1$.\par
    $H(s,0) = a(s)$，$H(s,1) = b(s)$，$H(0,t) = a(0) = b(0)$，$H(1,t) = a(1) = b(1)$.\par
        \begin{figure}[H]
        \centering
        \includegraphics[width=0.55\textwidth]{fig/10.25.png}
        \caption{同伦提升证明示意图}
    \end{figure}
    (1) $p \circ \tilde{H}(s,0) = H(s,0) = a(s)$，$\tilde{H}(0,0) = 0 \in \mathbb{E}^1$，所以 $\tilde{H}(s,0) = \tilde{a}(s)$.\par
    (2) $p \circ \tilde{H}(0,t) = H(0,t) = 1 \in S^1$，所以 $\tilde{H}(0,t) \in \mathbb{Z}$.\par
    因为 $\{0\} \times [0,1]$ 是连通的，$\tilde{H}(0, \cdot): [0,1] \to \mathbb{Z}$的像也是连通的，因此为整数.\par
    故$\tilde{H}(0,t) = \tilde{H}(0,0) = 0 \in \mathbb{E}^1$.\par
    (3) ... 类似于 (1)     (4) ... 类似于 (2)\par
    因此 $\tilde{H}: \tilde{a} \simeqdot \tilde{b}$.
\end{proof}

\begin{corollary}
    $\tilde{a}$ 的终点只依赖于 $a$ 的道路类，即 $a \simeqdot b$，$\tilde{a},\tilde{b}$分别为$a,b$的提升，且$\tilde{a}(0)=\tilde{b}(0)=0 \in \mathbb{E}^1$，则$\tilde{a} \simeqdot \tilde{b}$.
\end{corollary}

\section{群论复习}

\begin{definition}{核和像}
    对于同态$\Phi: G \to G_1$，定义$\mathrm{Ker}\,\Phi = \{g \in G \mid \Phi(g) = e_1\}$，称为$\Phi$的\textbf{核}，其中$e_1$是$G_1$的单位元.\par
    定义$\mathrm{Im}\,\Phi = \{h \in G_1 \mid \exists\,g \in G, \text{s.t. } h = \Phi(g)\}$，称为$\Phi$的\textbf{像}.
\end{definition}

除了无限循环群之外，还有有限循环群.\par
一个$n$阶循环群定义为$\mb{Z}_n=\mb{Z}/n\mb{Z}=\{\bar{0}, \bar{1}, \dots, \overline{n-1}\}$，其中的元素$\bar{q},q\in\{0,1,\cdots,n-1\}$表示除以$n$余$q$的等价类.\par
循环群中的二元运算定义为$\forall\,\overline{q_1}+\overline{q_2}=\overline{q_2+q_1}$，即为代表元相加所在的等价类.其中的单位元为$\bar{1}$，$\bar{1}$被称为\bluebf{生成元}，因为循环群中的所有元素都可以有生成元不断相加得到.\par
一个例子是二阶循环群$\mb{Z}_2=\mathbb{Z}/2\mathbb{Z} = \{\bar{0}, \bar{1}\}$，它的两个等价类分别代表了奇数和偶数.\par
下面介绍群之间的直积运算.\par
\begin{definition}{直积}
    设$G_1,G_2$是两个群，定义$G_1 \times G_2 = \{(a,b) \mid a \in G_1, b \in G_2\}$，并定义二元运算$\cdot$为：
    \[ (a_1, b_1) \cdot (a_2, b_2) = (a_1 a_2, b_1 b_2),\text{其中}\,a_1,a_2\in G_1, b_1,b_2 \in G_2 \]
    则$G_1\times G_2$称为$G_1$和$G_2$的\textbf{直积}，其中的单位元为$(e_1,e_2)$，$e_1,e_2$分别为$G_1,G_2$的单位元.
\end{definition}
例如：$\mathbb{Z} \times \mathbb{Z} = \{(n, m) \in \mathbb{Z} \times \mathbb{Z}\}$
$(n_1, m_1) + (n_2, m_2) = (n_1+n_2, m_1+m_2)$
$(\mathbb{Z}, +)$：单位元 $(0,0) \in \mathbb{Z} \times \mathbb{Z}$.

\section{积空间的基本群}

\begin{theorem}
    设$X, Y$是两个拓扑空间.$x_0 \in X, y_0 \in Y$.
    则 $\pi_1(X \times Y, (x_0, y_0)) \cong \pi_1(X, x_0) \times \pi_1(Y, y_0)$.
\end{theorem}

\begin{proof}
    设 $p_1: X \times Y \to X, p_2: X \times Y \to Y$ 为投影.它们诱导
    \[ (p_1)_*: \pi_1(X \times Y, (x_0, y_0)) \to \pi_1(X, x_0) \]
    \[ (p_2)_*: \pi_1(X \times Y, (x_0, y_0)) \to \pi_1(Y, y_0) \]
    
    定义 $\setjot[-4]\begin{aligned}[t]
        \Phi: \pi_1(X \times Y, (x_0, y_0)) &\to \pi_1(X, x_0) \times \pi_1(Y, y_0) \\
        [a] &\mapsto \left((p_1)_*[a], (p_2)_*[a]\right)
    \end{aligned} $\par
    则 $\Phi$ 是一个同态.\par
    (1) $\Phi$ 是满射.\par
    $\forall\,([b], [c]) \in \pi_1(X, x_0) \times \pi_1(Y, y_0)$.
    设 $\setjot[-5]\begin{aligned}[t]
        a: [0,1] &\to X \times Y  \\
        s &\mapsto (b(s), c(s))
    \end{aligned}$.\par
    则 $[a] \in \pi_1(X \times Y, (x_0, y_0))$，且$\Phi([a])= \left((p_1)_*[a], (p_2)_*[a]\right)\left([p_1 \circ a], [p_2 \circ a]\right) = ([b], [c])$.\par
    (2) $\Phi$ 是单射.\par
    设 $\Phi([a]) = \left([c_{x_0}], [c_{y_0}]\right)$.\par
    则 $(p_1)_*[a] = [c_{x_0}], (p_2)_*[a] = [c_{y_0}]$.$p_1 \circ a \simeq c_{x_0}, p_2 \circ a \simeq c_{y_0}$.\par
    设 $H_1: [0,1] \times [0,1] \to X$ 和 $H_2: [0,1] \times [0,1] \to Y$为同伦.\par
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.5\textwidth]{fig/10.29.png}
        \captionof{figure}{}
    \end{figure}
    令 $\setjot[-6]\begin{aligned}[t]
        H: [0,1] \times [0,1] &\to X \times Y \\
        (s, t) &\mapsto \left(H_1(s, t), H_2(s, t)\right)
    \end{aligned} $\par
    则 $H(s, 0) = \left(H_1(s, 0), H_2(s, 0)\right) = \left(p_1 \circ a(s), p_2 \circ a(s)\right) = a(s)$.\par
    \begin{wrapfigure}{r}{0.3\textwidth}
        \vspace{-5 em}
        \includegraphics[width=0.3\textwidth]{fig/10.30.png}
        \captionof{figure}{}
        \vspace{-1 em}
    \end{wrapfigure}
    $H(s, 1) = \left(H_1(s, 1), H_2(s, 1)\right) = (x_0, y_0)$.\par
    $H(0, t) = \left(H_1(0, t), H_2(0, t)\right) = (x_0, y_0)$.\par
    $H(1, t) = \left(H_1(1, t), H_2(1, t)\right) = (x_0, y_0)$.\par
    所以 $a \simeq c_{(x_0, y_0)}$，即 $[a] = 1 \in \pi_1(X \times Y, (x_0, y_0))$.
\end{proof}

\begin{instance}
    $\pi_1(S^1 \times S^1, (x_0, y_0)) \cong \pi_1(S^1, x_0) \times \pi_1(S^1, y_0) \cong \mathbb{Z} \times \mathbb{Z}$.
\end{instance}

\begin{instance}
    $\pi_1(S^1 \times D^n, (x_0, y_0)) \cong \pi_1(S^1, x_0) \times \pi_1(D^n, y_0) \cong \mathbb{Z} \times \{1\} = \mathbb{Z}$.
\end{instance}

\section{同伦等价}

\begin{definition}{同伦等价}
    设$X, Y$是两个拓扑空间，$f: X \to Y, g: Y \to X$是连续映射，如果$g \circ f \simeq \mathrm{id}_X: X \to X$且$f \circ g \simeq \mathrm{id}_Y: Y \to Y$，则称$X$和$Y$是\textbf{同伦等价的}，记作$X \simeq Y$.
\end{definition}

\begin{remark}
    $X \cong Y \Rightarrow X \simeq Y$.\par
    若$f: X \cong Y$，则存在$f^{-1}: Y \to X$使得$f^{-1} \circ f = \mathrm{id}_X, f \circ f^{-1} = \mathrm{id}_Y$.
\end{remark}

\begin{instance}
    $X \times [0,1] \simeq X$\par
    定义$\setjot[-4]\begin{aligned}[t]
        f: X \times [0,1] &\to X \\
        (x, s) &\mapsto x
    \end{aligned}$，$\setjot[-4]\begin{aligned}[t]
         g: X &\to X \times [0,1]\\
         x &\mapsto (x, 0)
    \end{aligned}$\par
    则$\setjot[-4]\begin{aligned}[t]
        f: X \times [0,1] &\to X \\
        (x, s) &\mapsto x
    \end{aligned}$，即$f \circ g = \mathrm{id}_X$，而$\setjot[-4]\begin{aligned}[t]
        g \circ f: X \times [0,1] &\to X \times [0,1] \\
        (x, s) &\mapsto (x, 0)
    \end{aligned}$.\par
    构造同伦$\setjot[-2]\begin{aligned}[t]
        H: X \times [0,1] \times [0,1] &\to X \times [0,1] \\
        (x, s, t) &\mapsto (x, ts)
    \end{aligned} $.\par
    则$H(x, s, 0) = (x, 0) = g \circ f(x, s)$，$H(x, s, 1) = (x, s) = \mathrm{id}_{X \times [0,1]}(x, s)$.
    所以$g \circ f \simeq id_{X \times [0,1]}$.\par
    类似地，进一步可以证明：$X \times \mathbb{E}^1 \simeq X$.
\end{instance}

\begin{proposition}
    同伦等价是拓扑空间之间的等价关系.
\end{proposition}

\begin{proof}
    需要验证以下三条性质：
    (1) $X \simeq X$.
    (2) $X \simeq Y \Rightarrow Y \simeq X$.
    (3) $X \simeq Y, Y \simeq Z \Rightarrow X \simeq Z$.\par
    前两条显然，下面证明第三条性质.设$X\simeq Y,Y\simeq Z$，则 \par
    \hspace{-2em}
    \begin{minipage}[t]{0.49\textwidth}
        \begin{center}
            \begin{tikzcd}[labels={font=\normalsize}]
            X \arrow[rr, "f_1", bend left] &  & Y \arrow[ll, "g_1", bend left]
            \end{tikzcd}
        \end{center}
        \vspace{-1 em}
        \begin{gather*}
            g_1 \circ f_1 \simeq \mathrm{id}_X \\[-5pt]
            f_1 \circ g_1 \simeq \mathrm{id}_Y
        \end{gather*}
    \end{minipage}
    \vspace{-1 em}
    \begin{minipage}[t]{0.49\textwidth}
        \begin{center}
            \begin{tikzcd}[labels={font=\normalsize}]
            Y \arrow[rr, "f_2", bend left] &  & Z \arrow[ll, "g_2", bend left]
            \end{tikzcd}
        \end{center}
        \vspace{-1 em}
        \begin{gather*}
            g_2\circ f_2 \simeq \mathrm{id}_Y \\[-5pt]
            f_2 \circ g_2 \simeq \mathrm{id}_Z
        \end{gather*}
    \end{minipage}\par
    \begin{center}
        $\begin{aligned}
        f_2 \circ f_2:X\to Z\\[-5pt]
        g_1 \circ g_2:Z\to X 
    \end{aligned}$
    \hspace{10em}
    \begin{tikzcd}[labels={font=\normalsize}]
        X \arrow[rr, "f_2\circ f_1", bend left] &  & Z \arrow[ll, "g_1\circ g_2", bend left]
    \end{tikzcd}
    \end{center}\par
    \vspace{-1 em}
    那么就有：$\setjot[-4]\begin{aligned}[t]
        &(g_1 \circ g_2) \circ (f_2 \circ f_1)= g_1 \circ (g_2 \circ f_2) \circ f_1 \simeq g_1 \circ id_Y \circ f_1 = g_1 \circ f_1 \simeq \mathrm{id}_X \\
        &(f_2 \circ f_1) \circ (g_1 \circ g_2)= f_2 \circ (f_1 \circ g_1) \circ g_2 \simeq f_2 \circ id_Y \circ g_2 = f_2 \circ g_2 \simeq \mathrm{id}_Z
    \end{aligned}$\par
    所以$X \simeq Z$.
\end{proof}

\begin{instance}
    $\mathbb{E}^n \simeq \{pt\}, n \geq 1$.\par
    设$\setjot[-6]\begin{aligned}[t]
        f: \mathbb{E}^n &\to \{pt\} \\
        x &\mapsto pt
    \end{aligned} $，$\setjot[-6]\begin{aligned}[t]
        g: \{pt\} &\to \mathbb{E}^n\\
        pt &\mapsto 0
    \end{aligned}$.
    则$\setjot[-6]\begin{aligned}[t]
         g \circ f: \mathbb{E}^n &\to \mathbb{E}^n\\
         x &\mapsto 0
    \end{aligned}$，$\setjot[-6]\begin{aligned}[t]
            f \circ g: \{pt\} &\to \{pt\} \\
            pt &\mapsto pt
    \end{aligned}$.\par
    显然$g \circ f \simeq \mathrm{id}_{\mathbb{E}^n}$ (通过直线同伦)，$f \circ g = \mathrm{id}_{\{pt\}}$.
\end{instance}

\begin{definition}{可缩}
    如果一个空间$X \simeq \{pt\}$，则称$X$是\textbf{可缩的}.
\end{definition}

\section{同伦等价空间的基本群}

设$X_0, X_1$是两个空间，$h \simeq k: X_0 \to X_1$.
取$x_0 \in X_0$.
考虑诱导同态$h_*: \pi_1(X_0, x_0) \to \pi_1(X_1, h(x_0))$和$k_*: \pi_1(X_0, x_0) \to \pi_1(X_1, k(x_0))$.
$h_*$和$k_*$有什么关系？

假设$H: X_0 \times [0,1] \to X_1$实现了$h \simeq k$.
存在一条从$h(x_0)$到$k(x_0)$道路$\setjot[-6]\begin{aligned}[t]
    b: [0,1] &\to X_1 \\
    t &\mapsto H(x_0, t)
\end{aligned}$.\par
令$\omega = [b]$.

\begin{proposition}{}\label{prop:10.5}
    $k_* = \omega_\# \circ h_*$.
    即\begin{tikzcd}[labels={font=\normalsize}]
                                                              &  & {\pi_1(X_1,h(x_0))} \arrow[dd, "\omega_\#"] \\
{[a]\in\pi_1(X_0,x_0)} \arrow[rru, "h_*"] \arrow[rrd, "k_*"'] &  &                                             \\
                                                              &  & {\pi_1(X_1,k(x_0))}                         
        \end{tikzcd}
\end{proposition}

\begin{proof}
    对于任意$[a] \in \pi_1(X_0, x_0)$，我们要证明
    $$ k_*[a] = \omega_\# \circ h_*[a] = \omega^{-1} \cdot h_*[a] \cdot \omega $$
    即 $\omega \cdot k_*[a] = h_*[a] \cdot \omega$.
    这只需证明 $b \cdot (k \circ a) \simeq (h \circ a) \cdot b$.
    
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.7\textwidth]{fig/10.31.png}
        \captionof{figure}{}
    \end{figure}
    
    考虑$X_0 \times [0,1]$上的正方形区域$[0,1] \times [0,1]$.
    令$\beta_0(s) = (s, 0), \beta_1(s) = (s, 1)$，$\gamma_0(t) = (0, t), \gamma_1(t) = (1, t)$.\par
    定义$F(s, t) = (a(s), t) \in X_0 \times [0,1]$.\par
    则$F \circ \beta_0(s) = (a(s), 0), F \circ \beta_1(s) = (a(s), 1)$.$F \circ \gamma_0(t)=F \circ \gamma_1(t) = (a(0), t) = (x_0, t)$.\par
    考虑映射$H \circ F: [0,1] \times [0,1] \to X_1$.\par
    则$H \circ F \circ \gamma_0(t) = H(x_0, t) = b(t)$.
    $H \circ F \circ \gamma_1(t) = H(x_0, t) = b(t)$.
    $H \circ F \circ \beta_0(s) = H(a(s), 0) = h(a(s))$.
    $H \circ F \circ \beta_1(s) = H(a(s), 1) = k(a(s))$.\par
    显然，在正方形中我们有$G: \gamma_0 \cdot \beta_1 \simeq \beta_0 \cdot \gamma_1$.\par
    于是$H\circ F\circ G=(H \circ F \circ \gamma_0) \cdot (H \circ F \circ \beta_1) \simeq (H \circ F \circ \beta_0) \cdot (H \circ F \circ \gamma_1)$,即$ b \cdot (k \circ a) \simeq (h \circ a) \cdot b$.
\end{proof}

\begin{corollary}
    假设$h \simeq k: X_0 \to X_1$.
    如果$h_*$是单射，或满射，或同构，那么$k_*$也是.
\end{corollary}

\begin{theorem}
    设 $X \simeq Y$，且 $f: X \to Y, g: Y \to X$ 满足 $\mathrm{id}_X \simeq g \circ f: X \to X$，$\mathrm{id}_Y \simeq f \circ g: Y \to Y$.
    取 $x_0 \in X$. 则 $f_*: \pi_1(X, x_0) \to \pi_1(Y, f(x_0))$ 是一个同构.
\end{theorem}

\begin{proof}
    \vspace{-2 em}
    \begin{center}
        \begin{tikzcd}[labels={font=\normalsize}]
{\pi_1(X,x_0)} \arrow[dd, "(\omega_1)_\#"] \arrow[dd, "\cong"'] \arrow[rr, "f_*"] &  & {\pi_1\big(Y,f(x_0)\big)} \arrow[dd, "\cong"'] \arrow[dd, "(\omega_2)_\#"] \arrow[lldd, "g_*"'] \\
                                                                                  &  &                                                                                                 \\
{\pi_1\big(X,g\circ f(x_0)\big)} \arrow[rr, "f_*"']                               &  & {\pi_1\big(Y,f\circ g\circ f(x_0)\big)}                                                        
\end{tikzcd}
    \end{center}\par
    (i) $\mathrm{id}_X \simeq g \circ f: X \to X$.\par
    由命题\ref{prop:10.5}，$g_*\circ f_* = (g \circ f)_* = (\omega_1)_\# \circ \mathrm{id}: \pi_1(X, x_0) \to \pi_1(X, g \circ f(x_0)) $\par
    即 $g_* \circ f_*$ 是一个同构，所以 $f_*$ 是单射，$g_*$ 是满射.\par
    (ii) $\mathrm{id}_Y \simeq f \circ g: Y \to Y$.\par
    则$f_*\circ g_*=(f \circ g)_* = (\omega_2)_\# \circ \mathrm{id}: \pi_1(Y, f(x_0)) \to \pi_1(Y, f \circ g \circ f(x_0)) $.\par
    即 $f_* \circ g_*$ 是一个同构，所以 $g_*$ 是单射，$f_*$ 是满射.\par
    因此 $g_*: \pi_1(Y, f(x_0)) \to \pi_1(X, g \circ f(x_0))$ 是一个同构.\par
    因为 $g_* \circ f_*: \pi_1(X, x_0) \to \pi_1(X, g \circ f(x_0))$ 是一个同构，
    所以 $f_*: \pi_1(X, x_0) \to \pi_1(Y, f(x_0))$ 是一个同构.
\end{proof}

\begin{corollary}
    设 $X, Y$ 是道路连通空间，则$X \simeq Y \implies \pi_1(X) \cong \pi_1(Y)$.
\end{corollary}

\begin{instance}
    设 $X$ 是道路连通空间. $X \times [0,1] \simeq X$.
    所以 $\pi_1(X \times [0,1]) \cong \pi_1(X)$.
\end{instance}

\begin{remark}
    $X \cong Y \implies X \simeq Y \implies \pi_1(X) \cong \pi_1(Y)$.
    $\pi_1(X) \not\cong \pi_1(Y) \implies X \not\simeq Y \implies X \not\cong Y$.
\end{remark}

\begin{note}
    \textbf{庞加莱猜想 (Poincaré Conjecture)}
    
    设 $M$ 是一个闭的 3 维流形.
    如果 $\pi_1(M) \cong \pi_1(S^3)$，那么 $M \cong S^3$.
    
    为了计算 $\pi_1(X)$，我们寻找一个更简单的空间 $Y$ 使得 $Y \simeq X$，并计算 $\pi_1(Y)$.
\end{note}

\section{形变收缩}

\begin{definition}{收缩映射}
    设 $X$是一个拓扑空间，$A \subset X$，$i: A \to X$为嵌入映射，$r: X \to A$.
    如果 $r|_A = \mathrm{id}_A: A \to A$，即 $r \circ i = \mathrm{id}_A: A \to A$，
    则称 $r$ 是一个\textbf{收缩映射}，且称 $A$ 是 $X$ 的一个\textbf{收缩核}.
\end{definition}

\begin{instance}
    \vspace{-2 em}
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.4\textwidth]{fig/10.32.png}
        \caption{收缩示例}
    \end{figure}
    $r: x \mapsto \begin{cases} x, & x \in A \\ x_0, & x \in B \end{cases}$ 是一个收缩映射.
\end{instance}

\begin{instance}
    $x_0 \in X$，$r: X \to \{x_0\}$，$x \mapsto x_0$ 是一个收缩映射.
\end{instance}

\begin{instance}
    $X, Y$ 是拓扑空间，$y_0 \in Y$，则$\setjot[-5]\begin{aligned}[t]
         r: X \times Y &\to X \times \{y_0\}
         \\(x, y) &\mapsto (x, y_0)
    \end{aligned}$ 是一个收缩.
\end{instance}

\begin{remark}
    $i_*: \pi_1(A, a_0) \to \pi_1(X, a_0)$ 是单射.
    $r_*: \pi_1(X, a_0) \to \pi_1(A, a_0)$ 是满射.
\end{remark}

\begin{definition}{形变收缩映射}
    设 $X$ 是拓扑空间，$A \subset X$，$i: A \to X$是嵌入映射，$r: X \to A$.
    如果 $r \circ i = \mathrm{id}_A$，且 $i \circ r \simeq \mathrm{id}_X \text{ rel } A$.
    则称 $r$ 是一个\textbf{形变收缩映射}，
    且称 $A$ 是 $X$ 的一个\textbf{形变收缩核}.
\end{definition}

\begin{remark}
    如果 $A$ 是 $X$ 的形变收缩核，那么 $X \simeq A$.
\end{remark}

假设 $H: X \times [0,1] \to X$ 实现了 $\mathrm{id}_X \simeq i \circ r \text{ rel } A$.
\begin{enumerate}
    \item $H(x, 0) = x, \forall x \in X$.
    \item $H(x, 1) = i \circ r(x) = r(x) \in A, \forall x \in X$.
    \item $H(a, t) = a, \forall a \in A, t \in [0,1]$.
\end{enumerate}

\begin{remark}
    $r \circ i(a) = r(a) \stackrel{(2)}{=} H(a, 1) \stackrel{(3)}{=} a$.
    所以 $\mathrm{id}_X \simeq i \circ r \text{ rel } A \implies r \circ i = \mathrm{id}_A$.\par
    因此，上述 $H$ 的存在性 $\iff r$ 是从 $X$ 到 $A$ 的形变收缩.
\end{remark}

\begin{instance}
    $X \times [0,1]$ 形变收缩到 $X \times \{0\}$.\par
    $\setjot[-6]\begin{aligned}[t]
        r: X \times [0,1] &\to X \times \{0\} \\
        (x, s) &\mapsto (x, 0)
    \end{aligned} $ 是一个收缩映射.\par
    定义$\setjot[-5]\begin{aligned}[t]
        H: X \times [0,1] \times [0,1] &\to X \times [0,1] \\
        (x, s, t) &\mapsto (x, (1-t)s)
    \end{aligned}$，则$H$实现了从$X \times [0,1]$到$X \times \{0\}$的形变收缩.\par
    (1) $H(x, s, 0) = (x, s)$.
    (2) $H(x, s, 1) = (x, 0) \in X \times \{0\}$.
    (3) $H(x, 0, t) = (x, 0)$.
\end{instance}

\begin{instance}
    $\mathbb{E}^n \setminus \{0\}$ 形变收缩到 $S^{n-1}$，$n \geq 1$.\par
    \begin{wrapfigure}{r}{0.3\textwidth}
        \vspace{-3 em}
        \includegraphics[width=0.3\textwidth]{fig/10.33.png}
        \captionof{figure}{到圆周上的形变收缩}
        \vspace{-1 em}
    \end{wrapfigure}
    $\setjot[-4]\begin{aligned}[t]
        r: \mathbb{E}^n \setminus \{0\} &\to S^{n-1} \\
        x &\mapsto \frac{x}{\|x\|}
    \end{aligned} $ 是一个收缩映射.\par
    定义$\setjot[-6]\begin{aligned}[t]
        H: (\mathbb{E}^n \setminus \{0\}) \times [0,1] &\to \mathbb{E}^n \setminus \{0\} \\
        (x, t) &\mapsto (1-t)x + t \frac{x}{\|x\|}
    \end{aligned}$.\par
    \begin{enumerate}[(1)]
        \item $H(x, 0) = x$.
        \item $H(x, 1) = \frac{x}{\|x\|} \in S^{n-1}$.
        \item $\forall\, x\in S^{n-1},\|x\|=1$，$H(x, t) = (1-t)x + tx = x$.
    \end{enumerate}
\end{instance}

\begin{instance}
    $\mathbb{E}^2 \setminus \{p_1, p_2, p_3\}$.
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.8\textwidth]{fig/10.34.png}
        \captionof{figure}{}
    \end{figure}
    $r: \mathbb{E}^2 \setminus \{p_1, p_2, p_3\} \to A$.
    $$ x \mapsto r(x) $$
    $r(x)$ 是包含 $x$ 的直线的终点.
    $H: X \times [0,1] \to X$.
    $$ (x, t) \mapsto (1-t)x + t \cdot r(x) $$
    (1) $H(x, 0) = x$.
    (2) $H(x, 1) = r(x) \in A$.
    (3) $H(a, t) = (1-t)a + t \cdot a = a, \forall a \in A$.
\end{instance}

\section{商空间的形变收缩}

\begin{theorem}\label{thm:10.6}
    $f: X \to Y$ 是一个商映射. $A \subset X$ 是一个形变收缩核. $B = f(A) \subset Y$.
    $H: X \times [0,1] \to X$ 实现了从 $X$ 到 $A$ 的形变收缩.
    如果 $f(x) = f(x') \implies f \circ H(x, t) = f \circ H(x', t)$，
    那么 $B$ 是 $Y$ 的一个形变收缩核.
\end{theorem}

\begin{proof}
    \vspace{-2 em}
    \begin{center}
        \begin{tikzcd}[labels={font=\normalsize}]
{\makebox[0pt][r]{${(x,t)\in\,}$}X\times[0,1]} \arrow[rr, "H"] \arrow[dd, "f\times\mathrm{id}"'] \arrow[rrdd, "f\circ H"] &  & X \arrow[dd, "f"]    \\
                                                                                                   &  &                      \\
{\makebox[0pt][r]{${(y,t)\in\,}$}Y\times[0,1]} \arrow[rr, "H_1"']                                                         &  & {Y \makebox[0pt][l]{${} \ni f \circ H(x,t)$}}
\end{tikzcd}
    \end{center}\par
    定义$\setjot[-2]\begin{aligned}[t]
        H_1: Y \times [0,1] &\to Y \\
        (y, t) &\mapsto f \circ H(x, t), x \in f^{-1}(y)
    \end{aligned} $.\par
    如果 $x' \in f^{-1}(y)$，那么 $f(x) = f(x') \implies f \circ H(x, t) = f \circ H(x', t)$.\par
    所以 $H_1(y, t)$ 独立于 $x \in f^{-1}(y)$ 的选取，即 $H_1$ 是良定义的.
    \begin{lemma}
        $f: X \to Y$ 是一个商映射，那么 $f \times \mathrm{id}: X \times [0,1] \to Y \times [0,1]$ 是一个商映射.
    \end{lemma}
    因为 $f \times \mathrm{id}$ 是一个商映射且 $f \circ H$ 是连续的，所以 $H_1$ 是连续的.并且：
    \begin{enumerate}[(1)]
        \item $H_1(y, 0) = f \circ H(x, 0) = f(x) = y, \forall\, y \in Y$.
        \item $H_1(y, 1) = f \circ H(x, 1) \in f(A) = B, \forall\, y \in Y$.
        \item $H_1(b, t) = f \circ H(a, t) = f(a) = b, \forall\, b \in B, t \in [0,1]$.
    \end{enumerate}

\end{proof}

\section{有限表现群}

\begin{definition}{群的自由积}
    设$G_1,G_2$是群，定义它们的\textbf{自由积}为：
    \[ G_1*G_2:=\{ \alpha_1\beta_1\alpha_2\beta_2\dots\alpha_n\beta_n \mid \alpha_i\in G_1,\beta_i\in G_2 \} \]
    其中运算法则如下：
    \[ \alpha_1\beta_1\dots\alpha_n\beta_n \cdot \alpha'_1\beta'_1\dots\alpha'_m\beta'_m = \alpha_1\beta_1\dots\alpha_n\beta_n \alpha'_1\beta'_1\dots\alpha'_m\beta'_m \]
\end{definition}

\begin{instance}
    考虑两个无限循环群$\mathbb{Z}=\{ \dots,a^{-2},a^{-1},1,a,a^2,\dots \}$，$\mathbb{Z}=\{ \dots,b^{-2},b^{-1},1,b,b^2,\dots \}$.\par
    则$\mathbb{Z}*\mathbb{Z}$.
    其凯莱图如下：
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.5\textwidth]{example-image}
        \caption{$\mathbb{Z}*\mathbb{Z}$的凯莱图}
    \end{figure}
\end{instance}

\begin{definition}{自由群}
    $\underbrace{\mathbb{Z}*\mathbb{Z}*\dots*\mathbb{Z}}_{n}=F_n$称为\textbf{秩为$n$的自由群}.
\end{definition}

\begin{definition}{生成元集}
    $\{g_1,g_2,\dots,g_n\}$是群$G$的\textbf{生成元集}，如果$G$中的每个元素都是有限个$g_i$及其逆元的乘积.
\end{definition}

\begin{definition}{正规闭包}
    设$G$是群，$S\subset G$是子集.包含$S$的$G$的最小\textbf{正规子群}称为$S$的\textbf{正规闭包}，记为$\overline{S}$.
    
    具体地，
    \[ \overline{S}:=\{ g_1 s_1^{\pm 1} g_1^{-1} \dots g_n s_n^{\pm 1} g_n^{-1} \mid g_i\in G, s_i\in S \} \]
    即$\overline{S}$由$\{ gsg^{-1} \mid g\in G, s\in S \}$生成.
    显然对于任意$h\in G$，有$h \left( g_1 s_1 g_1^{-1} \dots \right) h^{-1}\in \overline{S}$.
\end{definition}

\begin{definition}{自由群}
    自由群 $F_n = \langle g_1, g_2, \dots, g_n \rangle = \underbrace{\mathbb{Z} * \mathbb{Z} * \dots * \mathbb{Z}}_{n}$.
\end{definition}

\begin{definition}{有限表现群}
    设 $\{r_1, r_2, \dots, r_m\} \subset F_n$.
    商群 $F_n / \overline{\{r_1, r_2, \dots, r_m\}}$ 记为
    \[ \langle g_1, g_2, \dots, g_n \mid r_1, r_2, \dots, r_m \rangle = G \]
    其中 $g_i$ 称为\textbf{生成元}，$r_j$ 称为\textbf{关系元}.
    有限表现群是指具有有限个生成元和有限个关系元的群.
\end{definition}

\begin{instance}
    $\mathbb{Z}/n\mathbb{Z} = \langle g \mid g^n \rangle = \{ 1, g, g^2, \dots, g^{n-1} \}$.
\end{instance}

\begin{instance}
    $\mathbb{Z} \times \mathbb{Z} = \langle g_1, g_2 \mid g_1 g_2 g_1^{-1} g_2^{-1} \rangle$.
    关系 $g_1 g_2 g_1^{-1} g_2^{-1} = 1$ 蕴含 $g_1 g_2 = g_2 g_1$.\par
    一般地，$\underbrace{\mathbb{Z} \times \mathbb{Z} \times \dots \times \mathbb{Z}}_{n} = \langle g_1, \dots, g_n \mid g_i g_j g_i^{-1} g_j^{-1}, 1 \leq i < j \leq n \rangle$.
    这称为\textbf{自由阿贝尔群}.
\end{instance}

\section{Seifert-van Kampen 定理}

设 $X$ 是一个拓扑空间，$\{U, V\}$ 是 $X$ 的开覆盖.
假设 $U, V, U \cap V \neq \varnothing$ 都是道路连通的(从而 $X$ 是道路连通的). \par
则有以下交换图成立： \par
\begin{tikzcd}[labels={font=\normalsize}]
                                               &  & U \arrow[rrd, "j_1"]  &  &   \\
\makebox[0pt][r]{$\textcolor{red}{x_0\in}$\,}U\cap V \arrow[rru, "i_1"] \arrow[rrd, "i_2"'] &  &                       &  & X \\
                                               &  & V \arrow[rru, "j_2"'] &  &  
\end{tikzcd}
\begin{tikzcd}[labels={font=\normalsize}]
                                               &  & \pi_1(U, x_0) \arrow[rrd, "(j_1)_*"]  &  &   \\
\pi_1(U \cap V, x_0) \arrow[rru, "(i_1)_*"] \arrow[rrd, "(i_2)_*"'] &  &                       &  & \pi_1(X, x_0) \\
                                               &  & \pi_1(V, x_0) \arrow[rru, "(j_2)_*"'] &  &  
\end{tikzcd}\par
即：$j_1\circ i_1=j_2\circ i_2,(j_1)_* \circ (i_1)_* = (j_2)_* \circ (i_2)_*$.

定义$\setjot[-4]\begin{aligned}[t]
    \Phi: \pi_1(U, x_0) * \pi_1(V, x_0) &\to \pi_1(X, x_0) \\
    \alpha_1 \beta_1 \dots \alpha_n \beta_n &\mapsto (j_1)_*(\alpha_1) (j_2)_*(\beta_1) \dots (j_1)_*(\alpha_n) (j_2)_*(\beta_n)
\end{aligned}$，其中 $\alpha_i \in \pi_1(U, x_0), \beta_i \in \pi_1(V, x_0)$.则$\Phi$是一个同态.

\begin{lemma}
    $\Phi$ 是满射.
\end{lemma}

\begin{proof}
    设 $a: [0,1] \to X$ 是 $X$ 中以 $x_0$ 为基点的闭路.\par
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.6\textwidth]{fig/12.1.png}
        \caption{回路分割示意图}
    \end{figure}
    由Lebesgue引理，则可以选取足够大的$n$，使得$a$将每个区间 $\left[\frac{i-1}{n}, \frac{i}{n}\right]$完整映入$U$ 或 $V$ 中.\par
    令 $a_i = a|_{\left[\frac{i-1}{n}, \frac{i}{n}\right]}:[0,1]\to \left[\frac{i-1}{n},\frac{i}{n}\right]\stackrel{a}{\to} X$.\par
    $\forall\,i=1, \dots, n-1$，选择一条从 $x_0$ 到 $a\left(\frac{i}{n}\right)$的道路$h_i$.\par
    如果$a\left(\frac{i}{n}\right) \in U \cap V$，则选择 $h_i$ 完全位于 $U \cap V$ 中.否则，选择它位于 $U$ 或 $V$ 中(取决于点在哪里).\par
    定义回路 $\tilde{a}_i = h_{i-1} \cdot a_i \cdot \overline{h}_i$(其中 $h_0, h_n$ 为常值路径)，则$[\tilde{a}_i] \in \pi_1(U, x_0)$ 或 $\in \pi_1(V, x_0)$.\par
    令$\mathcal{V} = [\tilde{a}_1] \cdot [\tilde{a}_2] \dots [\tilde{a}_n] \in \pi_1(U, x_0) * \pi_1(V, x_0)$.
    则 $\Phi(\mathcal{V}) = [\tilde{a}_1 \cdot \tilde{a}_2 \dots \tilde{a}_n] = [a]$.
\end{proof}
基于以上结论，由于同态基本定理，就有$\pi_1(X, x_0) \cong \pi_1(U, x_0) * \pi_1(V, x_0) / \ker \Phi$.\par

\begin{lemma}
    设 $N$ 是 $\pi_1(U, x_0) * \pi_1(V, x_0)$ 中集合$\{ (i_1)_*(\alpha) (i_2)_*(\alpha^{-1}) \mid \alpha \in \pi_1(U \cap V, x_0) \}$
    的正规闭包，则 $N \leq \ker \Phi$.
\end{lemma}

\begin{proof}
    若$(i_1)_*(\alpha) (i_2)_*(\alpha^{-1}) \in \ker \Phi$，那么$\Phi\left((i_1)_*(\alpha) (i_2)_*(\alpha^{-1})\right)=1$.\par
    故$\Phi\left(g(i_1)_*(\alpha) (i_2)_*(\alpha^{-1})g^{-1}\right)=\Phi(g)\Phi\left((i_1)_*(\alpha) (i_2)_*(\alpha^{-1})\right)\Phi(g^{-1})=1$.\par
    因此只需证明对于任意 $\alpha \in \pi_1(U \cap V, x_0)$，有 $(i_1)_*(\alpha) (i_2)_*(\alpha^{-1}) \in \ker \Phi$.\par
    $\setjot[-3]\begin{aligned}[t]
        \Phi((i_1)_*(\alpha) (i_2)_*(\alpha^{-1})) &= (j_1)_*((i_1)_*(\alpha)) (j_2)_*((i_2)_*(\alpha^{-1})) \\
        &= (j_1)_*((i_1)_*(\alpha)) ((j_2)_*((i_2)_*(\alpha)))^{-1} \quad\left((j_1)_* \circ (i_1)_* = (j_2)_* \circ (i_2)_*\right) \\
        &= 1
    \end{aligned}$.
\end{proof}

\begin{lemma}
    $\ker \Phi \leq N$.
\end{lemma}

\begin{proof}
    略.
\end{proof}

因此，$\ker \Phi = N$.

\begin{theorem}[Seifert-van Kampen 定理]
    \[ \pi_1(X, x_0) \cong \pi_1(U, x_0) * \pi_1(V, x_0) / N \]
    其中 $N$ 是由 $\{ (i_1)_*(\alpha) (i_2)_*(\alpha^{-1}) \mid \alpha \in \pi_1(U \cap V, x_0) \}$ 生成的正规闭包.
\end{theorem}

\end{instance}

\begin{instance}
    设 $Y$ 是一个空间，$D^n$ 是 $n$ 维圆盘($n \geq 3$).
    $f: S^{n-1} \to Y$.
    $X = Y \cup_f D^n = Y \sqcup D^n / x \sim f(x), x \in S^{n-1} = \partial D^n$.
    
    则 $\pi_1(Y \cup_f D^n) \cong \pi_1(Y)$.
\end{instance}

\begin{proof}
    取 $p \in \text{Int}(D^n)$.
    $U = X - \{p\} \xrightarrow{\text{def. ret.}} Y$.
    $V$ 是 $p$ 的小圆盘邻域.
    $U \cap V = V - \{p\} \simeq S^{n-1}$.
    由于 $n \geq 3$， $S^{n-1}$ 是单连通的.
    所以
    \[
    \begin{aligned}
    \pi_1(X) &\cong \pi_1(U) * \pi_1(V) \\
    &\cong \pi_1(U) \cong \pi_1(Y)
    \end{aligned}
    \]
\end{proof}
