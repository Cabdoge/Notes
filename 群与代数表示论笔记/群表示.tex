\chapter{群表示}

\section{群表示的定义}

\begin{definition}[群表示]
    设$G$是一个群，$V$是域$\mathbb{F}$上的线性空间，则群的\bluebf{表示}是$G$到$GL(V)$的群同态$\setjot[-3]\begin{aligned}[t]
    \rho\colon G &\to GL(V) \\
    g &\mapsto \rho(g)
    \end{aligned}$.称$V$的维数$\mathrm{dim}_{\mathbb{F}}\,V$为该表示的\bluebf{次数}，记为$\mathrm{deg}\,\rho$.
\end{definition}

为了使$\rho$是一个满足上述条件的群同态，我们首先要保证$\rho$将$G$中的元素映射到$GL(V)$.其次，$\rho$必须满足群同态保持运算的性质，即满足以下两个条件：
\begin{enumerate}[label=(\roman*)]
    \item $\forall\, g\in G,\rho(g)\in GL(V)$，即$\rho(g)$是$V$上的可逆线性变换.
    \item $\forall\, g_1,g_2\in G,\rho(g_1\cdot g_2)=\rho(g_1)\cdot\rho(g_2)$.
\end{enumerate}

有时候，直接证明$\rho$将元素映为可逆线性变换较为复杂，因此上述的条件(i)也可以等价地替换为以下条件：
\begin{enumerate}[label=(\roman*)]
    \setcounter{enumi}{2}
    \item $\rho(1)=1_V$，其中$1$是$G$中单位元，$1_V$是$V$上的恒等变换.
\end{enumerate}

下面我们可以证明条件$(i)$和$(iii)$是等价的.
\begin{proof}[条件(i)和(iii)的等价性]\par
    首先证明$(i)\Rightarrow(iii)$：由于$\rho$是群同态，我们有$\rho(1)=\rho(1\cdot 1)=\rho(1)\cdot\rho(1)$，即$\rho(1)^2=\rho(1)$.\par
    又因为$\rho(1)\in GL(V)$，所以$\rho(1)$必须是恒等变换$1_V$.\par
    接下来证明$(iii)\Rightarrow(i)$：对于任意$g\in G$，我们有$\rho(g)\cdot\rho(g^{-1})=\rho(g\cdot g^{-1})=\rho(1)=1_V$.\par
    故$\rho(g)$是可逆的，因此$\rho(g)\in GL(V)$.
\end{proof}

除了使用线性变换的角度定义群表示，群表示的本质是群在向量空间上的线性作用，因此也可以从以下观点定义群表示：
\begin{definition}[群作用观点的群表示]
    设$G$是一个群，$V$是域$\mb{F}$上的线性空间，如果存在$G$在$V$上的线性作用，即存在映射$\setjot[-4]\begin{aligned}[t]
    G\times V &\to V \\
    (g,v) &\mapsto g\cdot v
    \end{aligned}$满足以下条件：
    \begin{enumerate}[label=(\roman*)]
        \item $\forall\, g\in G,u,v\in V,g\cdot(v+w)=g\cdot v+g\cdot w$.
        \item $\forall\, g\in G,a\in\mb{F},v\in V$，$g\cdot(av)=a(g\cdot v)$.
        \item $\forall\, v\in V,\, 1\cdot v=v$.
        \item $\forall\, g_1,g_2\in G,\, v\in V,\, (g_1g_2)\cdot v=g_1\cdot(g_2\cdot v)$.
    \end{enumerate}
    则称$V$是$G$的一个$\mb{F}-$表示.
\end{definition}
上述定义中的条件(i)和(ii)保证了$G$在$V$上的作用是线性的，条件(iii)和(iv)说明这是一个群作用.\par
可以证明上面两个对于群表示的定义是相容的，只要有一个$G$在$V$上的线性作用，就可以定义$G$的表示为$g\to\rho(g)v\triangleq g\cdot v$.反之如果有群表示$\rho$，就可以定义$G$在$V$上的线性作用为$g\cdot v\triangleq \rho(g)v$.因此我们可以根据需要选择其中一个定义来研究群表示.

\begin{instance}
    \begin{enumerate}
        \item 设$G$是一个群，$V$是域$\mb{F}$上的线性空间，则$\setjot[-8]\begin{aligned}[t]
    \rho\colon G &\to GL(V) \\
    g &\mapsto 1_V
    \end{aligned}$.它将群$G$中的每个元素映射到$V$上的恒等变换，因此称之为\bluebf{平凡表示}.\par
        \item 设$G$是一个有限群，$V=\mb{F}$，则$GL(V)\cong \mb{F}^*=\mb{F}\setminus\{0\}$.\par
        由于$G$是有限群，其中任意一个元素$g$的阶数有限，即$\exists\, m\in\mb{N}$，使得$g^m=1$.\par
        那么$\rho(g)^m=\rho(g^m)=\rho(1)=1_V$，从而$\rho(g)$的$m$次单位根.\par
        因而群表示为$\setjot[-6]\begin{aligned}[t]
            \rho:G &\to GL(V)\cong\mb{F}^*=\mb{F}\setminus\{0\} \\
            g &\mapsto \rho(g)\triangleq \xi(\xi^m=1,m\,\text{为}\,g\,\text{的阶数})
        \end{aligned}$.\par
        由于$V$的维数是1，因此上述表示称为\bluebf{一次表示}.
        \item 设$V$的基为$\{g\in G\}$，则$V=\left\{\sum_{i=1}^na_ig_i\right\}$.\par
        考虑群$G$在$V$上的左正则作用，则群表示为$\setjot[-2]\begin{aligned}[t]
            \rho:G&\to GL(V) \\
            h&\mapsto\rho(h)\left(\sum_{i=1}^n{a_ig_i}\right)\triangleq \sum_{i=1}^na_i(hg_i)
        \end{aligned}$\par
        由于它源自于群$G$在$V$上的左正则作用，因此称之为\bluebf{正则表示}，它是一种特殊的置换表示.\par
        如果取二阶循环群$S_2=\{(1),(12)\}$.那么$\rho((1))=1_V$.\par
        而$\rho((12))(1)=(12)(1)=(12),\rho((12))(12)=(12)(12)=(1)$，故$\rho((12))=\begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}$.
        \item 设群$G$作用在集合$S$上，线性空间$V$以$S$为基，则可以定义群表示$\setjot[-6]\begin{aligned}[t]
            \rho:G&\to GL(V) \\
            g&\mapsto\rho(g)(s)\triangleq g\cdot s\in S
        \end{aligned}$\par
        任取$g\in G$，$\rho(g)(s_1,s_2,\cdots,s_n)=(s_1,s_2,\cdots,s_n)M$，其中$M$是一个置换矩阵，因此该表示称为\bluebf{置换表示}.\par
        如果取$G=S^3$，作用在$S=\{1,2,3\}$上，$V=\mb{F}\cdot S$.\par
        那么置换表示(以(12)为例)为$\setjot[-2]\begin{aligned}[t]
            \rho:G=S^3&\to GL(V) \\
            (12)&\mapsto \rho((12))(S)=(S)\begin{pmatrix}
                0 & 1 & 0 \\
                1 & 0 & 0 \\
                0 & 0 & 1
            \end{pmatrix}
        \end{aligned}$
    \end{enumerate}
\end{instance}


\section{子表示，商表示和表示同态}
类似于研究线性空间的子空间、商空间，拓扑空间的子空间、商空间等，下一步要研究群表示的子表示、商表示等概念，以便于我们更好地理解群表示的结构.\par

\begin{definition}[子表示]
    设$\rho$是$G$在$V$上的表示$\rho:G\to GL(V)$，设$U\subseteq V$，若$\rho|_U:G\to GL(U)$是$G$在$U$上的表示，即$\forall\, g\in G,u\in U,\, \rho(g)u\in U$，则称$\rho|_U$是$G$在$V$上的\bluebf{子表示}.
\end{definition}

对于上述的定义，如果从群作用的角度来看，$\rho|_U$是子表示当且仅当$G$在$U$上的群作用是封闭的，即$G$在$U$上的群作用将$U$映射到$U$中.\par
对于任意一个群$G$，向量空间$V$和上面的群作用$\rho$，总有$\rho|_V$和$\rho|_\varnothing$这两个子表示，它们是两个\bluebf{平凡子表示}，因此在后续研究过程中的一般情况下当我们说子表示时，默认考虑非平凡的子表示.\par
一个非平凡的子表示的构造方式是考虑一个群$G$的正则表示，定义$x\triangleq\sum_{g\in G}g,U\triangleq\mb{F}\cdot x$.那么就有$\rho(h)(x)\triangleq h\cdot x=x\in U$，这是因为$x$是$G$中所有元素之和，而$h$作用在$G$是对其中元素作置换，因此将所有元素之和置换后仍然是所有元素之和.因此$\rho|_U$是子表示.\par

除此之外，我们可以定义商表示.
\begin{definition}[商表示]
    设$(\rho,V)$是群$G$的表示，$U\subseteq V$，则$G$的\bluebf{商表示}$\setjot[-2]\begin{aligned}[t]
        \rho_{U/V}:G&\to GL(V/U) \\
        g&\mapsto \rho_{V/U}(g):V/U\to V/U
    \end{aligned}$定义为$\forall\, g\in G,x+U\in V/U,g(x+U):=gx+U$.
\end{definition}

设$(V,\rho)$为$G$的表示以及$U\subseteq V$，设$(\eta_1,\cdots,\eta_k)$是$U$的一组基，则$\rho|_U(g)(\eta_1,\cdots,\eta_k)=(\eta_1,\cdots,\eta_k)R_{U}$.将上述的基扩充为$V$上的基$(\eta_1,\cdots,\eta_k,\eta_{k+1},\cdots,\eta_n)$，并且考虑$\rho$的商表示$\rho_{V/U}$，那么类似地有$\rho_{V/U}(g)(\eta_{k+1}+U,\cdots,\eta_n+U)=(\eta_{k+1},\cdots,\eta_n)R_{V/U}$.\par
那么$\rho(g)$对应的矩阵就是一个分块准对角矩阵，即$\rho(g)(\eta_1,\cdots,\eta_n)=(\eta_1,\cdots,\eta_n)\begin{pmatrix}
    R_U & 0 \\
    0 & R_{V/U}
\end{pmatrix}$.\par
一般地，表示的分解可以推广到有限个的情形.如果表示可以分解为若干个表示的直和$V=U_1\oplus U_2\oplus\cdots\oplus U_k$，那么$\rho(g)=\begin{pmatrix}
    R_{U_1} & 0 & \cdots & 0 \\
    0 & R_{U_2} & \cdots & 0 \\
    \vdots & \vdots & \ddots & \vdots \\
    0 & 0 & \cdots & R_{U_k}
\end{pmatrix}$\par

这自然引出了一个问题，如果我们有一个群表示和一个子表示，那么这个群表示能否分解为该子表示和其补子表示的直和.抛去表示，我们首先知道关于向量空间和其子空间，有以下特殊性质.
\begin{theorem}[]
    域$\mb{F}$上的线性空间$V$都可以分解成子空间$U$和其补空间$W$的直和.
\end{theorem}
上述定理的补空间可以使用商空间$V/U$来构造.但表示的结构要求每一个子空间都要对$G$的作用封闭，因此很遗憾，上述问题无法对所有的群表示成立.于是我们退而求其次一个群表示满足什么条件时可以分解，让我们先通过一个例子寻找什么样的群表示可以做到这一点.
\begin{instance}
    设$G=S^3,V=\mb{F}\{1,2,3\}$，考虑$G$在$V$上的置换表示$\setjot[-6]\begin{aligned}[t]
    \rho:G&\to GL(V)&& \\
    g &\mapsto \rho(g):\setjot[-6]\begin{aligned}[t]
    V&\to V \\
    k&\mapsto g\cdot k,\forall\, k\in\{1,2,3\}
    \end{aligned}
    \end{aligned}$ \par
    取$W=\mb{F}\{1+2+3\}$，则$(W,\rho|_W)$为子表示.问题是是否存在子空间$U$，使得$V=W\oplus U$，并且$\rho|_U$是子补表示.\par
    先取$W$的补子空间，一种方式是直接取$W$的正交补空间$U=W^\perp=\left\{x\in V|\langle x,1+2+3\rangle=0\right\}$，其中$\langle i,j\rangle =\delta_{ij}$.\par
    由于$x\in V,x=a\cdot 1+b\cdot 2+c\cdot 3,a,b,c\in\mb{F}$.\par
    那么$\langle x,1+2+3\rangle=a+b+c=0\Rightarrow U=\{a\cdot 1+b\cdot 2+c\cdot 3:a+b+c=0,a,b,c\}$.\par
    $\forall\,g\in S^3,u\in U,g\cdot u=g(a\cdot 1+b\cdot 2+c\cdot 3)=a\cdot g(1)+b\cdot g(2)+c\cdot g(3)\in U$.\par
    进而可以看出$\rho|_U$也是$\rho$的子表示，所以$V=W\oplus U$作为表示也是直和.
\end{instance}
上述例题告诉我们对于对称群$G=S_n$和它在$\mb{C}$上的表示$\rho:G\to GL(V)$，若$W\subseteq V$为子表示，则$\exists\, \text{子表示}\, U,\st V=W\oplus U$.更进一步地我们有以下结论
\begin{theorem}[]
    设$(\rho,V)$是$\mb{C}$上的群表示，$W\subseteq V$是其子表示，则$\exists\, \text{子表示}\, U\subseteq V$，使得$V=W\oplus U$.
\end{theorem}
\begin{proof}
    设$p:V\to W$是$V$到$W$的投影，定义$p'\triangleq \frac{1}{|G|}\sum_{g\in G}\rho(g)p\rho(g^{-1})$.\par
    \begin{enumerate}[label=(\roman*)]
        \item \textbf{\boldmath $p'$是投影}\par
        $\forall\, v\in v,\rho(g^{-1})(v)\in V,p\rho(g^{-1})(v)\in W,\rho(g)p\rho(g^{-1})(v)\in W$.\par
        $\forall\, \in W$，$p'(x)=\frac{1}{|G|}\sum_{g\in G}\rho(g)p\rho(g^{-1})(x)\xlongequal{\rho(g^{-1})(x)\in W}\frac{1}{|G|}\sum_{g\in G}\rho(g)\rho(g^{-1})(x)=\frac{1}{|G|}\sum_{g\in G}x=x$.\par
        \item \textbf{\boldmath $\ker p'$是子表示，即$\forall\, x\in\ker p',\rho(g)(x)\in\ker p'$}\par
        $\setjot[-2]\begin{aligned}[t]
            p'(\rho(g)(x)) &=\frac{1}{|G|}\sum_{h\in G}\rho(h)p\rho(h^{-1})(\rho(g)(x)) =\frac{1}{|G|}\sum_{h\in G}\rho(h)p\rho(h^{-1}g)(x) \\
            &=\frac{1}{|G|}\rho(g)\sum_{h\in G}\rho(g^{-1}h)p\rho(h^{-1}g)(x) \\
            &=\frac{1}{|G|}\rho(g)\sum_{h'\in G}\rho(h')p\rho(h'^{-1})(x) \\
            &=\rho(g)p'(x)=0
        \end{aligned}$ \par
        $\Rightarrow \rho(g)(x)\in\ker p'$
    \end{enumerate}
\end{proof}\par
由此我们就完成了证明，证明的核心在于利用原先的投影$p$构造了一个新的投影$p'$，使得$p'$满足$\rho(g)p'=p'\rho(g)$，从而保证了$\ker p'$是子表示.而在上述证明过程中，数域为$\mb{C}$的要求主要体现在，在$\mb{C}$中，不论群$G$的阶是多少，它总不为域下的零，因此可以作为分母，保证了$p'$定义的合理性.\par
因此对于上述问题，当$G$是有限群且$\mathrm{char}\,(\mb{F})\nmid |G|$时，证明依旧成立，它在后续被我们称为Maschke定理.此时称$G$的表示是\bluebf{常表示}，否则称其为\bluebf{模表示}.\par
有限群的表示论在上述两个类型的研究有本质的不同，前者结构相对优美，有更好的结果.而对于后者的研究则相对复杂.\par
在开始下一个环节之前，补充一些对称群的相关知识.\par
对于对称群$S_n$，其生成元为对换$(i,i+1)=s_i,1\leq i\leq n-1$，并且满足如下三条生成关系：
(i)$s_i^2=1$\hspace{2em}(ii)$s_is_j=s_js_i,|i-j|>1$\hspace{2em}(iii)$s_is_{i+1}s_i=s_{i+1}s_is_{i+1}$ \par
因此在后续研究对称群时，实际上只需要考虑对称群的生成元以及它们之间的关系即可.\par
在了解完表示的子表示和商表示后，现在要讨论的是表示之间有何关系.
\begin{definition}[表示同态，$G-$模同态]
    设$(\rho_1,V_1),(\rho_2,V_2)$是$G$的两个表示，如果线性映射$f:V_1\to V_2$使得下述交换图成立 \par
    \begin{center}
    \begin{tikzcd}[labels={font=\normalsize}]
V_1 \arrow[dd, "\rho_1(g)"'] \arrow[rr, "f"] &  & V_2 \arrow[dd, "\rho_2(g)"] \\
                                             &  &                             \\
V_1 \arrow[rr, "f"]                          &  & V_2                        
\end{tikzcd}
    \end{center}
    即$\forall\, g\in G,f\rho_1(g)=\rho_2(g)f$，则称$f$是\bluebf{表示同态}或\bluebf{$G-$模同态}.
\end{definition}
\begin{definition}[表示同构]
    设$(\rho_1,V_1),(\rho_2,V_2)$是两个表示，$f:V_1\to V_2$是表示同态，如果$f$是双射，则称$f$是\bluebf{表示同构}，称\bluebf{$(\rho_1,V_1)\cong(\rho_2,V_2)$}.
\end{definition}
\begin{instance}
    设$G$是群，$V$的基为$\{g\in G\}$，定义$\rho':G\to GL(V),\exists\, w\in W,\st \{\rho'(g)(w)|g\in G\}$是$W$的基，则$(\rho,W)\cong(\rho_{\mathrm{reg}},V_{\mathrm{reg}})$\par
    注意到正则表示中的基元素可写作$\rho_\mathrm{reg}(1)g$，取$\setjot[-6]\begin{aligned}[t]
        f:V_\mathrm{reg}&\to W \\
        \rho_\mathrm{reg}(1)g&\mapsto \rho'(g)(w)
    \end{aligned} $作为线性空间的同构.\par
    那么$\rho'(g)f(h)=\rho'(g)f\left(\rho_\mathrm{reg}(1)(h)\right)=\rho'(g)\rho'(h)(w)=\rho'(gh)(w)$.\par
    $f\rho_\mathrm{reg}(g)(h)=f(gh)=f(\rho_{\mathrm{reg}}(1)(gh))=\rho'(gh)(w)$.\par
    因此$f\rho_\mathrm{reg}(g)=\rho'(g)f$，$f$是表示同态，又$f$是$V_1\to V_2$的同构，因此$f$是表示同构.
\end{instance}

表示同态有以下两个符合直觉的重要性质：
\begin{property}[表示同态]
    设$(\rho_1,V_1),(\rho_2,V_2)$是$G$的两个表示，$f:(\rho_1,V_1)\to(\rho_2,V_2)$为表示同态
    \begin{enumerate}
        \item $\ker f\triangleq \{v\in V_1|f(v)=0\}$是$V_1$的子表示
        \item $\im f \triangleq \{f(v)|v\in V_1\}$是$V_2$的子表示
    \end{enumerate}
\end{property}
\begin{proof}
    \begin{enumerate}
        \item $\forall\, x\in \ker f,f\rho_1(g)(x)=\rho_2(g)f(x)=\rho_2(g)(0)=0$，所以$\rho_1(g)(x)\in \ker f$
        \item $\forall\, y\in \im f,y=f(x)$，则$\rho_2(g)(y)=\rho_2(g)f(x)=f\rho_1(g)(x)\in \im f$
    \end{enumerate}
\end{proof}

下面一个引理是我们后面常用的引理.
\begin{lemma}[]
    设$(\rho_1,V_1),(\rho_2,V_2),(\rho_3,V_3)$是$G$的表示，$f:V_1\to V_2,g:V_2\to V_3$为表示同态，若$\ker f\subseteq \ker g$，则$\exists\, !h:\im f\to V_3,\st g=hf$.
    \begin{center}
        \begin{tikzcd}[labels={font=\normalsize}]
V_1 \arrow[rr, "f"] \arrow[dd, "g"'] &  & {\mathrm{Im}\, f} \arrow[lldd, "{\exists\, !h}", dashed] \\
                                     &  &                                                          \\
V_3                                  &  &                                                         
\end{tikzcd}
    \end{center}
\end{lemma}
\begin{proof}
    定义$\setjot[-4]\begin{aligned}[t]
        h:\im f&\to V_3 \\
        f(a)&\mapsto h(f(a))\triangleq g(a)
    \end{aligned}$\par
    若$f(a)=0\Rightarrow a\in\ker f\subseteq \ker g$，则$h(f(a))=g(a)=0$，故$h$良定.\par
    下证唯一性.\par
    若$\exists\, h_1\neq h_2$均满足$h_1f=h_2f=g$.\par
    则$\exists\, a\in \im f,\st h_1(a)\neq h_2(a)$，但由于$a\in \im f,a=f(b)\Rightarrow h_1f(b)\neq h_2f(b)$，与$h_1f=h_2f$矛盾，从而唯一.
\end{proof}

记$\Hom_G(V_1,V_2)=\left\{f:V_1\to V_2,f\,\text{为表示同态}\right\}$，$\Hom_F(V_1,V_2)=\left\{f:X_1\to V_2,f\,\text{为线性映射}\right\}$，则二者都是线性空间，自然可以定义群$G$在上面的表示，特别地由于$\Hom_G(V_1,V_2)\subseteq \Hom_F(V_1,V_2)$，我们只需要在$\Hom_F(V_1,V_2)$上定义表示即可.\par
\begin{definition}[]
    定义$G$在$W=\Hom_F(V_1,V_2)$上的表示为$\setjot[-2]\begin{aligned}[t]
        \rho(g):W&\to W \\
        f &\mapsto \rho(g)f: \begin{aligned}
            V_1 &\to V_2 \\
            v_1&\mapsto\rho(g)f(v_1)\triangleq \rho_2(g)\left(f\left(\rho_1(g^{-1})(v_1)\right)\right)
        \end{aligned} 
    \end{aligned}$
\end{definition}
为了保证定义的合理性，我们要验证$\rho$确实是一个群表示，即满足$\rho(g_1g_2)=\rho(g_1)\rho(g_2)$.\par
$\rho(gh)(f)(v_1)\triangleq \rho_2(gh)\left(f\left(\rho_1(h^{-1}g^{-1})\right)(v_1)\right)$.\par
$\rho(g)\rho(h)(f)(v_1)\triangleq \rho(g)\left(\rho_2(h)\left(f\left(\rho_1(h^{-1})\right)(v_1)\right)\right)$，其中$\rho_2(h)\left(f\left(\rho_1(h^{-1})\right)\right)$是$V_1\to V_2$的线性映射，因此有$\rho(g)\left(\rho_2(h)\left(f\left(\rho_1(h^{-1})\right)(v_1)\right)\right)=\rho_2(g)\rho_2(h)\left(f\left(\rho_1(h^{-1})\rho_2(g^{-1})\right)(v_1)\right)$.\par
因此上述定义下的$(\rho,W)$确实是一个群表示.


\section{Maschke定理}

在上一节中我们已经证明了什么样的表示可以分解为其子表示和补子表示的直和，现在来正式定义表示的直和并进一步研究表示的分解问题.

\begin{definition}[表示的直和]
    设$(\rho_i,V_i),i=1,2,\cdots,s$均为$G$的表示，$V=V_1\oplus V_2\oplus\cdots\oplus V_s$，则$G$在$V$上的表示为
    \[\setjot[-2]\begin{aligned}[t]
        \rho:G&\to GL(V) \\
        g&\mapsto \rho(g):\setjot[-2]\begin{aligned}[t]
            V_1 &\to V_2 \\
            \rho(g)(v_1,\cdots,v_s)&\triangleq \left(\rho_1(v_1),\rho_2(v_2),\cdots,\rho_s(v_s)\right)
        \end{aligned}
    \end{aligned}\]
\end{definition}

上述对表示直和的定义是自然的，表示的直和就是将多个表示的向量空间进行直和，并且定义群作用为在每个分量上分别作用.\par
下面一个定理说明了一个线性空间的直和什么时候成为表示的直和.
\begin{theorem}[]
    设$V$和$V_i,i=1,2,\cdots,s$均是群$G$的表示，则$V$是表示$V_i$的直和，即$V=V_1\oplus V_2\oplus\cdots \oplus V_s$$\iff$$V$有子表示$V_i',i=1,2,\cdots,s$，且满足
    \begin{enumerate}[label=(\roman*)]
        \item $V_i'\cong V_i,\forall\, i$
        \item $V=V_1'+V_2'+\cdots+V_s'$
        \item $V_i'\cap \left(V_{i+1}'+\cdots+V_s'\right)=0,1\leq i\leq s-1$
    \end{enumerate}
\end{theorem}

关于一个表示是否有子表示，我们也有以下的定理，它实际上就是我们之前构造的与群作用可交换的投影的一般形式.

\begin{theorem}[]
    设$(\rho,V)$是群$G$的表示，$U\subseteq V$是子表示，则$\exists\, W,\st V=U\oplus V\iff\exists\, p:V\to U,p|_U=\mathrm{Id}$且$p$是$G-\!$模同态.
\end{theorem}
\begin{proof}
    $\Rightarrow$：定义$\setjot[-8]\begin{aligned}[t]
        p:V=U\oplus W&\to U \\
        (u,w) &\mapsto u
    \end{aligned}$.\par
    则$\forall\, u\in U$，$p(u)=u$，故$p|_U=\mathrm{id}$.\par
    $\forall\,g\in G,w=(v,u)\in W$，$p\rho(g)(w)=p\rho(g)(v,u)=p(\rho|_V(g)v,\rho|_U(g)u)=\rho|_U(g)u$.\par
    $\rho|_U(g)p(w)=\rho|_U(g)u$.因此$p$是$G-\!$模同态.\par
    $\Leftarrow$：定义$W\triangleq \ker p$.\par
    \begin{enumerate}[label=(\roman*),leftmargin=4em]
        \item {\boldmath{$V=U+W$}}\par
        $\forall\, v\in V$，$v=p(v)+(v-p(v))$.\par
        由$p$的定义可知，$p(v)\in U$.$p\left(v-p(v)\right)=p(v)-p(p(v))=p(v)-p(v)=0$，故$v-p(v)\in\ker p=W$.\par
        因此$V\subseteq U+W$.\par
        另一方面$W=\ker p\subseteq V,U\subseteq V$，所以$U+W\subseteq V$.
        \item {\boldmath{$U\cap W={0}$}}\par
        $\forall\, u\in U\cap W$，$p(u)=u$，又由于$u\in W=\ker p$，所以$p(u)=0$，因此$u=0$.
        \item \textbf{\boldmath $W$是子表示}\par
        由于$p$是表示同态，所以$\forall\, g\in G,p\rho(g)(w)=p\rho(g)(v-p(v))=p\rho(g)(v)-p\rho(g)p(v)=p\rho(g)(v)-\rho(g)p(v)=0$.\par
        因此$\rho(g)(w)\in W$,所以$W$是子表示.
    \end{enumerate}\par
\end{proof}

下面回到之前证明的结论，即若$\mathrm{char}\, F\nmid|G|$，那么$G$的表示可以分解为其子表示和补子表示的直和.一个自然的问题就是上述命题的逆命题是否依旧成立，于是我们有以下推论.

\begin{corollary}[]
    设$G$是有限群，$(\rho,V)$是其表示，且$\forall\, \text{子表示}\,U\subseteq V$，$\exists\, W,\st V=U\oplus W$，则$\mathrm{char}\, F\nmid |G|$.
\end{corollary}
\begin{proof}
    使用反证法，设$\mathrm{char}\, F \mid |G|$，$|G|=n$.\par
    考虑正则表示$FG=V,U=F\sum_{x\in G}{x}$.\par
    所以$\exists\, W,\st FG=U\oplus W$.\par
    那么其中的单位元可以写作$1=a\sum_{x\in G}+w,w\in W\Rightarrow x=1-a\sum_{x\in G}x$.\par
    设$G=\{g_1,g_2,\cdots,g_n\}$，由于$W$是子表示，那么$\forall\,i,g_iw\in W,\sum_{i=1}^n g_iw=0$.\par
    $g_iw=g_i-a\sum_{x\in G}{g_i x}=g_i-a\sum_{x\in G}x$.\par
    因此$\sum_{i=1}^n g_iw=\sum_{i=1}^n\left(g_i-a\sum_{x\in G}x\right)=(1-na)\sum_{x\in G}x$.\par
    由于$\mathrm{char}\, F\mid n$，故$1-na=1\Rightarrow \sum_{i=1}^n{g_iw}=\sum_{x\in G}x\in U$.\par
    故$\sum_{i=1}^n{g_iw}=0\Rightarrow \sum_{i=1}^n{g_i}=0$.矛盾!
\end{proof}

由此，若$G$有限，$\mathrm{char}\, F\nmid|G|$，则任意表示均可以写作$V=\bigoplus_{i=1}^s U_i$，其中$U_i$不可以再分解为非平凡的子表示，称之为\bluebf{不可约}.若$V$可以分解为不可约的表示的直和，则称$V$\bluebf{完全可约}.进而，Maskche定理可以表述为：

\begin{theorem}[Maschke定理]
    设$G$是一个有限群，则其任意表示完全可约$\iff \mathrm{char}\, F\nmid |G|$
\end{theorem}

如果$G$是无限群，取$\mb{F}=\mb{C}$，则对某些子表示$U$，可能不存在子表示$W$，使得$V=U\oplus W$.
\begin{instance}
    设$G=\mb{R}^+,\setjot[-6]\begin{aligned}[t]
        \rho:G&\to GL(\mb{C}^2) \\
        r&\mapsto \rho(r):\setjot[-4]\begin{aligned}[t]
            \mb{C}^2&\to \mb{C}^2 \\
            (\varepsilon_1,\varepsilon_2)&\mapsto(\varepsilon_1,\varepsilon_2)\begin{pmatrix}
                1 & \log r \\
                0 & 1
            \end{pmatrix}
        \end{aligned}
    \end{aligned}$\par
    由于若$V=U_1\oplus U_2$，则可以找到一组基，使得$\rho(g)$是一个分块对角矩阵.\par
    若不论基的选取，则$\rho(g)$应当与分块对角矩阵相似.\par
    在本例中，即$\exists\, T\in GL(\mb{C}^2),\st TMT^{-1}=\begin{pmatrix}
    \lambda_1 & 0\\
    0 & \lambda_2 
    \end{pmatrix}$，其中$M=\begin{pmatrix}
        1 & \log r \\
        0 & 1
    \end{pmatrix}$.\par
    由于相似变换不改变矩阵的特征值，因此$TMT^{-1}=\begin{pmatrix}
    1 & 0 \\
    0 & 1
    \end{pmatrix}$，得$TM=T$.\par
    那么$M=E_2$，矛盾!
\end{instance}

\section{特殊的表示}
下面我们来看一些特殊的表示，首先是反轭表示.
\begin{definition}[反轭表示]
    设$(\rho,V)$是$G$的表示，令$V^*=\Hom_F(V,F)$，定义其\bluebf{反轭表示}为 \[\setjot[-2]\begin{aligned}[t]
        \rho^*:G&\to GL(V^*) \\
        g&\mapsto \rho^*(g): \setjot[-2]\begin{aligned}[t]
            V^* &\to V^* \\
            f &\mapsto \rho^*(g)(f):\rho^*(g)(f)(v)\triangleq f\left(\rho(g^{-1})(v)\right)
        \end{aligned}
    \end{aligned}\]
\end{definition}
反轭表示就是将群在向量空间上的表示转化为在其对偶空间上的表示.现在，如果取$V$中的基$\{\varepsilon_1,\cdots,\varepsilon_n\}$及其对偶基$\{f_1,\cdots,f_n\}$满足$f_i(\varepsilon_j)=\delta_{ij}$.那么$\rho(g)$和$\rho^*(g)$在集下分别对应两个矩阵$R_g=(b_{ij})_{n\times n}$和$R^*_g=(a_{ij})_{n\times n}$，则$R_g$和$R^*_g$满足何等条件.\par
首先$\rho^*(g)(f_1,\cdots,f_n)=(f_1,\cdots,f_n)R^*_g$.设$R^*_g=(a_{ij})_{n\times n}$，因此$\rho^*(g)f_k=\sum_{i=1}^n f_ia_{ik}$.\par
将其作用在$V$的每一个基上，$\rho^*(g)f_k(\varepsilon_\ell)=\sum_{i=1}^nf_ia_{ik}(\varepsilon_\ell)=a_{\ell k}$.\par
另一方面，$\rho^*(g)f_k(\varepsilon_l)=f_k\left(\rho(g^{-1})(\varepsilon_\ell)\right)$，因此需要考虑$\rho(g^{-1})(\varepsilon_\ell)$的结果.\par
$\rho(g^{-1})(\varepsilon_1,\cdots,\varepsilon_n)=(\varepsilon_1,\cdots,\varepsilon_n)R_{g}^{-1}$，设$R_{g}^{-1}=(b_{ij})_{n\times n}$，故$\rho(g^{-1})(\varepsilon_\ell)=\sum_{i=1}^n\varepsilon_i b_{i\ell}$.\par
$\Rightarrow f_k\left(\rho(g^{-1})(\varepsilon_\ell)\right)=f_k\left(\sum_{i=1}^n\varepsilon_i b_{i\ell}\right)=\sum_{i=1}^n b_{i\ell}f_k(\varepsilon_i)=b_{k\ell}$.\par
因此$a_{\ell k}=b_{k\ell}$，即$R^*_g=\left(R_{g}^{-1}\right)^T$.\par
有了反轭表示，实际上也有表示的共轭表示，定义如下.
\begin{definition}[共轭表示]
    设$(\rho,V)$是$G$的表示，$N\vartriangleleft G$，定义其\bluebf{共轭表示}为$\setjot[-2]\begin{aligned}[t]
        \rho_N:N&\to GL(V) \\
        n &\mapsto \rho(g^{-1}ng)
    \end{aligned}$
\end{definition}
事实上，上述表示可以视作现将$G$的表示和$N\to G$的群同态的复合，因此上述定义可以不依赖于正规子群和共轭作用.\par
\begin{definition}[表示的提升]
    设$(\rho,V)$是$G_1$的表示，$f$是$G_2\to G_1$的群同态，则$\eta=\rho\circ f:G_2\to GL(V)$是$G_2$的表示，称为\bluebf{表示的提升}.
\end{definition}
\begin{instance}
    (1) 若$H\subseteq G$，则自然有$H\xlongrightarrow{p}G\xlongrightarrow{\rho}GL(V)$，其中$p:H\to G$是嵌入.\par
    (2) 设$N\vartriangleleft G$，则有$G\to G/N \xlongrightarrow{\rho} GL(V)$.为了保证良定义，需要$N\subseteq \ker p$.
\end{instance}

\section{表示的张量积}
线性空间还有一种特殊的运算---张量积，对于线性空间的张量积，我们有以下三个等价定义.\par
\begin{definition}[线性空间的张量积]
    \begin{enumerate}[label=\textbf{定义\arabic*},leftmargin=3.5em]
        \item 设$f:V_1\times V_2\to W$是双线性映射，$V_1$的基为$\{\varepsilon_i\}$，$V_2$的基为$\{\eta_j\}$，则由$\{f(\varepsilon_i,\eta_j)\}$张成的空间$W$称为$V_1$和$V_2$的\bluebf{张量积}，记为$W=V_1\otimes V_2$.
        \item 设$V_1,V_2$是线性空间，$(a+a',b)-(a,b)-(a,b'),(a,b+b')-(a,b)-(a,b'),(ka,b)-k(a,b),(a,kb)-k(a,b),\forall\, a,a'\in V_1,b,b'\in V_2,k\in F$张成$V_1\times V_2$的子空间$K$，则定义$V_1\otimes V_2\triangleq F(V_1\times V_2)/K$.
        \item 线性空间$V_1,V_2$的张量积$V_1\otimes V_2$是一个配备双线性映射$f:V_1\times V_2\to V_1\otimes V_2$的线性空间，且满足：对任意双线性映射$h:V_1\times V_2\to V$，存在唯一一个线性映射$\tilde{h}:V_1\times V_2\to V$，使得$h=\tilde{h}\circ f$.\par
        \begin{tikzcd}[labels={font=\normalsize}]
V_1\times V_2 \arrow[dd, "f"'] \arrow[rr, "{\forall\, h}"]    &  & V \\
                                                              &  &   \\
V_1\oplus V_2 \arrow[rruu, "{\exists\, !\tilde{h}}"', dashed] &  &  
\end{tikzcd}
    \end{enumerate}
\end{definition}
上述的定义3就是著名的张量积的泛性质.前两者的等价性是很容易验证的，要验证定义3与前两者的等价性，我们需要验证两点，即定义1和定义2中的$V_1\otimes V_2$具有泛性质以及若$W$满足泛性质，$W\cong V_1\otimes V_2$.

\begin{lemma}[]
    设$W_1,W_2$均满足上述泛性质，则$W_1\cong W_2$.
\end{lemma}
\begin{proof}
\begin{tikzcd}[labels={font=\normalsize}]
V_1\times V_2 \arrow[rr, "h"] \arrow[dd, "f"']       &  & W_2 \\
                                                     &  &     \\
W_1 \arrow[rruu, "{\exists\, ! \tilde{h}}"', dashed] &  &    
\end{tikzcd}
\begin{tikzcd}[labels={font=\normalsize}]
V_1\times V_2 \arrow[rr, "f"] \arrow[dd, "h"']       &  & W_1 \\
                                                     &  &     \\
W_2 \arrow[rruu, "{\exists\, ! \tilde{f}}"', dashed] &  &    
\end{tikzcd}
\begin{tikzcd}[labels={font=\normalsize}]
V_1\times V_2 \arrow[rr, "f"] \arrow[dd, "f"']       &  & W_1 \\
                                                     &  &     \\
W_1 \arrow[rruu, "{\exists\, ! \mathrm{id}_{W_1}}"', dashed] &  &    
\end{tikzcd}\par
由上述交换图，有$h=\tilde{h}f,f=\tilde{f}h=\tilde{f}\tilde{h}f$.\par
因此，由第三张交换图可知$\tilde{f}\tilde{h}=\mathrm{id}_{W_1}$.同理，$\tilde{h}\tilde{f}=\mathrm{id}_{W_1}$.因此$W_1\cong W_2$.
\end{proof}

对于张量空间$V_1\otimes V_2$，其中一个特点在于$av_1\otimes v_2=v_1\otimes (av_2)=a(v_1\otimes v_2)$，表明域的作用在张量积上是相容的，可以随意移动的.\par
下面需要定义表示的张量积，首先需要先定义线性变换的张量积.\par
\begin{definition}[线性变换的张量积]
    设$f_1\otimes f_2\in \End\left(V_1\otimes V_2\right)=\left\{f:V_1\otimes V_2\to V_1\otimes V_2\right\}$，$\left(f_1\otimes f_2\right)\left(v_1\otimes v_2\right)\triangleq f_1(v_1)\otimes f_2(v_2)$.
\end{definition}
线性映射往往会对应矩阵，如果$f_1$和$f_2$分别对应矩阵$R_1=(a_{ij})_{n\times n},R_2=(b_{ij})_{m\times m}$，则$f_1\otimes f_2$对应的矩阵为$R_1\otimes R_2$，由此就有矩阵的张量积.\par
$\setjot[-2]\begin{aligned}[t]
    (f_1\otimes f_2)(\varepsilon_i\otimes \eta_j) &= f_1(\varepsilon_i)\otimes f_2(\eta_j)=\sum_{k=1}^n\varepsilon_ka_{ki}\otimes\sum_{\ell=1}^n\eta_\ell b_{\ell j} \\
    &=\sum_{k=1}^n\sum_{\ell=1}^m\left(a_{ki}b_{\ell j}\right)\varepsilon_k\otimes \eta_\ell
\end{aligned}$\par
因此如果我们选定基的顺序，就可以得到$R_1\otimes R_2$的矩阵.一般来说，我们定义两个矩阵的张量积，即\bluebf{Kronecker积}为$R_1\otimes R_2=\begin{pmatrix}
a_{11}R_2 & a_{12}R_2 & \cdots & a_{1n}R_2 \\
a_{21}R_2 & a_{22}R_2 & \cdots & a_{2n}R_2 \\
\vdots & \vdots & \ddots & \vdots \\
a_{n1}R_2 & a_{n2}R_2 & \cdots & a_{nn}R_2
\end{pmatrix}$.\par
下一步就可以自然地定义表示的张量积了.
\begin{definition}[表示的张量积]
    设$(\rho_1,V_1)$和$(\rho_2,V_2)$是$G$的表示，则定义其张量积为$\setjot[-1]\begin{aligned}[t]
        \rho_1\otimes \rho_2:G&\to GL(V_1\otimes V_2) \\
        g&\mapsto (\rho_1\otimes\rho_2)(g)\triangleq \rho_1(g)\otimes\rho_2(g)
    \end{aligned}$
\end{definition}
由于$\rho_1(g)$和$\rho_2(g)$都是线性变换，因此最后的$\rho_1(g)\otimes\rho_2(g)$就是之前定义的线性变换的张量积.\par
介绍表示的张量积的一个重要作用就是可以利用张量积将之前介绍的反轭表示和线性映射联系起来.
\begin{theorem}[]
    $V^*\otimes W\cong \Hom_F(V,W)$作为表示是同构的
\end{theorem}
\begin{proof}
    定义$\setjot[-6]\begin{aligned}[t]
        \varphi:V^*\otimes W&\to \Hom_F(V,W) \\
        f\otimes w&\mapsto \varphi\left(f\otimes w\right):\setjot[-2]\begin{aligned}[t]
            V&\to W\\
            v&\mapsto f(v)w
        \end{aligned}
    \end{aligned}$
    \begin{enumerate}[label=(\roman*),leftmargin=3.5em]
        \item \textbf{\boldmath $\varphi$是单射}\par
        设 $x \in \ker\varphi$，设$x = \sum_{i=1}^k f_i \otimes w_i$.\par
        不失一般性，我们可以选取 $\{w_1, w_2, \dots, w_k\}$ 为 $W$ 中一组线性无关的向量.\par
        因为 $x \in \ker\varphi$，所以$\forall\,v \in V$，$\varphi(x)(v) = \varphi\left(\sum_{i=1}^k f_i \otimes w_i\right)(v) = \sum_{i=1}^k f_i(v)w_i = 0$.\par
        由于$\{w_1, \dots, w_k\}$是线性无关的.因此，$\forall\,v \in V,\forall\,i \in \{1, \dots, k\}$，$f_i(v) = 0$.\par
        这意味着对于每一个 $i$，$f_i$ 都是 $V^*$ 中的零泛函，即 $f_i = 0$.\par
        故$x = \sum_{i=1}^k 0 \otimes w_i = 0$，所以$\ker\varphi = \{0\}$，即 $\varphi$ 是单射.
        \item \textbf{\boldmath $\varphi$是满射}\par
        $\dim(V^*\otimes W)=\dim(V^*)\dim W=\dim V\dim W =\dim(\Hom_F(V,W))$.\par
        又$\varphi$是单射，故$\varphi$是满射.
        \item \textbf{\boldmath $\varphi$是$G$-模同态}\par
        设$\rho_1^*,\rho_2,\rho,\rho_{\Hom}$分别是$G$在$V^*,W,V^*\otimes W,\Hom_F(V,W)$上的表示.则需要验证$\forall\, g\in G,f\in V^*,w\in W,v\in V,\varphi(\rho(g)(f \otimes w))(v) = (\rho_{\Hom}(g)\varphi(f \otimes w))(v)$.\par
        对于左边，根据表示的张量积定义，$\rho(g)(f \otimes w) = \rho_1^*(g)(f) \otimes \rho_2(g)(w)$.故$\varphi(\rho(g)(f \otimes w))(v) = \varphi(\rho_1^*(g)(f) \otimes \rho_2(g)(w))(v) = (\rho_1^*(g)(f))(v) \cdot (\rho_2(g)(w))$.\par
        根据反轭表示的定义，$\varphi(\rho(g)(f \otimes w))(v)(\rho_1^*(g)(f))(v) = f(\rho_1(g^{-1})(v))$.\par
        因此$\varphi(\rho(g)(f \otimes w))(v)=f(\rho_1(g^{-1})(v)) \cdot (\rho_2(g)(w))$.\par
        对于右边，根据$\Hom_F(V,W)$上表示的定义，$(\rho_{\Hom}(g)\varphi(f \otimes w))(v) = \rho_2(g)(\varphi(f \otimes w)(\rho_1(g^{-1})(v)))$.\par
        根据$\varphi$的定义，$\varphi(f \otimes w)(\rho_1(g^{-1})(v)) = f(\rho_1(g^{-1})(v)) \cdot w$.\par
        因此$(\rho_{\Hom}(g)\varphi(f \otimes w))(v) = \rho_2(g)(f(\rho_1(g^{-1})(v)) \cdot w) = f(\rho_1(g^{-1})(v)) \cdot (\rho_2(g)(w))$.\par
    \end{enumerate}

    综上所述，$\varphi$是双射的$G$-模同态，故$V^*\otimes W\cong \Hom_F(V,W)$作为表示是同构的.
\end{proof}
