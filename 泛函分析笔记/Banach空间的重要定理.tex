\chapter{Banach空间的重要定理}


\section{超限归纳法与 Zorn 引理}

在正式介绍诸多定理之前，由于要讨论无限维赋范线性空间的性质，因此需要引入一些集合论的工具，以便处理无限的情景.

\begin{definition}[偏序关系]\index{p@偏序关系}
    设 $S$ 是一个集合，$\lesssim$ 是 $S$ 上的偏序关系，即满足：
    \begin{itemize}
        \item 自反性：$x \lesssim x$.
        \item 传递性：$x \lesssim y, y \lesssim z \Rightarrow x \lesssim z$.
        \item 反对称性：$x \lesssim y, y \lesssim x \Rightarrow x = y$.
    \end{itemize}
\end{definition}

集合 $T \subseteq S$ 是\bluebf{链}\index{l@链}（全序集），若 $\forall\, x, y \in T$，要么 $x \lesssim y$，要么 $y \lesssim x$.\par
集合 $B \subseteq (S, \lesssim)$ 有\bluebf{上界}\index{s@上界} $u \in S$，若 $\forall\, x \in B, x \lesssim u$.\par
称 $m \in S$ 为\bluebf{极大元}\index{j@极大元}，若 $\forall\, y \in S, m \lesssim y \Rightarrow m = y$.\par

基于以上定义，我们可以引入 Zorn 引理，它可以用来证明很多集合的存在性问题.
\begin{lemma}[Zorn 引理]\index{z@Zorn引理}
    设 $(S, \lesssim)$ 是一个偏序集，若 $S$ 的每个链都有上界，则 $S$ 中存在极大元.
\end{lemma}

\begin{remark}
    事实上：Zorn 引理 $\iff$ 选择公理 $\iff$ 良序原理 $\iff$ Tychonoff 定理.\par
    Zorn 引理 $\Rightarrow$ Hahn-Banach 定理.\par
    Gödel 在 1938 年证明了选择公理独立于 ZF 公理体系.
\end{remark}

在Zorn引理的框架下，可以证明很多有意思的结论，例如Hamel证明了以下两个结论：
\begin{enumerate}[label=(\arabic*),leftmargin=2em]
    \item  $\mb{R}$ 有 $\mb{Q}$ 上的基；
    \item  $\exists\, f \colon \mb{R} \to \mb{R}$，满足 $f(x+y)=f(x)+f(y)$，但 $f(x) \neq cx, \forall\, c$.
\end{enumerate}\par
上面的第一个结论就是Hamel基的最初始形态，而Hausdorff才真正证明了Hamel基的存在性.

\begin{theorem}[Hausdorff, 1932]\index{h@Hamel基}
    每个非零线性空间都有一组Hamel基.
\end{theorem}

\begin{proof}
    令 $\mcal{S}$ 是 $V$ 中所有线性无关子集的集合族.定义 $\mcal{S}$ 上的偏序$\lesssim$为
    \begin{equation*}
        L \lesssim L' \iff L \subseteq L'.
    \end{equation*}\par
    设 $\mcal{C}=\{L_\alpha\}_{\alpha \in I}$ 是 $\mcal{S}$ 的一个全序子集，则 $L = \bigcup_{\alpha\in I} L_\alpha$ 是 $\mcal{S}$ 的一个上界.\par
    为了说明 $L \in \mcal{S}$，任取 $\{v_1, \dots, v_n\} \in L$，则 $\forall\, i \in \{1, \dots, n\}, \exists\, \alpha_i \in I$, s.t. $v_i \in L_{\alpha_i}$.\par
    由于 $\mcal{C}=\{L_\alpha\}_{\alpha \in I}$ 是全序集，因此 $\exists\, j \in \{1, \dots, n\},\st L_{\alpha_j}\gtrsim L_{\alpha_i}, \forall\, i \in \{1, \dots, n\}$.\par
    $\Rightarrow \{v_1, \dots, v_n\} \in L_{\alpha_j}$.由 $L_{\alpha_j}$ 线性无关知 $v_1, \dots, v_n$ 线性无关.\par
    因此 $L$ 线性无关，故 $L \in \mcal{S}$.\par
    由 Zorn 引理，存在极大元 $\mcal{B} \in \mcal{S}$，且 $\mcal{B}$ 线性无关.\par
    下证 $V = \mathrm{span}(B)$.\par
    反证，若不然，则 $\exists\, v \in V \setminus \spann(\mcal{B})$，考虑 $B \cup \{v\}$.\par
    取 $\{v_1, \dots, v_n\} \subset B \cup \{v\},\st \sum_{i=1}^n c_i v_i = 0$.\par
    \begin{itemize}
        \item 若所有的 $v_i$ 均在 $\mcal{B}$ 中，由于 $\mcal{B} \in \mcal{S}$，则所有的 $c_i$ 均为 $0$.
        \item 若其中一个 $v_i$（不妨设为 $v_1$）为 $v$，则 $c_1 v + \sum_{j=2}^n c_j v_j = 0 \Rightarrow v = \sum_{j=2}^n -\frac{c_j}{c_1} v_j \Rightarrow v \in \mathrm{span}(B)$，矛盾！ \qedhere
    \end{itemize}
\end{proof}

下面可以利用Hamel基简单证明 Hamel 的第 2 个结论：\par
将 $\mb{R}$ 视为 $\mb{Q}$ 上的向量空间，则 $\{1, \sqrt{2}\}$ 线性无关.\par
通过上述定理，可将 $\{1, \sqrt{2}\}$ 延拓为一个 Hamel 基 $\{e_i\}_{i \in I}$，s.t. $e_1 = 1, e_2 = \sqrt{2}$.\par
定义 $f(1) = \sqrt{2}, f(\sqrt{2}) = 1, f(e_i) = e_i, \forall\, e_i \neq 1, \sqrt{2}$.\par
利用 $\mb{Q}-\!$线性关系将 $f$ 延拓为 $f \colon \mb{R} \to \mb{R}$.\par

接下来简单介绍序数的概念.
\begin{instance}
    $\{0, 1, 2, \dots, 12\}$ 的序数为 $13$.\par
    $\mb{N}$ 的序数为 $\omega$.\par
    $\mb{Z}$ 的序数为 $\omega + \omega$.
\end{instance}

\begin{definition}
    设$P, Q$是序数为$\alpha, \beta$ 的良序集，则定义其序数的序关系为
    \begin{equation*}
        \alpha < \beta \iff P\,\text{是}\, Q \,\text{前面的片断}\iff\,\text{对某个}\,b\in\mb{Q},\exists\, \text{同构}\,\Phi:P \to \{x \in Q :x < b\}.
    \end{equation*}
\end{definition}

因此对于序数，同构是保序的双射.所有的序数构成的集合也是一个良序集.\par
由上述定义，定义$\mb{N}$ 的序数为 $\omega = \aleph_0$，$\mb{R}$ 的序数为 $2^{\aleph_0} = \aleph_1$.著名的关于$\aleph_0$和$\aleph_1$的命题是
\begin{assumption}[连续统假设, CH]\index{l@连续统假设}
    不存在序数在 $\aleph_0$ 与 $\aleph_1$ 之间的集合.
\end{assumption}
Gödel 在 1938 年证明了连续统假设在 ZFC 中不可证.Cohen 在 1963 年更进一步证明了连续统假设独立于 ZFC.因此关于连续统假设，我们无法得知它的真值.\par
基于序数理论，我们可以推广自然数中的数学归纳法为序数中的有超限归纳法\index{c@超限归纳法}：若 $[(\forall\, \beta < \alpha) P(\beta)] \to P(\alpha)$，则 $P(\alpha), \forall\, \alpha$.

在超限归纳法的框架下，可以证明下述定理.
\begin{theorem}
    $\exists\, A \subseteq \mb{R}$，使得 $A$ 与 $\mb{R} \setminus A$ 与 $\mb{R}$ 中每个不可数闭集相交于至少一点.
\end{theorem}
\begin{proof}
    存在 $\aleph_1$ 个不同的不可数闭集，设为 $\{F_\alpha \colon \alpha < \aleph_1\}$是上述集合的集合族，则其上良序.\par
    每个 $F_\alpha$ 的基数为 $\aleph_1$.\par
    设 $x_\alpha, y_\alpha$ 是 $F_\alpha$ 中任意两点.在第 $\alpha$ 步，可以取 $x_\alpha \neq y_\alpha$，使得它们和任意 $x_\beta, y_\beta, \beta < \alpha$ 不相等.\par
    设 $\mcal{A} = \{x_\alpha \colon \alpha < \aleph_1\}$.由构造可知，$A \cap F_\alpha \neq \varnothing, \forall\, \alpha$.\par
    但 $\mb{R} \setminus A$ 包含了所有的 $y_\alpha, \forall\, \alpha < \aleph_1, y_\alpha \in F_\alpha \Rightarrow (\mb{R} \setminus A) \cap F_\alpha \neq \varnothing, \forall\, \alpha$.
\end{proof}




\section{Hahn-Banach 定理}


设$X$是一个向量空间，$X'=\mcal{L}\left(X,\mb{K}=\mb{R}\,\text{或}\,\mb{C}\right)$为其代数对偶.若$X$是一个赋范线性空间，则$X^*=\mcal{B}(X,\mb{K})$.一个重要的观察是，若$Y\leq X$，$f\in X^*$，则$\|f|_Y\|_{Y^*}\leq\|f\|_{X^*}$.\par
我们知道，在线性空间中，若有一个子空间上的函数，可以通过将子空间上的基底扩展为整个空间的基底来将函数延拓到整个空间.自然地，当整个空间被赋予范数后，若$Y< X$，且有一个$f\in Y^*$，还能否将$f$延拓到$F\in X^*$，即$F|_Y=f$.\par
在这个问题中，我们主要聚焦于次线性泛函这一特殊的函数.\par
\begin{definition}[次线性泛函]\index{c@次线性泛函}
    设$X$是一个实线性空间，称$P:X\to\mb{R}$是一个\colorbf{次线性泛函}，如果其满足：
    \begin{enumerate}[label=(\roman*)]
        \item $\forall\, x,y\in X,P(x+y)\leq P(x)+P(y)$
        \item $\forall\, x\in X,\lambda\geq 0,P(\lambda x)=\lambda P(x)$
    \end{enumerate}
\end{definition}
我们之前定义的范数、半范数实际都是一种次线性泛函.在次线性泛函的框架下，关于上述问题，分析学家给出的解答是：

\begin{theorem}[Hahn-Banach 延拓定理]\index{h@Hahn-Banach定理}
    设 $X$ 是实线性空间，$Y \leq X$ 是子空间，$p \colon X \to \mb{R}$ 是次线性泛函.若 $f \in Y'$ 满足 $\forall\, x \in Y,f(x) \leq p(x)$，则存在 $F \in X'$ 使得 $F|_Y = f$ 且 $\forall\, x \in X,F(x) \leq p(x)$.
\end{theorem}
\begin{proof}
    首先，考虑 $x_0 \notin Y$，将 $f \in Y^*$ 延拓到 $\tilde{f} \in (\langle x_0 \rangle \oplus Y)^*$，其中 $\langle x_0 \rangle = \mathrm{span}\{x_0\}$.\par
    由于$\tilde{f}(y + \alpha x_0) = \tilde{f}(y) + \alpha \tilde{f}(x_0) = f(y) + \alpha \tilde{f}(x_0)$，因此只需定义 $\tilde{f}(x_0)$.\par
    由于 $f(y) + \alpha \tilde{f}(x_0) = \tilde{f}(y + \alpha x_0) \leq p(y + \alpha x_0)$：
    \begin{itemize}
        \item 若 $\alpha = 0$，则 $f(y) \leq p(y)$，已成立.
        \item 若 $\alpha > 0$，则\begin{align*}
            &\tilde{f}(x_0) \leq \frac{p(y + \alpha x_0) - f(y)}{\alpha} = p\left(\frac{y}{\alpha} + x_0\right) - f\left(\frac{y}{\alpha}\right), \forall\, y \in Y, \alpha > 0 \\
            &\Longleftrightarrow\tilde{f}(x_0) \leq p(y' + x_0) - f(y'), \forall\, y' \in Y.
        \end{align*}
        \item 若 $\alpha < 0$，则\begin{align*}
            &\tilde{f}(x_0) \geq \frac{p(y - (-\alpha) x_0) + f(y)}{-\alpha} = -p\left(\frac{y}{-\alpha} - x_0\right) + f\left(\frac{y}{-\alpha}\right), \forall\, y \in Y, \alpha < 0  \\
            &\Longleftrightarrow \tilde{f}(x_0) \geq f(y'') - p(y'' - x_0), \forall\, y'' \in Y.
        \end{align*}
    \end{itemize}\par
    综合以上，即需满足 $\sup_{\tilde{y} \in Y} [f(\tilde{y}) - p(\tilde{y} - x_0)] \leq \tilde{f}(x_0) \leq \inf_{y \in Y} [p(y + x_0) - f(y)]$.\par
    即
    \begin{equation*}
        \forall\, y,\tilde{y} \in Y, f(\tilde{y}) - p(\tilde{y} - x_0) \leq p(y + x_0) - f(y).
    \end{equation*}\par
    但 $p(y + x_0) + p(\tilde{y} - x_0) - f(y) - f(\tilde{y}) \geq p(y + \tilde{y}) - f(y + \tilde{y}) \geq 0$，故可以定义这样的 $\tilde{f}(x_0)$.\par
    因此可以作一步延拓 $\tilde{f} \in \left(\langle x_0 \rangle \oplus Y\right)^*,\st \tilde{f}|_Y = f$.\par
    对任意余维数，定义 
    \begin{equation*}
            \mcal{G} = \left\{\left(\widetilde{Y}, \tilde{f}\right) : Y \leq \widetilde{Y} \leq X, \tilde{f} \in (\widetilde{Y})^*, \text{s.t. } \tilde{f} \leq p, \tilde{f}\text{ 是 }\,f\text{ 的延拓}\right\}.
    \end{equation*}
    定义其上的偏序为：$(\widetilde{Y}_1, \tilde{f}_1) \lesssim (\widetilde{Y}_2, \tilde{f}_2) \iff \widetilde{Y}_1 \leq \widetilde{Y}_2, \tilde{f}_2|_{\widetilde{Y}_1} = \tilde{f}_1$.\par
    设 $\mcal{C}=\left\{(\widetilde{Y}_\alpha, \tilde{f}_\alpha)\right\}_{\alpha \in I}$ 是 $\mcal{G}$ 的全序子集.\par
    则 $\widetilde{Y} = \bigcup_{\alpha \in I} \widetilde{Y}_\alpha, \tilde{f} \in (\widetilde{Y})^*,\st \tilde{f}(y) = f_\alpha(y), y \in Y_\alpha$ 为$\mcal{C}$的一个上界.\par
    因此，由 Zorn 引理，$\mcal{G}$ 有极大元 $(\widetilde{Y}_*, \tilde{f}_*)$.\par
    下证 $\widetilde{Y}_* = X$.\par
    反证，若 $\exists\, x_* \in X \setminus \widetilde{Y}_*$.\par
    考虑$\widetilde{Y}_*\oplus \langle x_*\rangle$，由前述一步延拓，$\exists\, \tilde{f} \in (\widetilde{Y}_* \oplus \langle x_* \rangle)^*,\st\tilde{f}|_{\widetilde{Y}_*} = \tilde{f}_*$.\par
    但 $(\widetilde{Y}_*, \tilde{f}_*) \lnsim(\widetilde{Y}_* \oplus \langle x_* \rangle, \tilde{f})$，与$\left(\widetilde{Y}_*,\tilde{f}_*\right)$最大矛盾.
\end{proof}

由Hahn-Banach定理，就可以把子空间上的实线性泛函保范地延拓到整个空间上.
\begin{corollary}
    设 $X$ 是实赋范线性空间，$Y \leq X, f \in Y^* = \mcal{B}(Y, \mb{R})$，则 $\exists\, F \in X^*,\st F|_Y = f$ 且 $\|F\| = \|f\|$.
\end{corollary}
\begin{proof}
    设 $p(x) := \|f\|_{Y^*} \|x\|_X$ 是一个范数$(f \neq 0)$.\par
    注意到我们有 $f(y) \leq p(y)$.\par
    由 Hahn-Banach 定理，$\exists\, F \in X^*$, s.t. $F|_Y = f, F(x) \leq p(x) = \|f\| \|x\|$.\par
    使用$-x$代换得，
    \begin{equation*}
    F(-x) = -F(x) \leq \|f\| \|-x\| \Rightarrow F(x) \geq -\|f\| \|x\|.
    \end{equation*}
    故 $|F(x)| \leq \|f\| \|x\| \Rightarrow \|F\| \leq \|f\|$.\par
    又 $\|F\| \geq \|f\|$ 显然，因此 $\|F\| = \|f\|$.
\end{proof}

\begin{remark}
    注意：Hahn-Banach 延拓仅作用在 $F \colon X \to \mb{R}$ 上，而不是对其它赋范线性空间 $F \colon X \to Z$.
\end{remark}

\begin{theorem}[Philips, 1948]
    不存在 $\ell^\infty \to c_0$ 的有界线性投影.
\end{theorem}
等价地，$\mathrm{Id} \colon c_0 \to c_0$ 不可被延拓为 $\ell^\infty \to c_0$ 的有界线性算子.这就说明了Hahn-Banach定理对其无效.\par
事实上：设 $X$ 是一个 Banach 空间，如果对任意的闭子空间 $Y \leq X$，所有的投影 $X \to Y$ 都有界，则 $X$ 同构于 Hilbert 空间 (J. Lindenstrauss - Tzafriri).\par

上述 Hahn-Banach 定理对复数域依旧成立，但需要将次线性泛函加强为半范数.

\begin{theorem}[复 Hahn-Banach 延拓定理]\index{h@Hahn-Banach定理!f@复数域}
    设 $X$ 为复线性空间，$Y \leq X$ 是子空间且 $p\colon X \to \mb{R}$ 是半范数，若 $f \in Y^*$ 满足 $|f(x)| \leq p(x)$，则 $\exists\, F \in X^*$，$\st F|_Y = f$，$F \leq p$.
\end{theorem}

下面来看更多 Hahn-Banach 定理的应用.

\begin{instance}
    1. 设 $X$ 是 $\mb{R}$ 上的赋范线性空间，则 $\forall\, x_0 \in X \setminus \{0\}$，$\exists\, f \in X^*$ $\st \|x_0\| = f(x_0)$，$\|f\|_{X^*} = 1$，称 $f$ 是 $X$ 的\bluebf{范数泛函}.\index{f@范数泛函}

    \begin{proof}
        在 $\langle x_0 \rangle = \mathrm{span}\{x_0\}$ 上定义 $f$：$f(\lambda x_0) = \lambda \|x_0\|$.
        通过 Hahn-Banach 定理将其延拓至 $X^*$ 上.
    \end{proof}
    \noindent
    2. 设 $X$ 是非零赋范线性空间.\par
    \begin{enumerate}[label=(\arabic*),leftmargin=2em]
        \item $X$上线性泛函的范数可以等价地定义如下：$\forall\,f \in X^*$，$\|f\|_{X^*} = \sup \{|f(x)|: x \in X, \|x\|_X = 1\}$.
        \item 对偶地，也可以利用有界线性泛函定义$X$中元素的范数：$\forall\, x \in X$，$\|x\|_X = \sup \{|f(x)|: f \in X^*, \|f\|_{X^*} = 1\}$.
    \end{enumerate}

    \begin{proof}
        对(2)：\par
        “$\geq$” 若 $f \in X^*$，$\|f\|_{X^*} = 1$，则 $|f(x)| \leq \|f\|_{X^*} \|x\|_X = \|x\|_X$.\par
        “$\leq$” 由 (1) 中范数泛函知可达到.
    \end{proof}
\end{instance}
对上述的第2个例子，不自然的点是，$X$中的元素可以使用$X^*$定义，但$X^*$也是用$X$定义而非再对$X^*$求对偶.在其余Banach空间，这种性质未必成立，我们必须研究二次对偶.

\begin{definition}[二次对偶]\index{e@二次对偶}
    赋范线性空间 $X$ 的\colorbf{二次对偶}为 $X^{**} = (X^*)^*$.
\end{definition}

一般地，$X \lneqq  X^{**}$，但 $X$ 可以自然嵌入 $X^{**}$.

\begin{definition}[典范等距嵌入]
    设 $X$ 是赋范线性空间，称
    \begin{align*}
        J = J_X: X &\to X^{**} = (X^*)^*, \\
        x &\mapsto (J(x))(f) = f(x),\forall\, f \in X^*,
    \end{align*}
    为\colorbf{典范等距嵌入}.\index{d@典范等距嵌入}\par
\end{definition}
事实上，我们可以证明$J_X$是单射，且它是等距的，即 $\|J(x)\|_{X^{**}} = \|x\|_X$.但是上述嵌入通常显然不是满射，即 $X \lneqq X^{**}$.
\begin{attention}
    即使有$X\cong X^{**}$，我们依旧无法得到$J_X$是满射，因此$J_X$不是等距同构，上述结论由 James 在 1940s 得到.他构造了一个使得二次对偶$J^{**}$和$J$同构的空间，但$J_X$的余维数为1，这个空间被称作 \bluebf{James空间}.
\end{attention}

由以上的讨论可知$J_X$的满射性质是十分特殊的，因此需要特别强调满足该性质的空间.
\begin{definition}[自反空间]\index{z@自反空间}
    如果 $J_X$ 是满射，则称 $X$ 是一个\colorbf{自反 Banach 空间}.
\end{definition}

\begin{instance}
    $(L^p)^*$ 自反 $\iff 1 < p < \infty$.\par
    事实上 $(L^p)^* \cong L^q$，其中 $\frac{1}{p} + \frac{1}{q} = 1$.
\end{instance}

\begin{instance}
    设 $X$ 是赋范线性空间，则 $X^* = \mcal{B}(X, \mb{R})$ 在 $X$ 上分离点，即 $\forall\, x \neq y \in X$，$\exists\, f \in X^*$，$\st f(x) \neq f(y)$.\par
    要证明上述命题只需要取 $f$ 为 $x-y$ 的范数泛函即可.
\end{instance}
从该例中可看出$X^*$中的元素极为丰富，至少能将$X$中的任意两个不同点分离.\par
另外，从几何层面来看，若 $X$ 有限维，则 $\ker f$是 $X$ 中的超平面，它将这两个点分到超平面的上下半空间.上述结论在无穷维依旧成立，此处的超平面定义为余维数为1的子空间.

\begin{lemma}[]
    设 $X$ 是赋范线性空间，$f \in X^* \setminus \{0\}$，则 $\forall\, x_0 \in X \setminus \ker f$，$X = \mathrm{span}(\ker f \cup \{x_0\})$.
\end{lemma}

\begin{proof}
    $\forall\, x \in X$，$x - \frac{f(x)}{f(x_0)} x_0 \in \ker f$.\par
    因此$\forall\, x\in X,\exists\, y\in\ker f,\st x=y+\lambda x_0$.其中$\lambda=\frac{f(x)}{f(x_0)}$.\par
    故$X = \mathrm{span}(\ker f \cup \{x_0\})$.
\end{proof}
由上述引理可知，$\mathrm{codim}(\ker f) := \dim (X / \ker f) = 1$，因此若 $f \in X^* \setminus \{0\}$，$\ker f$ 始终是一个超平面.\par
进一步，$X^*$ 分离点和闭子空间.\par
\begin{lemma}
设 $X$ 是 Banach 空间，设 $\{0\} \neq Y \lneq X$ 是闭子空间，$x_0 \in X \setminus Y$。则 $\exists\, f \in X^*$，$\st \|f\| = 1$，$f|_Y = 0$，$f(x_0) = \mathrm{dist}(x_0, Y) > 0$.
\end{lemma}
\begin{proof}
    令 $\delta = \mathrm{dist}(x_0, Y) > 0$，定义
    \begin{align*}
        g\colon Y \oplus \langle x_0 \rangle &\to \mb{R},\\[-4pt]
        (y + \lambda x_0) &\mapsto \lambda \delta,\forall\, y \in Y, \lambda \in \mb{R},
    \end{align*}
    而后利用 Hahn-Banach 定理将其延拓至 $f$.\par
    记$Z = Y \oplus \langle x_0 \rangle$，下证 $\|g\|_{Z^*} = 1$.\par
    \fbox{$\|g\|_{Z^*} \leq 1$}: $\forall\, \lambda \in \mb{R}$，$\forall\, y \in Y$，\begin{align*}
        |g(y + \lambda x_0)| &= |\lambda| \delta = |\lambda| \inf \left\{\left\|x_0 + \tilde{y}\right\| : \tilde{y} \in Y\right\}\\ 
        &= |\lambda| \inf \left\{\left\|x_0 + \frac{\tilde{y}}{\lambda}\right\| : \tilde{y} \in Y\right\} \\
        &\leq \|\lambda x_0 + y\|
    \end{align*}\par
    \fbox{$\|g\|_{Z^*} \geq 1$}: 即证 $\forall\, \varepsilon > 0$，$\|g\|_{Z^*} \geq 1 - \varepsilon$.\par
    $\forall\, \varepsilon > 0$，要找到 $z \in Z \setminus \{0\}$，$\st |g(z)| \geq (1 - \varepsilon)\|z\|$.\par
    事实上，可找到 $y \in Y$，使得 $\|x_0 - y\| < \delta + \eta$，$\eta > 0$.\par
    \begin{equation*}
        \frac{|g(x_0 - y)|}{\|x_0 - y\|} = \frac{\delta}{\|x_0 - y\|} > \frac{\delta}{\delta + \eta} \geq 1 - \varepsilon \Rightarrow \eta = \eta(\varepsilon) = \frac{\delta \varepsilon}{1 - \varepsilon}.
    \end{equation*}
    故 $x_0 - y$ 即为所要的 $z$.
\end{proof}

上述定理证明利用了留一个$\varepsilon-\!\!$空间的思想.如果直接证明不等号较为困难，则可以先证明它和相差$\varepsilon$的不等式成立，然后令$\varepsilon\to 0$.\par
在此基础上，线性泛函可以分离点，可以分离一个点与闭子空间，也可以分离两个凸集.\par
进一步地，引入 Hahn-Banach 分离定理：

\begin{theorem}[Hahn-Banach 分离定理]\index{h@Hahn-Banach定理!l@分离定理}
    设 $X$ 是 $\mb{R}$ 上的赋范线性空间，设 $A, B \subset X$ 是非空不交凸集，则：
    \begin{enumerate}[label=(\arabic*),leftmargin=2em]
        \item 若 $A$ 是开集，$\exists\, f \in X^*$，$\lambda \in \mb{R}$，$\st f(a) < \lambda \leq f(b)$，$\forall\, a \in A, b \in B$.
        \item 若 $A$ 是紧集，$B$ 是闭集，$\exists\, g \in X^*$，$c_1, c_2 \in \mb{R}$，$\st g(a) \leq c_1 < c_2 \leq g(b)$，$\forall\, a \in A, b \in B$.
    \end{enumerate}
\end{theorem}

\begin{definition}[吸收集]\index{x@吸收集}
    设 $X$ 是 赋范线性空间，$A \subset X$ 是\colorbf{吸收的} $\iff \forall\, x \in X, \exists\, t > 0, \st\, x \in tA$.
\end{definition}

上述的一个等价描述是 $\bigcup_{t > 0} tA = X$.

\begin{definition}[平衡集]\index{p@平衡集}
    设 $X$ 是 $\mb{R}$ 或 $\mb{C}$ 上的赋范线性空间，$B \subseteq X$ 是\colorbf{平衡的} $\iff \forall\, \lambda \in \mb{F}, \st\, |\lambda| \leq 1$, $\lambda B \subset B$.
\end{definition}

下面考虑吸收的，平衡的 $X$ 的凸集 $B$ 应该具有怎样的性质，最直观的感受是它应当类似于单位球.

\begin{definition}[Minkowski 泛函]\index{m@Minkowski泛函}
    设 $X$ 是赋范线性空间，$A \subset X$ 是吸收集，定义 
    \begin{align*}
        \mu_A\colon X &\to [0, \infty),\\[-2pt]
        x &\mapsto \inf \{t > 0 : x \in tA\},
    \end{align*}
    称为 \colorbf{Minkowski 泛函}.
\end{definition}

\begin{proposition}[]
    设 $p$ 是 $X$ 上的半范数，则 $\mcal{B} = \{p < 1\}$ 是 $0$ 的吸收的、平衡的凸邻域，且 $\mu_{\mcal{B}} = p$.
\end{proposition}

\begin{proof}
    \fbox{$\mcal{B}$ 是平衡的.}\par
    $x \in \mcal{B} \Rightarrow p(x) < 1 \Rightarrow p(\lambda x) = |\lambda| p(x) < |\lambda| \leq 1$, $\forall\, |\lambda| \leq 1$.\par
    即 $\lambda x \in \mcal{B}$，故 $\mcal{B}$ 是平衡集.\par
    
    \fbox{$\mcal{B}$ 是凸集.}\par
    $\forall\, x, y \in \mcal{B}$，$\theta \in (0, 1)$，$p(\theta x + (1-\theta)y) \leq \theta p(x) + (1-\theta)p(y) < \theta + 1 - \theta = 1$.\par
    $\Rightarrow \theta x + (1-\theta)y \in \mcal{B}$，故 $\mcal{B}$ 是凸集.\par

    \fbox{$\mcal{B}$ 是吸收的.}\par
    $\forall\, x \in X$，取 $s > p(x)$，则 $p\left(\frac{x}{s}\right) = \frac{1}{s} p(x) < 1 \Rightarrow \frac{x}{s} \in \mcal{B} $.\par
    $\Rightarrow x \in s\mcal{B}$，故 $\mcal{B}$ 是吸收的.\par

    \fbox{$\mu_{\mcal{B}} = p$}.\par
    由 $\mcal{B}$ 吸收可知 $\mu_{\mcal{B}} \leq p$.下证 $a > \mu_{\mcal{B}}(x) \Rightarrow a > p(x)$.\par
    证明其逆否命题：若 $a \leq p(x)$，则 $p\left(\frac{x}{a}\right) \geq 1 \Rightarrow x \notin a\mcal{B} \Rightarrow a \leq \mu_{\mcal{B}}(x)$.
\end{proof}

从一定角度来看，研究 $\mu_{\mcal{B}}$ 和 研究半范数 $p$ 是相同的.

\begin{proposition}[]
    若 $\mcal{A} \subset X$ 是吸收凸集，则 $\mu_\mcal{A}$ 是次线性泛函.\par
    若 $\mcal{A}$ 还是平衡的，则 $\mu_\mcal{A}$ 是半范数.
\end{proposition}

\begin{proof}
    先证若$\mcal{A}$是吸收凸集，则$\mu_\mcal{A}$ 是次线性泛函.\par
    \fbox{正齐次性} $\forall\, \alpha > 0$，由 Minkowski 泛函的定义：
    \begin{equation*}
        \mu_\mcal{A}(\alpha x) = \inf\{t > 0 : \alpha x \in t\mcal{A}\} = \inf\ab\{t > 0 : x \in \frac{t}{\alpha}\mcal{A}\}= \alpha \inf\{t > 0 : x \in t\mcal{A}\} = \alpha \mu_\mcal{A}(x).
    \end{equation*}
    又因 $\mcal{A}$ 吸收，$0 \in \mcal{A}$ 且 $\forall\, t>0$, $0 \in t\mcal{A}$，取下确界即得$\mu_\mcal{A}(0) = 0$.

    \fbox{三角不等式} $\forall\, x, y \in X$, $\varepsilon > 0$，令
    \begin{equation*}
        t = \mu_\mcal{A}(x) + \varepsilon,\quad s = \mu_\mcal{A}(y) + \varepsilon,
    \end{equation*}
    故 $x \in t\mcal{A}$, $y \in s\mcal{A}$. 由于 $\mcal{A}$ 是凸集，有
    \begin{equation*}
        \frac{x+y}{t+s} = \frac{t}{t+s} \cdot \frac{x}{t} + \frac{s}{t+s} \cdot \frac{y}{s} \in \frac{t}{t+s}\mcal{A} + \frac{s}{t+s}\mcal{A} \subset \mcal{A}.
    \end{equation*}
    这表明 $\frac{x+y}{t+s} \in \mcal{A}$，从而 $x+y \in (t+s)\mcal{A}$，由 Minkowski 泛函的定义：
    \begin{equation*}
        \mu_\mcal{A}(x+y) \leq t+s = \mu_\mcal{A}(x) + \mu_\mcal{A}(y) + 2\varepsilon.
    \end{equation*}
    令 $\varepsilon \to 0$，即得 $\mu_\mcal{A}(x+y) \leq \mu_\mcal{A}(x) + \mu_\mcal{A}(y)$.\par
    若 $\mcal{A}$ 还是平衡的，下证$\mu_\mcal{A}$ 是半范数.\par
    \fbox{绝对齐次性} 若 $\lambda = 0$，显然成立.\par
    若 $\lambda \neq 0$，考虑极分解 $\lambda = |\lambda| e^{i\theta}$. 由正齐次性，只需证 $\mu_\mcal{A}(e^{i\theta}x) = \mu_\mcal{A}(x)$.\par
    由 $\mcal{A}$ 的平衡性，对任意 $t > 0$，由于 $|e^{\pm i\theta}| = 1$，
    \begin{align*}
        x \in t\mcal{A} &\Rightarrow e^{i\theta}x \in t\,e^{i\theta}\mcal{A} \subset t\mcal{A} , \\[-5pt]
        e^{i\theta}x \in t\mcal{A} &\Rightarrow x \in t\,e^{-i\theta}\mcal{A} \subset t\mcal{A}.
    \end{align*}
    故 $\{t > 0 : x \in t\mcal{A}\} = \{t > 0 : e^{i\theta}x \in t\mcal{A}\}$，取下确界即得 $\mu_\mcal{A}(e^{i\theta}x) = \mu_\mcal{A}(x)$. 故
    \begin{equation*}
        \mu_\mcal{A}(\lambda x) = \mu_\mcal{A}(|\lambda| e^{i\theta}x) = |\lambda|\,\mu_\mcal{A}(e^{i\theta}x) = |\lambda|\,\mu_\mcal{A}(x).\qedhere
    \end{equation*}
\end{proof}

\begin{proof}[证明(Hahn-Banach 分离定理)]\par
    \hspace{-1.6 em}(1) 任取 $a_0 \in A, b_0 \in B$，令 $x_0 = b_0 - a_0$，由于 $A \cap B = \varnothing$，$x_0 \neq 0$.\par
    \begin{claim}{$\Omega = A - B + x_0 = (A - a_0) + (B - b_0)$ 是 $0$ 的凸开集，$\st x_0 \notin \Omega$.}
        \begin{itemize}
        \item $0 \in \Omega$.
        \item $\Omega = A - B + x_0 = \bigcup_{b \in B} A - (b - x_0)$ 为开集的并，故 $\Omega$ 为开集.
        \item 类似地，易证 $\Omega$ 是凸集.
        \item 若 $x_0 \in \Omega$，则 $0 \in A - B$，从而 $A \cap B \neq \varnothing$，故 $x_0 \notin \Omega$.\qedhere
    \end{itemize}
    \end{claim}

    \vspace{-1.3em}
    因此，$\Omega$ 是吸收集，则 $\mu_\Omega$ 是次线性泛函且 $\mu_\Omega(x_0) \geq 1$.\par
    构造 
    \begin{align*}
        f_0\colon \langle x_0 \rangle &\to \mb{R},\\[-6pt]
        \lambda x_0 &\mapsto \lambda, \forall\, \lambda \in \mb{R}.
    \end{align*}
    则 $f_0 \in \langle x_0 \rangle^*$且$f_0(x_0) = 1 \leq \mu_\Omega(x_0)$.\par
    由 Hahn-Banach 延拓定理，$\exists\, f \in X^*$，$\st f|_{\langle x_0 \rangle} = f_0$ 且在 $X$ 上 $f \leq \mu_\Omega$.\par
    显然，$\forall\, x \in X$，$f(x) = -f(-x) \geq -\mu_\Omega(-x)$.\par
    因此，若设 $\mathbb{B}_r(0) \subset \Omega$
    \begin{equation*}
        \|f\| = \sup \{|f(x)|: \|x\| \leq 1\} \leq \sup \{\mu_\Omega(x): \|x\| \leq 1\} \leq \frac{1}{r}.
    \end{equation*}
    由此$f \in X^*$.\par
    任取 $a \in A, b \in B$，
    \begin{equation*}
        f(a) - f(b) = f(a-b+x_0) - 1 \leq \mu_\Omega(a-b+x_0) - 1.
    \end{equation*}\par
    因为 $a-b+x_0 \in \Omega$，$\mu_\Omega(a-b+x_0) < 1$，所以 $f(a) - f(b) < 0$.\par
    $\Rightarrow f(a) < \lambda \equiv \sup_A f \leq f(b)$.\par
    \hspace{-1.6 em}(2) $A$ 紧，$B$ 闭，$A \cap B = \varnothing$.\par
    此时，$\delta = \mathrm{dist}(A, B) > 0$，构造 $\widetilde{A} = \left\{x \in X: \mathrm{dist}(x, A) < \frac{\delta}{2}\right\}$.\par
    则 $\widetilde{A}$ 是开集且 $\widetilde{A} \cap B = \varnothing$.\par
    应用 (1) 的结论即可.
\end{proof}

\section{Baire 纲定理}

下面进入一个和Hahn-Banach定理的表述似乎完全不同的定理.设 $X$ 是一个非空完备度量空间，我们想讨论下拓扑和测度意义下的集合大小.\par
\begin{center}
\begin{tabular}{|c|c|c|}
    \hline
    & 大 & 小 \\
    \hline
    拓扑 & 剩余集 & 贫乏集 \\
    &  & (第一纲集) \\
    \hline
    测度 & 全测集 & 零测集 \\
    \hline
\end{tabular}
\end{center}
在测度意义下，零测集是小的集合.与之相对的，其补集全测集是最大的集合.那么在拓扑意义下，该如何进行分类呢? Baire在其博士论文给出了详尽的分类.

\begin{definition}[无处稠密，第一纲集，第二纲集]\index{w@无处稠密}\index{d@第一纲集}\index{s@剩余集}\index{d@第二纲集}
    \begin{enumerate}[label=(\arabic*)]
        \item 设 $E \subset X$，如果 $\overline{E}$ 没有内点，称 $E$ \colorbf{无处稠密}.
        \item 称 $E$ 是\colorbf{贫乏集}或\colorbf{第一纲集}，当且仅当 $E = \bigcup_{i=1}^\infty N_i$，$N_i$ 无处稠密.
        \item $E \subset X$ 是\colorbf{剩余集}，当且仅当 $X \setminus E$ 是第一纲集.
        \item $E \subset X$ 是\colorbf{第二纲集}，当且仅当 $E$ 不是第一纲集.
    \end{enumerate}
\end{definition}

为了直观展示拓扑和测度两个角度下集合的大小可能会发生变化.可以考虑$\mb{Q}$和Cantor集$\mcal{C}$.\par
$\mb{Q}$是$\mb{R}$的稠密子集，因此$\overline{\mb{Q}}=\mb{R}$是第二纲集.$\mcal{C}$是闭集，因此$\overline{\mcal{C}}=\mcal{C}$，而$\mcal{C}$没有内点，因此$\mcal{C}$无处稠密，是第一纲集.但从测度意义上看，$\mb{Q}$可数，$\mcal{C}$不可数.这表明Baire纲是一个拓扑概念，和数量无关.

假设 $(X, P)$ 是概率测度空间，我们有两种方式定义一个概率很大的事件：
\begin{enumerate}[label=(\arabic*),leftmargin=2em]
    \item 一个典型事件是剩余集.
    \item 一个必然事件有 $P(E) = 1$.
\end{enumerate}
注意：它们通常从本质上不同.

\begin{instance}
    考虑 $X = [0, 1]^2$，设 $\{U_n\}_{n \in \mb{N}}$ 为布朗运动的轨道，$E = \bigcup_{n=1}^\infty U_n$ 是第一纲集，但 $P(X \setminus E) = 0$.
\end{instance}

\begin{lemma}
    $X \setminus \mathring{A} = \overline{X \setminus A}$，$X \setminus \overline{B} = \mathrm{int}(X \setminus B)$.
\end{lemma}
\begin{proof}
    \fbox{$X \setminus \mathring{A} \subseteq \overline{X \setminus A}$} $\forall\, x \in X \setminus \mathring{A}$，则 $\forall\, \varepsilon > 0$，$\exists\, y_\varepsilon \in B_\varepsilon(x) \setminus A$.\par
    则 $x$ 是 $y_\varepsilon$ 的一个聚点 $\Rightarrow x \in \overline{A^c} = \overline{X \setminus A}$.\par
    \fbox{$X \setminus \mathring{A} \supseteq \overline{X \setminus A}$} 设 $z \in \overline{X \setminus A}$，则 $\exists\, \{z_j\} \subset X \setminus A$，$\st z_j \to z$.\par
    则 $\forall\, \varepsilon > 0$，$B_\varepsilon(z)$ 包含某个 $z_j$，因此 $z \notin \mathring{A}$.
\end{proof}

下面介绍一些第一纲集和剩余集的性质.
\begin{property}
    \begin{enumerate}[label=(\arabic*),leftmargin=2em]
        \item 若 $E \subset X$ 是第一纲集，$F \subset E$，则 $F$ 是第一纲集.
        \item 第一纲集的可数并是第一纲集.
        \item 剩余集的可数交是剩余集.
        \item 稠密开集的可数交是剩余集.
    \end{enumerate}
\end{property}



\begin{proof}[证明 (性质 (4))]
    设 $\{\mathcal{O}_n\}_{n=1}^\infty$ 是 $X$ 中稠密开集.\par
    要证 $X \setminus \bigcap_{n=1}^\infty \mathcal{O}_n = \bigcup_{n=1}^\infty (X \setminus \mathcal{O}_n)$ 是第一纲集.\par
    下证 $X \setminus \mathcal{O}_n$ 无处稠密.\par
    由于 $\mathcal{O}_n$ 开，故 $\mathrm{int}(\overline{X \setminus \mathcal{O}_n}) = \mathrm{int}(X \setminus \mathcal{O}_n) = X \setminus \overline{\mathcal{O}_n} \xlongequal{\text{稠密}} X \setminus X = \varnothing$.
\end{proof}

\begin{theorem}[Baire 纲定理]\index{b@Baire纲定理}
    设 $X$ 是非空完备度量空间，$\{\mathcal{O}_n\}_{n=1}^\infty$ 是 $X$ 中一列稠密开集，则 $\bigcap_{n=1}^\infty \mathcal{O}_n$ 稠密.
\end{theorem}
\begin{proof}
    只需证明对任意开球 $B_0 = \mathbb{B}_{\varepsilon}(x)$，$\bigcap_{n=1}^\infty \mathcal{O}_n \cap B_0 \neq \varnothing$.\par
    由于 $\mathcal{O}_1$ 是稠密开集，$\exists\, \delta_1 > 0, x_1 \in X$，$\st \overline{\mathbb{B}_{\delta_1}(x_1)} \subset B_0 \cap \mathcal{O}_1$.\par
    设 $B_1 = \mathbb{B}_{\delta_1}(x_1)$，它在 $B_0 \cap \mathcal{O}_1$ 内部.\par
    类似地，$\exists\, B_2 = \mathbb{B}_{\delta_2}(x_2)$，$\st \overline{\mathbb{B}_{\delta_2}(x_2)} \subset B_1 \cap \mathcal{O}_2 \subset B_0 \cap \mathcal{O}_1$.\par
    以此类推，可取 $\{B_j\}$，$\st \overline{B_j} \subset \overline{B_{j-1}} \cap \mathcal{O}_j$，$\forall\, j \in \mb{N}$，且要求 $\mathrm{rad}(B_j) < \frac{1}{j}$.\par
    因此有 $\overline{B_0} \supset \overline{B_1} \supset \overline{B_2} \supset \cdots$.\par
    则 $\{x_j\}$ 构成 Cauchy 列，因此收敛到 $x_* \in X$.\par
    于是 $x_* \in \bigcap_{n=1}^\infty \mathcal{O}_n \cap B_0$.
\end{proof}



下面的推论更直观地展示了剩余集的结构，它表明剩余集包含一个稠密的 $G_\delta$ 集.因此，剩余集是非常大的集合.
\begin{corollary}
    设 $X$ 是非空完备度量空间，则 $E \subset X$ 是剩余集当且仅当 $E$ 包含一个稠密的 $G_\delta$ 集.
\end{corollary}
\begin{proof}
    \fbox{$\Leftarrow$} 设 $E \supset \bigcap_{n=1}^\infty \mathcal{O}_n$，其中 $\bigcap_{n=1}^\infty \mathcal{O}_n$ 稠密，$\mathcal{O}_n$ 为稠密开集.\par
    由性质 (4)，$\bigcap_{n=1}^\infty \mathcal{O}_n$ 是剩余集，从而 $E$ 是剩余集.\par
    \fbox{$\Rightarrow$} 设 $E \subset X$ 是剩余集，则 $X \setminus E = \bigcup_{n=1}^\infty M_n$，其中 $M_n$ 无处稠密.\par
    因此 $E = X \setminus \bigcup_{n=1}^\infty M_n = \bigcap_{n=1}^\infty (X \setminus M_n) \supset \bigcap_{n=1}^\infty (X \setminus \overline{M_n})$.\par
    \begin{claim}{$X \setminus \overline{M_n}$ 稠密.}
    事实上，$\overline{X \setminus \overline{M_n}} = X \setminus \mathrm{int}(\overline{M_n}) = X \setminus \varnothing = X$，故 $X \setminus \overline{M_n}$ 稠密.
    \end{claim}\par
    由 Baire 纲定理，$\bigcap_{n=1}^\infty (X \setminus \overline{M_n})$ 稠密.
\end{proof}

另一个推论表明，无处稠密集的闭包没有内部，而作为它的可数并，第一纲集没有内部，因此第一纲集是非常小的集合(至少不能包含任意区间).
\begin{corollary}
    设 $X$ 是非空完备度量空间，则 $X$ 中任意第一纲集没有内部.
\end{corollary}
\begin{proof}
    若 $M = \bigcup_{n=1}^\infty E_n$，$E_n$ 无处稠密，则 $X \setminus M = \bigcap_{n=1}^\infty (X \setminus E_n)$.\par
    $E_n$ 无处稠密 $\Rightarrow X \setminus E_n$ 是剩余集 $\Rightarrow X \setminus E_n$ 包含稠密 $G_\delta$ 集.\par
    则 $\bigcap_{n=1}^\infty (X \setminus E_n)$ 仍包含一个稠密 $G_\delta$ 集，因此$X\setminus M $稠密.\par
    $\Rightarrow \overline{X \setminus M} = X \Rightarrow \mathring{M} = \varnothing$.
\end{proof}

\begin{theorem}
    设 $X = C([0, 1])$ 并配备最大模范数，则处处不可导的函数构成剩余集.
\end{theorem}
\begin{proof}
    \begin{align*}
        D &= \{f \in C([0, 1]) : f \text{在某处可导}\}\\
        &\subset \bigcup_{M=1}^\infty \bigcup_{n=1}^\infty \left\{f \in C([0, 1]) : \exists\, x \in [0, 1], \forall\, y \in [0, 1],\right.\\
        &\hspace{120pt}\left.\st 0 < |x - y| < \frac{1}{n}, \left|\frac{f(x) - f(y)}{x - y}\right| \leq M\right\} \\
        &\triangleq \bigcup_{M,n=1}^\infty D_{n, M}.
    \end{align*}\par
    下证 $\forall\, n, M \in \mb{N}$，$D_{n, M}$ 无处稠密.\par
    对 $f \in D_{n, M}$，对 $f$ 增加高频小幅振荡，可得 $\tilde{f} \notin D_{n, M}$.
\end{proof}
\subsection*{Banach-Mazur 游戏}\index{b@Banach-Mazur游戏}
为了更直观地刻画剩余集和第一纲集，可以使用 Banach-Mazur 游戏.\par
规则：给定 $A \subset \mb{R}$.\par
玩家 (I) 选择闭区间 $I_1$.\par
玩家 (II) 选择闭区间 $I_2 \subset I_1$.\par
玩家 (I) 选择闭区间 $I_3 \subset I_2$.\par
$\cdots$\par
两个玩家交替选择了一个区间套 $I_1 \supset I_2 \supset I_3 \supset \cdots$.\par
若 $\bigcap_{n=1}^\infty I_n \cap A \neq \varnothing$，则 (I) 获胜，否则 (II) 获胜.\par
\begin{remark}
    从上面的规则中可以看出，若 $A$ 包含区间，则玩家 (I) 有必胜策略.
\end{remark}

\begin{corollary}
    包含区间的集合不是第一纲集.
\end{corollary}

玩家 (II) 的必胜策略是指，$\forall\, n \in \mb{N}$，$\exists$ 映射函数 $f_n$，$\st I_{2n} = f_n(I_1, \cdots, I_{n-1})$ 满足 $I_{2n} \subset I_{2n-1} \subset \cdots$.


\begin{theorem}
    玩家 (II) 有必胜策略当且仅当 $A$ 是第一纲集.
\end{theorem}

\begin{proof}
    “$\Longleftarrow$” $A = \bigcup_{n=1}^\infty A_n$，$A_n$ 无处稠密，则 $\mathbb{R}\setminus\overline{A_n}=\mathrm{int}(\mb{R} \setminus A_n) \neq \varnothing$.\par
    因此，(II)总能选 $I_{2n} \subset \mathrm{int}(\mb{R} \setminus A_n)$.\par
    “$\Longrightarrow$” 假设 (II) 有必胜策略.\par
    Step 1: 选取闭区间 $\tilde{I}_1, \tilde{I}_2, \tilde{I}_3, \cdots$，$\st J_i = f_1(\tilde{I}_i)$ 两两不交，且 $\bigsqcup_i J_i$ 稠密.\par
    $\Rightarrow A_1 := \mb{R} \setminus \bigcup_i J_i$ 无处稠密.\par
    Step 2: 在 $J_i$ 中选取 $\tilde{I}_{i1}, \tilde{I}_{i2}, \tilde{I}_{i3}, \cdots$，$\st J_{ij} = f_2(\tilde{I}_i, \tilde{J}_i, \tilde{I}_{ij})$ 两两不交，且 $\bigsqcup_j J_{ij}$ 在 $J_i$ 中稠密.\par
    $\Rightarrow A_2 := \mb{R} \setminus \bigcup_{i,j} J_{ij}$ 无处稠密.\par
    以此类推形成 $J_i, J_{ij}, J_{ijk}, \cdots $\par
    \begin{claim}{$A \subseteq \bigcup_{n=1}^\infty A_n$.}
    反证，设 $x \in A \setminus \bigcup_{n=1}^\infty A_n$.\par
    则 $x \notin A_1 \Rightarrow \exists\, i, \st x \in J_i$.\par
    $x \notin A_2 \Rightarrow \exists\, j, \st x \in J_{ij}$.\par
    $\cdots$\par
    那么 $x$ 落在某个玩家 (II) 必胜策略生成的区间套中.\par
    $\Rightarrow x \notin A$，矛盾.
    \end{claim}

    因此 $A$ 为第一纲集.
\end{proof}

我们证明了玩家 (II) 有必胜策略 $\iff A$ 是第一纲集.\par
同理玩家 (I) 有必胜策略 $\iff \exists$ 区间 $I, \st I \cap (\mb{R} \setminus A)$ 是第一纲集.\par
\begin{remark}
    $\exists\, A \subset \mb{R}, \st$ (1) $A$ 不是第一纲集. (2) $\forall$ 区间 $I$, $I \setminus A$ 不是第一纲集.\par
此时 (I), (II) 均无必胜策略.
\end{remark}

回忆 Bernstein 集 $A$，$A$ 会和任意不可数闭集有交点. 
若 $A \cap [0, 1]$ 是第一纲集，则 $[0, 1] \setminus A$ 是剩余集.\par
$\Rightarrow [0, 1] \setminus A$ 包含 Cantor 集. \par
$\Rightarrow$ 但 $A \cap \mcal{C} \neq \varnothing$. \par
类似地，$[0, 1] \setminus A$ 也是第一纲集.

\begin{instance}
    $\mathbb{Q} = \{q_i\}_{i\in\mathbb{N}}$，令 $I_{ij} = \left(q_i - \frac{1}{2^{i+j}}, q_i + \frac{1}{2^{i+j}}\right)$，$G_j = \bigcup_{i=1}^\infty I_{ij}$，$B = \bigcap_{j=1}^\infty G_j$.\par
    则 $|G_j| \leq \sum_{i=1}^\infty |I_{ij}| \leq \sum_{i=1}^\infty 2 \cdot \frac{1}{2^{i+j}} = 2^{-j+1}$.\par
    因此 $|B| = 0$.\par
    此外，$B$ 是 $G_\delta$ 集，且 $\mathbb{Q} \subseteq B$，故 $B$ 是稠密的，进而 $B$ 是剩余集.
\end{instance}

\begin{corollary}
    $\mathbb{R} = A \cup B, A \cap B = \varnothing$，其中 $B$ 是零测的剩余集，$A$ 是第一纲的全测集.
\end{corollary}

Baire 纲定理有一系列重要的等价形式与应用. 以下将这些命题按逻辑层次组织起来，最终归结为完备度量空间是第二纲集这一核心结论.

\begin{lemma}[闭无处稠密集的可数并的内部为空][lem:closed-nowhere-dense]
    设 $X$ 是非空完备度量空间，$\{F_n\}_{n=1}^\infty$ 是 $X$ 中一列闭的无处稠密集，则
    \begin{equation*}
        \mathrm{int}\biggl( \bigcup_{n=1}^\infty F_n \biggr) = \varnothing.
    \end{equation*}
\end{lemma}
\begin{proof}
    令 $M = \bigcup_{n=1}^\infty F_n$，则 $X \setminus M = \bigcap_{n=1}^\infty (X \setminus F_n)$. \par
    每个 $F_n$ 是闭的无处稠密集，故 $X \setminus F_n$ 是稠密开集.\par
    由 Baire 纲定理，$\bigcap_{n=1}^\infty (X \setminus F_n)$ 稠密，从而 $\overline{X \setminus M} = X$，即 $\mathring{M} = \varnothing$.
\end{proof}

\begin{theorem}[$F_\sigma$ 集的 Baire 纲判据][thm:fsigma-baire]
    设 $X$ 是非空度量空间，$X = \bigcup_{n=1}^\infty F_n$，其中 $F_n$ 均为闭集. 则若$X$完备，则$\exists\, n_0 \in \mathbb{N},\ \st \mathring{F_{n_0}} \neq \varnothing$.
\end{theorem}
\begin{proof}
    设 $X$ 完备. 反证：若不然，每个 $F_n$ 均为闭的无处稠密集.\par
    则由\cref{lem:closed-nowhere-dense} 知 $\mathrm{int}\ab(\bigcup_{n=1}^\infty F_n)=\mathring{X}=X=\varnothing$，与$X$非空矛盾.
\end{proof}

\begin{lemma}[完备度量空间是第二纲集]\label{lem:second-category}
    非空完备度量空间不是第一纲集.
\end{lemma}
\begin{proof}
    若 $X$ 是第一纲集，即 $X = \bigcup_{n=1}^\infty N_n$，$N_n$ 无处稠密. \par
    取 $F_n = \overline{N_n}$，则 $F_n$ 是闭的无处稠密集，且 $X \subseteq \bigcup_{n=1}^\infty F_n$z.\par
    由引理 \ref{lem:closed-nowhere-dense} 知 $\mathring{X} = \varnothing$，但 $X$ 是非空完备度量空间，$\mathring{X} = X \neq \varnothing$，矛盾.
\end{proof}


基于这一观察，可以得到下述定理：

\begin{theorem}
    设 $X$ 是一个无穷维 Banach 空间，则其任意 Hamel 基均不可数.
\end{theorem}
\begin{proof}
    假设 $X$ 有一组可数 Hamel 基 $\{e_i\}_{i\in\mathbb{N}}$，即 $X = \mathrm{span}\{e_i\}$.\par
    等价地，$X = \bigcup_{i=1}^\infty V_i$，其中 $V_i = \mathrm{span}\{e_1, \dots, e_i\}$.\par
    由定理 \ref{thm:fsigma-baire}，$\exists\, i \in \mathbb{N}$，$V_i = \overline{V_i}$ 有非空内部，则存在开球 $\mathbb{B}_r(x_0) \subset V_i$.\par
    $\Rightarrow V_i \supseteq \mathbb{B}_r(x_0) - \mathbb{B}_r(x_0) \supseteq \mathbb{B}_{2r}(0)$.\par
    由于 $V_i$ 是向量空间，$V_i \supseteq \bigcup_{\lambda>0} \lambda \mathbb{B}_{2r}(0) = X$.\par
    得出 $X = V_i = \mathrm{span}\{e_1, \dots, e_i\}$ 是有限维的，矛盾.
\end{proof}

\begin{corollary}
    $c_{00}$ 为由有限项非零的数列构成的空间，不存在任何范数使 $c_{00}$ 为 Banach 空间.
\end{corollary}
\begin{proof}
    $e_i = (0, 0, \dots, \underbrace{1}_{\text{第 } i \text{ 项}}, 0, \dots)$ 构成 $c_{00}$ 的 Hamel 基.\par
    但 $e_i$ 可数，因而 $c_{00}$ 不能被赋予使之成为 Banach 空间的范数.
\end{proof}

进一步地，还可以使用上述性质研究函数的特点.我们可以轻易构造出在 $\mathbb{R} \setminus \mathbb{Q}$ 上连续，在 $\mathbb{Q}$ 上间断的函数.但反过来并不成立.

\begin{theorem}
    不存在 $f \colon \mathbb{R} \to \mathbb{R}$ 恰好在 $\mathbb{Q}$ 上连续.\par
    更一般地，若 $X, Y$ 是度量空间，$f \colon X \to Y$，则 $C_f = \{x \in X : f \text{ 在 } x \text{ 处连续}\}$ 是 $G_\delta$ 集.
\end{theorem}
\begin{proof}
    若一般结论成立，对应到 $\mathbb{R}$ 上的函数，则 $C_f = \mathbb{Q}$ 为稠密 $G_\delta$ 集，因而 $\mathbb{Q}$ 为剩余集，矛盾.\par
    下证一般结论成立.\par
    定义振幅函数 $\mathrm{Osc} f(x) = \lim_{r \to 0} \mathrm{diam}[f(\mathbb{B}_r(x))]$.\par
    则 $x \in C_f \iff \mathrm{Osc} f(x) = 0$，那么 $C_f = \bigcap_{n=1}^\infty \left\{\mathrm{Osc} f < \frac{1}{n}\right\}$.\par
    断言：$\forall\, M > 0$，$\{\mathrm{Osc} f < M\}$ 是 $X$ 中开集.\par
    事实上，若 $x \in X$ 满足 $\mathrm{Osc} f(x) < M$，$\exists\, r > 0,\st \mathrm{diam} f[\mathbb{B}_{r}(x)] < M$.\par
    则 $\forall\, y \in \mathbb{B}_{\frac{r}{2}}(x)$，$\mathbb{B}_{\frac{r}{2}}(y) \subset \mathbb{B}_{r}(x)$，有 $\mathrm{diam} f[\mathbb{B}_{\frac{r}{2}}(y)] \leq \mathrm{diam} f[\mathbb{B}_{r}(x)] < M$.\par
    从而 $\mathrm{Osc} f(y) < M \Rightarrow \mathbb{B}_{\frac{r}{2}}(x) \subseteq \{\mathrm{Osc} f < M\}$.故 $\{\mathrm{Osc} f < M\}$ 是开集.断言成立.
\end{proof}

另外一个定理反应了实空间中函数的部分刚性.

\begin{theorem}
    设 $f \in C^\infty(\mathbb{R})$ 且 $\forall\, x \in \mathbb{R}$, $\exists\, n \in \mathbb{N}$, s.t. $f^{(n)}(x) = 0$，则 $f$ 是多项式函数.
\end{theorem}
\begin{proof}
    设 $\Omega = \bigcup\{(a, b) \subset \mathbb{R}: f|_{(a,b)} \text{ 是多项式}\}$，下证 $\Omega = \mathbb{R}$.\par
    由定义可知 $\Omega$ 是开集，故写作互不相交的开区间并$\Omega = \bigsqcup _j (a_j, b_j)$.\par
    \begin{claim}{$\overline{\Omega} = \mathbb{R}$.}
        只需验证 $\forall\, c < d$, $[c, d] \cap \Omega \neq \varnothing$.\par
    实际上，由假设，$ [c, d] = \bigcup_{n=1}^\infty ([c, d] \cap E_n),\mathbb{R}=\bigcup_{n=0}^\infty E_n$，其中 $E_n = \{x \in \mathbb{R}: f^{(n)}(x) = 0\}$.\par
    因此完备度量空间$[c,d]$可写作$F_\sigma$集，由 Baire 纲定理，$\exists\, n \in \mathbb{N}$, s.t. $[c, d] \cap E_n$ 有非空内部.\par
    也即，$\exists\, e < \hat{f},\st (e, \hat{f}) \subseteq [c, d] \cap E_n$，故$\forall\, x \in (e, \hat{f})$, $f^{(n)}(x) = 0$.\par
    由于$n$不依赖于$x$，$f$ 是 $(e, \hat{f})$ 上多项式 $\Rightarrow (e, \hat{f}) \subseteq \Omega \cap [c, d]$.
    \end{claim}

    下证 $X = \mathbb{R} \setminus \Omega = \varnothing$.\par
    注意到 $X$ 在 $\mathbb{R}$ 中闭，因而$X$是完备度量空间.\par
    由于 $\overline{\Omega}=\mathbb{R}$，若 $X \neq \varnothing$，则 $X$ 中无孤立点，即对任意 $x \in X$，总存在序列 $x_i \in X \setminus \{x\}$ 使得 $x_i \to x$.\par
    由 $f$ 的假设，$X = \bigcup_{n=1}^\infty (X \cap E_n)$.\par
    由Baire 纲定理，$\exists\, n' \in \mathbb{N}$, s.t. $X \cap E_{n'}$ 在 $X$ 中有非空相对内部.\par
    即存在开区间 $(g, h)\subset X\cap E_{n'},\st \forall\, x \in X \cap (g, h)$, $f^{(n')}(x) = 0$.\par
    不失一般性，可取 $\{x_i\} \subset X \cap (g, h)$，满足 $x_i \neq x$ 且 $x_i \to x$.\par
    则 $f^{(n'+1)}(x) = \lim_{i\to\infty} \frac{f^{(n')}(x_i) - f^{(n')}(x)}{x_i - x} = 0$.\par
    由此归纳可得，$\forall\, m \geq n'$，在 $X \cap (g, h)$ 上 $f^{(m)} \equiv 0$.\par
    \begin{claim}{在 $(g, h) \cap \Omega$ 上，$f^{(n')} \equiv 0$.}
        由于 $\overline{\Omega} = \mathbb{R}$，只需证明$\forall\, i\in\mb{N}$，若$(a_i,b_i)\cap (g,h)=\varnothing$，则在$(a_i,b_i)$上，$f^(n')\equiv 0$.\par
    由于 $(g, h) \cap X \neq \varnothing$，$(g, h)$ 不可能完全包含在任何一个使得 $(a_i, b_i)$ 成为连通分支的完整区间中，所以 $(g, h)$ 一定包含 $(a_i, b_i)$ 的至少一个端点.\par
    不失一般性，设 $a_i \in (g, h)$，则 $a_i \in X \cap (g, h)$，因此 $\forall\, m \geq n'$，$f^{(m)}(a_i) = 0$.\par
    若 $f$ 在 $(a_i, b_i)$ 上是 $r$ 阶多项式，假设 $r \ge n'$，则 $f^{(r)}(a_i) \neq 0$，导致矛盾.故 $r < n'$.\par
    进而在 $(a_i, b_i)$ 上，$f^{(n')} \equiv 0$.
    \end{claim}

    结合 $X \cap (g,h)$ 上和 $\Omega \cap (g,h)$ 上的结论，在 $(g, h)$ 上恒有 $f^{(n')} \equiv 0$.\par
    从而 $f$ 是 $(g, h)$ 上的至多 $n'-1$ 次多项式，导致 $(g, h) \subseteq \Omega$.\par
    这与 $(g, h) \cap X \neq \varnothing$ 既而 $X = \mathbb{R} \setminus \Omega$ 矛盾. 故 $X$ 必为空集.
\end{proof}

\section{开映射定理与逆算子定理}

\begin{theorem}[逆算子定理]\index{n@逆算子定理}
    设 $X, Y$ 是 Banach 空间，$T \in \mcal{B}(X,Y)$.若 $T$ 是双射，则 $T^{-1} \in \mcal{B}(Y,X)$.
\end{theorem}
它是开映射定理的直接推论.

\begin{theorem}[开映射定理]\index{k@开映射定理}
    设 $X, Y$ 是 Banach 空间，$T \in \mcal{B}(X,Y)$.若 $T$ 是满射，则 $T$ 是开映射.
\end{theorem}
\begin{proof}
    $T$ 是满射 $\iff Y = T(X) = T\left(\bigcup_{n=1}^\infty \mathbb{B}_n^X(0)\right) = \bigcup_{n=1}^\infty {T(\mathbb{B}_n^X(0))}$.\par
    由于 $Y$ 是 Banach 空间，由 Baire 纲定理，$\exists\, n_0 \in \mathbb{N},\st \overline{T(\mathbb{B}_{n_0}^X(0))}$ 有非空内部.\par
    亦即，$\exists\, y_0 \in Y, \rho_0 > 0,\st \mathbb{B}_{\rho_0}^Y(y_0) \subseteq \overline{T(\mathbb{B}_{n_0}^X(0))}$.\par
    由于等式右边集合对称，有 $\mathbb{B}_{\rho_0}^Y(-y_0) \subseteq \overline{T(\mathbb{B}_{n_0}^X(0))}$.\par
    又因为等式右边是凸集，$\mathbb{B}_{\rho_0}^Y(0) \subseteq \frac{1}{2} \mathbb{B}_{\rho_0}^Y(y_0) + \frac{1}{2} \mathbb{B}_{\rho_0}^Y(-y_0) \subseteq \overline{T(\mathbb{B}_{n_0}^X(0))}$.\par
    假设我们已经证明了更强的结论：$\mathbb{B}_{\rho_0}^Y(0) \subseteq T(\mathbb{B}_{n_0}^X(0))$.\par
    那么任取 $X$ 中开集 $U$ 以及 $y \in T(U)$，则存在 $x \in U$ 使得 $y = Tx$.\par
    由于 $U$ 是开集，故存在 $r > 0$ 使得 $\mathbb{B}_r^X(x) = x + \mathbb{B}_r^X(0) \subseteq U$.\par
    利用$T$ 的线性和更强的结论，我们有 $T(\mathbb{B}_r^X(0)) = \frac{r}{n_0} T(\mathbb{B}_{n_0}^X(0)) \supseteq \frac{r}{n_0} \mathbb{B}_{\rho_0}^Y(0) = \mathbb{B}_{\frac{r\rho_0}{n_0}}^Y(0)$.\par
    所以 $T(U) \supseteq T(\mathbb{B}_r^X(x)) = Tx + T(\mathbb{B}_r^X(0)) \supseteq y + \mathbb{B}_{\frac{r\rho_0}{n_0}}^Y(0) = \mathbb{B}_{\frac{r\rho_0}{n_0}}^Y(y)$，这说明 $T(U)$ 包含 $y$ 的一个邻域，故 $T(U)$ 是开集，即 $T$ 是开映射.
\end{proof}

\begin{lemma}[Banach]\index{b@Banach引理}
    若 $\forall\, y \in \mathbb{B}_1^Y(0)$，$\dist(y, T(\mathbb{B}_c^X(0))) < \varepsilon$，则 $\mathbb{B}_1^Y(0) \subseteq T\left(\mathbb{B}_{\frac{c}{1-\varepsilon}}^X(0)\right)$.
\end{lemma}
\begin{proof}
    任取 $y \in \mathbb{B}_1^Y(0)$，目标是找到 $z \in X$ 使得 $y = Tz$ 且 $\|z\| < \frac{c}{1-\varepsilon}$.\par
    由于 $\forall\, y \in \mathbb{B}_1^Y(0)$，$\dist(y, T(\mathbb{B}_c^X(0))) < \varepsilon$，存在 $x_0 \in \mathbb{B}_c^X(0)$，s.t. $\|y - Tx_0\| < \varepsilon$.\par
    则 $\frac{y - Tx_0}{\varepsilon} \in \mathbb{B}_1^Y(0)$，因此$\exists\, \tilde{x}_1 \in \mathbb{B}_c^X(0)$，s.t. $\left\|\frac{y - Tx_0}{\varepsilon} - Tx_1\right\| < \varepsilon$.\par
    记 $x_1 = \varepsilon x_1 \in \mathbb{B}_{\varepsilon c}^X(0)$，则 $\|y - T(x_0 + x_1)\| < \varepsilon^2$.\par
    以此类推，可取到 $x_j \in \mathbb{B}_{\varepsilon^j c}^X(0),\st \left\|y - T\left(\sum_{i=0}^j x_i\right)\right\| < \varepsilon^{j+1}$.\par
    由于 $X$ 是 Banach 空间，$\sum x_i\to z\in X,\st y= Tz$.\par
    由$\|z\|=\sum_{j=1}^\infty\|x_j\|=\sum_{j=1}^\infty \varepsilon^j c < \frac{c}{1-\varepsilon}$可知引理成立
\end{proof}

下面的推论将帮助我们补全开映射定理中空缺的部分.
\begin{corollary}
    若 $T(\mathbb{B}_{n_0}^Y(0))$包含$\mathbb{B}_1^Y(0)$的一个稠密子集，则它包含整个$\mathbb{B}_1^Y(0)$.
\end{corollary}
\begin{proof}
    条件等价于 $\forall\, \varepsilon > 0$，$\mathbb{B}_{1}^Y(0) \subseteq T(\mathbb{B}_{n_0}^X(0)) + \mathbb{B}_{\varepsilon}^Y(0)$且$\mathrm{dist}(y, T(\mathbb{B}_{n_0}^X(0))) < \varepsilon$.\par
    由前述引理，$\forall\,\varepsilon\in(0,1),\mathbb{B}_{1}^Y(0) \subseteq T\left(\mathbb{B}_{\frac{n_0}{1-\varepsilon}}^X(0)\right)$.\par
    利用线性性可得，$\forall\, \varepsilon \in (0, 1)$，$\mathbb{B}_{(1-\varepsilon)}^Y(0) \subseteq T(\mathbb{B}_{n_0}^X(0))$.\par
    从而 $\mathbb{B}_{1}^Y(0) = \bigcup_{\varepsilon \in (0, 1)} \mathbb{B}_{1-\varepsilon}^Y(0) \subseteq T(\mathbb{B}_{n_0}^X(0))$.
\end{proof}
\begin{remark}
    在开映射定理中，我们要求 $X, Y$ 都是 Banach 空间，但二者参与证明的方式不同.
\end{remark}

若 $X, Y$ 是 Banach 空间，$T \in \mcal{B}(X,Y)$，则 $\ker T = T^{-1}\{0\}$ 是 $X$ 的闭子空间. 从而 $X/\ker T$ 为商范数下 Banach 空间.
可以定义$\setjot[-4]\begin{aligned}[t]
    \hat{T}\colon X/\ker T &\to Y \\
    [x]=x+\ker T &\mapsto  Tx
\end{aligned} $
这是 $X/\ker T$ 到 $Y$ 的连续双射.
因此，在开映射定理下，若 $T$ 是满射，则 $Y \cong X/\ker T$.
类似于抽象代数中的同态基本定理.

\begin{corollary}
    设 $X$ 是赋范线性空间，使得存在两个范数 $\|\cdot\|_1, \|\cdot\|_2$ 使其完备. 若 $\|\cdot\|_1 \lesssim \|\cdot\|_2$，即 $\exists\, M > 0, \st \forall\, x \in X, \|x\|_1 \leq M\|x\|_2$，则 $\|\cdot\|_2 \lesssim \|\cdot\|_1$，即 $\|\cdot\|_1 \approx \|\cdot\|_2$.
\end{corollary}
\begin{proof}
    考虑 $\mathrm{Id}\colon (X, \|\cdot\|_2) \to (X, \|\cdot\|_1)$.\par
    那么$\| \cdot \|_1 \leq M \| \cdot \|_2 \iff \|\mathrm{Id}\| \le M < \infty$.\par
    由逆算子定理，$\mathrm{Id}^{-1}$ 也连续，从而 $\|\cdot\|_2 \lesssim \|\cdot\|_1$.
\end{proof}

观察到，若 $\forall\, x \in X, \|x\|_1 \leq M\|x\|_2$，则 $\mathbb{B}_1^{(X, \|\cdot\|_2)}(0) \subseteq M\mathbb{B}_1^{(X, \|\cdot\|_1)}(0)$.
因此，若 $\|\cdot\|_1 \approx \|\cdot\|_2$，则 $\exists\, c > 0, \st \mathbb{B}_{c^{-1}}^{(X, \|\cdot\|_2)}(0) \subseteq \mathbb{B}_1^{(X, \|\cdot\|_1)}(0) \subseteq \mathbb{B}_c^{(X, \|\cdot\|_2)}(0)$.
从而，$(X, \|\cdot\|_1)$ 与 $(X, \|\cdot\|_2)$ 有相同的拓扑.

下面研究另两个定理，闭图像定理与一致有界原理.

\begin{definition}[图像]
    设 $f\colon A \to B$，其图像为 $\Gamma_f = \{(x, f(x)) \colon x \in A\} \subset A \times B$.
\end{definition}

\begin{theorem}[闭图像定理]\index{b@闭图像定理}
    设 $X, Y$ 是 Banach 空间，$T \in \mcal{L}(X, Y)$，则 $T \in \mcal{B}(X, Y) \iff \Gamma_T$ 在 $(X \times Y, \|\cdot\|_X + \norm{\,\cdot\,}_Y)$ 中是闭集.
\end{theorem}
\begin{proof}
    \fbox{$\Rightarrow$} 若 $T \in \mcal{B}(X, Y)$，则 $T$ 连续. 设 $\{(x_n, y_n)\} \subset \Gamma_T, \st (x_n, y_n)$ 在范数下收敛到 $(x, y)$.\par
    则 $y_n = Tx_n \to Tx$ 且 $y_n \to y$. 由极限的唯一性，得 $Tx = y$，故 $(x, y) \in \Gamma_T$.\par
    \fbox{$\Leftarrow$} 若 $\Gamma_T$ 闭，考虑投影$\setjot[-6]\begin{aligned}[t]
        \pi_X\colon X \times Y &\to X\\
        (x, y) &\mapsto x
    \end{aligned} $和$\setjot[-6]\begin{aligned}[t]
        \pi_Y\colon X \times Y &\to Y \\
        (x, y) &\mapsto y
    \end{aligned} $.\par
    若 $\pi_X|_{\Gamma_T}(x, Tx) = 0$，则 $x = 0$，从而 $Tx = 0$，故 $\pi_X|_{\Gamma_T}$ 单.\par
    显然 $\pi_X|_{\Gamma_T}$ 满，故 $\pi_X|_{\Gamma_T}\colon \Gamma_T \to X$ 是双射.\par
    由于 $\Gamma_T$ 是 $X \times Y$ 的闭子空间，因此它是 Banach 空间.\par
    由逆算子定理，$(\pi_X|_{\Gamma_T})^{-1}$ 是有界的.
    因此 $T = \pi_Y \circ (\pi_X|_{\Gamma_T})^{-1}$ 有界.
\end{proof}

下面看闭图像定理的应用.
设 $X$ 是 Banach 空间，$M, N$ 是闭子空间使得 $M \oplus N = X$. 考虑投影 $P\colon X \to M, x \mapsto m$，其中 $x = m + n, m \in M, n \in N$，则 $P$ 有界.
\begin{proof}
    由闭图像定理，只要证 $\Gamma_P$ 是闭的.
    设 $\Gamma_P$ 中的序列 $(x_j, Px_j) \to (x, y)$\par
    因为 $Px_j \in M$ 且 $M$ 是闭子空间，则$M$是Banach空间，故$y \in M$.\par
    同时 $x_j - Px_j \in N$，由于 $N$ 也是闭子空间，故 $x_j - Px_j \to x - y \in N$.\par
    因此 $x = y + (x - y)$ 且 $y \in M, x - y \in N$. 按投影定义知 $Px = y$，即 $(x, y) \in \Gamma_P$. \par
    故 $\Gamma_P$ 闭，$P$ 有界.
\end{proof}

\begin{definition}
    设 $M$ 是 Banach 空间 $X$ 的闭子空间，则称 $M$ 是\colorbf{可补的}\index{k@可补子空间}当且仅当存在闭子空间 $N$，使得 $M \cap N = \{0\}, M + N = X$.
\end{definition}

上述表示和线性空间的直和分解的表述类似，那么类似于线性空间中的结论，可以思考投影和可补子空间之间的关系.

\begin{definition}[投影]\index{t@投影}
    设 $X$ 是 赋范线性空间，则 $P\colon X \to X$ 称为\colorbf{投影}，当且仅当 $P^2 = P$.
\end{definition}

之后，我们记 $\ker P = \mcal{N}(P)$，$\mcal{R}(P)=\mathrm{ran} P = P(X)$ 分别为 $X$ 的闭子空间.
进一步，可以观察到若 $P \in \mcal{B}(X)$ 是投影，则 $X = \mcal{N}(P) \oplus \mcal{R}(P)$.
事实上，$\mcal{N}(P) = P^{-1}(0), \mcal{R}(P) = \mcal{N}(I-P)$.
\begin{itemize}[label=$\circ$]
    \item 若 $x \in \mcal{N}(P) \cap \mcal{R}(P)$，则 $Px = 0$，且 $\exists\, y \in X$, $\st x = Py \Rightarrow P(Py) = Px = 0$.\par
    又 $P(Py) = Py = x$，故 $x = 0$.
    \item $\forall\, x \in X$，$x = Px + (I - P)x$. 则 $Px \in \mcal{R}(P), (I - P)x \in \mcal{N}(P)$.
\end{itemize}

\begin{theorem}
    设 $M$ 是 Banach 空间 $X$ 的闭子空间，则 $M$ 可补当且仅当 $M$ 是 $X$ 的一个有界投影的像.
\end{theorem}
\begin{proof}
    \fbox{$\Leftarrow$} 由上述分析即得.\par
    \fbox{$\Rightarrow$} 若 $X = M \oplus N$，构造$\setjot[-6]\begin{aligned}[t]
        P\colon X &\to M \\
        m + n &\mapsto m
    \end{aligned}$\par
    由前述结论，$P \in \mcal{B}(X)$，且 $M = \mcal{R}(P), N = \mcal{N}(P)$.
\end{proof}

向前看，对于 Hilbert 空间 $H$，有
\begin{proposition}
    设 $M$ 是 Hilbert 空间 $\mathcal{H}$ 的任意闭子空间，则 $M$ 是可补的.
\end{proposition}
\begin{proof}
    $\mathcal{H} = M \oplus M^\perp$.
\end{proof}

上述命题的逆命题也成立.
\begin{proposition}
    若 Banach 空间 $X$ 的任意闭子空间可补，则 $X$ 是 Hilbert 空间.
\end{proposition}

\begin{theorem}
    设 $X$ 是 Banach 空间，若 $M$ 是 $X$ 的闭子空间且为有限维或余有限维的，则 $M$ 可补.
\end{theorem}
\begin{proof}
    证明 $\dim M < \infty$ 的情形.\par
    设 $\{e_1, \dots, e_n\}$ 是 $M$ 的 Hamel 基，则 $\forall\, x \in M$，$x = \alpha_1(x) e_1 + \dots + \alpha_n(x) e_n$，其中 $\alpha_j \in M^*, \forall\, 1 \leq j \leq n$.\par
    由 Hahn-Banach 延拓定理，可以把 $\alpha_j$ 延拓为 $\tilde{\alpha}_j \in X^*$，则 $N = \bigcap_{j=1}^n \ker \tilde{\alpha}_j$ 为 $M$ 的代数补空间，且因为 $\tilde{\alpha}_j$ 连续，可知 $N$ 闭. 从而 $M$ 可补.\par

    证明 $\mathrm{codim}\, M < \infty$ 的情形.\par
    由于 $\mathrm{codim}\, M = n < \infty$，商空间 $X/M$ 为 $n$ 维向量空间.\par
    设 $\{e_1 + M, \dots, e_n + M\}$ 为 $X/M$ 的一组基，取 $N = \mathrm{span}\{e_1, \dots, e_n\}$.\par
    由基的定义易得，任意 $x \in X$ 可唯一分解为 $M$ 中的元素与 $N$ 中的元素之和，即 $X = M \oplus N$.\par
    由于 $N$ 为有限维子空间，故 $N$ 为闭子空间.\par
    由于 $M$ 与 $N$ 皆为闭子空间，对应的投影算子是有界的.因此 $M$ 是可补的.
\end{proof}

Banach-Steinhaus定理实际上可以包含一系列定理和推论.

\begin{theorem}[A]\index{b@Banach-Steinhaus定理}
    \wideline
    设 $X$ 是完备度量空间，$\{f_i \colon i \in \mcal{I}\} \subset C^0(X, \mb{R})$，则下述命题有且仅有一成立：
    \begin{enumerate}[label=(\roman*)]
        \item 存在非空开集 $U \subset X$ 与 $M > 0$，$\st \sup_{i \in \mcal{I}} |f_i(x)| \leq M, \forall\, x \in U$.
        \item 存在稠密 $G_\delta$ 集 $E \subset X$，$\st \sup_{i \in \mcal{I}} |f_i(x)| = \infty, \forall\, x \in E$.
    \end{enumerate}
\end{theorem}
\begin{proof}
    考虑 $\psi(x) := \sup_{i \in \mcal{I}} |f_i(x)|$.\par
    设 $V_n = \{\psi > n\} = \bigcup_{i \in \mcal{I}} \{|f_i| > n\}$，由 $f_i$ 连续，知 $V_n$ 为开集.\par
    若 (i) 不成立，则对任意非空开集 $U \subset X$，$V_n \cap U \neq \varnothing, \forall\, n \in \mb{N}$.\par
    因此，每个 $V_n$ 都是稠密开集.\par
    由 Baire 纲定理，$E := \bigcap_{n=1}^\infty V_n$ 是一个稠密 $G_\delta$ 集. 这证明了 (ii).
\end{proof}
定理A的结论表明，在完备度量空间中，函数族 $\{f_i\}$ 的一致有界性与逐点无界性之间存在二分法：要么在某个非空开集上一致有界，要么在一个稠密 $G_\delta$ 集上逐点无界.

\begin{corollary}[B]
    设 $X$ 是完备度量空间，$\{f_i \colon i \in \mcal{I}\} \subset C^0(X, \mb{R})$，若 $\forall\, x \in X$，$\sup_{i \in \mcal{I}} |f_i(x)| < \infty$，则存在非空开集 $U \subset X$ 与 $M > 0$，$\st \sup_{i \in \mcal{I}} |f_i(x)| \leq M, \forall\, x \in U$.
\end{corollary}
上述推论的条件排除了定理 A 中 (ii) 情况，故得到 (i) 的结论，它将函数族的逐点有界提升到在某个非空开集上的一致有界.\par
上面两个定理都未对函数族作过多限制，现在我们考虑线性算子族的情况.

\begin{corollary}[C]
    设 $X$ 是 Banach 空间且 $Y$ 是 赋范线性空间，设 $\{T_i \in \mcal{B}(X, Y) \colon i \in \mcal{I}\}$，则要么 $\sup_{i \in \mcal{I}} \|T_i\| < \infty$，要么存在稠密$G_\delta$ 集 $E \subset X$，$\st \sup_{i \in \mcal{I}} \|T_ix\| = \infty, \forall\, x \in E$.
\end{corollary}
\begin{proof}
    令 $f_i(x) := \|T_ix\|$ 是定理 A 中所述函数族.\par
    若推论后半不成立，由定理 A，$\exists\, \mathbb{B}_{r_0}(x_0) \subset X$，$M > 0$，$\st \sup\{\|T_i x\| \colon i \in \mcal{I}, x \in \mathbb{B}_{r_0}(x_0)\} \leq M$.\par
    注意到 $\forall\, x \in \mathbb{B}_{r_0}(0) \subset \mathbb{B}_{\tfrac{r_0}{2}}(x_0) - \mathbb{B}_{\tfrac{r_0}{2}}(x_0)$，$\|T_i x\| \leq \sup\{\|T_i z\| \colon z \in \mathbb{B}_{\tfrac{r_0}{2}}(x_0)\} + \sup\{\|T_i z\| \colon z \in -\mathbb{B}_{\tfrac{r_0}{2}}(x_0)\} \leq 2M$.\par
    令$v=\frac{x}{r_0}$，则$\|T_i\| = \sup_{v \in \mathbb{B}_1(0)} \|T_i v\| = \frac{1}{r_0}\sup_{x \in \mathbb{B}_{r_0}(0)} \|T_i x\| \leq \frac{2M}{r_0},\forall\, i\in\mcal{I}$.
\end{proof}
证明的核心是，对于 $T \in \mcal{B}(X, Y)$，若 $\sup\{\|Tx\| \colon x \in \mathbb{B}_{r_0}(x_0)\} \leq M$，则 $\|T\| \leq C(M, r_0)$，$C$ 为与 $M, r_0$ 有关的常数.

\begin{corollary}[D, 一致有界原理]\index{y@一致有界原理}
    设 $X$ 是 Banach 空间，$Y$ 是 赋范线性空间，设 $\{T_i \colon i \in \mcal{I}\} \subset \mcal{B}(X, Y)$. 若 $\forall\, x\in X$，$\exists\,M_x,\st\allowbreak\sup_{i \in \mcal{I}} \|T_i x\|_Y \leq M_x$，则 $\exists\, M, \st \sup_{i \in \mcal{I}} \|T_i\|_{X \to Y} \leq M$.
\end{corollary}
该推论将逐点有界提升到算子范数的一致有界.

\begin{corollary}[E]
    设 $X$ 是 Banach 空间，$Y$ 是 赋范线性空间，设 $\{T_n\}_{n \in \mb{N}} \subset \mcal{B}(X, Y)$. 若 $\forall\, x \in X$，逐点极限 $Tx = \lim_{n \to \infty} T_nx$ 存在，则 $T \in \mcal{B}(X, Y)$ 且 $\|T\| \leq \liminf_{n \to \infty} \|T_n\|$.
\end{corollary}
\begin{proof}
    由于逐点极限存在，可知 $\sup_{n \in \mb{N}} \|T_n x\| < \infty, \forall\, x \in X$.\par
    由推论 D，$\sup_{n \in \mb{N}} \|T_n\| < \infty$.\par
    则 $\|Tx\| = \liminf_{n \to \infty} \|T_n x\| \leq \left(\liminf_{n \to \infty} \|T_n\|\right) \|x\|, \forall\, x \in X$.\par
    故 $\|T\| \leq \liminf_{n \to \infty} \|T_n\|$.
\end{proof}
因此逐点收敛极限的算子是下半连续的.
