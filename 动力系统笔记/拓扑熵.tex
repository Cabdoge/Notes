\chapter{拓扑熵}

\section{拓扑熵的定义}

在紧致度量空间上，可以类似于测度熵定义连续映射的拓扑熵，首先，需要介绍一些概念.\par

\begin{definition}{开覆盖的交}
    设$\alpha,\beta$是$X$的开覆盖，记$\alpha\vee\beta=\{A\cap B|A\in\alpha,B\in\beta\}$，称为$\alpha$和$\beta$的\textbf{交}.
\end{definition}

\begin{definition}{加细}
    设$\alpha,\beta$是$X$的开覆盖，如果$\forall\,B\in\beta$，$\exists\,A\in\alpha$使得$A\supseteq B$，则称$\beta$是$\alpha$的\textbf{加细}，记作$\alpha<\beta$.
\end{definition}

\begin{definition}{}
    设$\alpha$是开覆盖，定义$N(\alpha)=\min \left\{\#\gamma|\gamma\subset \alpha,\gamma\,\text{是}\,X\,\text{的开覆盖}\right\}$.
\end{definition}
由于$X$是紧致度量空间，因此$N(\alpha)$总是存在的且有限的.

\begin{definition}{拓扑熵}
    设$\alpha$是$X$的开覆盖，定义$H(\alpha)=\ln{N(\alpha)}$，称为$\alpha$的\textbf{熵}.\par
    设$T:X\to X$是连续映射，$\alpha$是$X$的开覆盖，定义$h(T,\alpha)=\lim_{n\to\infty}{\frac{1}{n}H\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)}$，称为\textbf{$\bm{T}$关于$\bm{\alpha}$的拓扑熵}.\par
    定义$T$的\textbf{拓扑熵}为$h(T)=\sup_{\alpha}h(T,\alpha)$.
\end{definition}

\begin{lemma}{}
    设$\alpha,\beta$是$X$的开覆盖，$T:X\to X$连续映射，则
    \begin{enumerate}[(1)]
        \item $H(\alpha\vee \beta)\leq H(\alpha)+H(\beta)$
        \item $N(T^{-1}\alpha)\leq N(\alpha)$，如果$T$是满射，则等号成立
        \item $\alpha\leq \beta\Rightarrow H(\alpha)\leq H(\beta)$
    \end{enumerate}
\end{lemma}
\begin{proof}
    \begin{enumerate}[(1)]
        \item 设$\alpha',\beta'$分别是$\alpha,\beta$的开覆盖，使得$\#\alpha'=N(\alpha),\#\beta'=N(\beta)$.\par
        $\alpha'\vee\beta'=\{A\cap B|A\in\alpha',B\in\beta'\}$\par
        则$\#\alpha'\vee\beta'\geq N(\alpha)\cdot N(\beta)$
        $\Rightarrow N(\alpha\vee\beta)\leq\#\alpha'\vee\beta'\leq N(\alpha)\cdot N(\beta)$\par
        $\Rightarrow H(\alpha\vee\beta)\leq H(\alpha)+H(\beta)$
        \item 设$\alpha'$是$\alpha$的开覆盖，使得$\#\alpha'=N(\alpha)$.\par
        则$T^{-1}\alpha$是$X$的开覆盖，且$\# T^{-1}\alpha\geq N(T^{-1}\alpha)$.\par
        又$\#\alpha'=\# T^{-1}\alpha$，因此$N(\alpha)\geq N(T^{-1}\alpha)$\par
        如果$T$是满射，那么考虑$T^{-1}\alpha$的子覆盖$\beta$，使得$\#\beta=N(T^{-1}\alpha)$\par
        由$\beta$是$X$的开覆盖$\beta\subseteq T^{-1}\alpha$，$T$是满射，则$T(\beta)$是$X$的开覆盖且$T(\beta)\subseteq\alpha$\par
        $\Rightarrow \#T(\beta)\geq N(\alpha)\Rightarrow N(T^{-1}\alpha)\geq N(\alpha)\Rightarrow N(\alpha)=N(T^{-1}\alpha)$
    \end{enumerate}
\end{proof}

和测度熵类似，我们同样需要证明定义拓扑熵时使用的极限的存在性.\par
\begin{corollary}{}
    $\lim_{n\to\infty}{\frac{1}{n}H\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)}$ 存在.
\end{corollary}
\begin{proof}
    令$a_n=H \left(\bigvee_{i=0}^{n-1}{T^{-i}\alpha}\right)$，那么$\setjot[-2]\begin{aligned}[t]
        a_{n+m} &= H \left(\bigvee_{i=0}^{n+m-1}{T^{-i}\alpha}\right) \\
        &= H \left(\left(\bigvee_{i=0}^{m-1}{T^{-i}\alpha}\right)\vee\left(\bigvee_{i=m}^{n+m-1}{T^{-j}\alpha}\right)\right) \\
        &\leq H \left(\bigvee_{i=0}^{m-1}{T^{-i}\alpha}\right) + H \left(\bigvee_{i=m}^{n+m-1}{T^{-i}\alpha}\right) \\
        &=a_m+H \left(T^{-m}\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)\right) \\
        &\leq a_m+H \left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right) \\
        &=a_m+a_n
    \end{aligned}$\par
    由引理\ref{lem:3.2}可知，上述极限存在
\end{proof}

为了说明拓扑熵在同构的连续映射下不变，需要介绍紧致拓扑空间中连续映射的同构概念.
\begin{definition}{拓扑共轭}
    设$T_1:X_1\to X_1$和$T_2:X_2\to X_2$是紧致空间上的连续映射，如果存在$\pi:X_1\to X_2$连续满映射，使得交换图成立 \par
    \begin{tikzcd}
X_1 \arrow[dd, "\pi"'] \arrow[rr, "T_1"] &  & X_1 \arrow[dd, "\pi"] \\
                                          &  &                        \\
X_2 \arrow[rr, "T_2"']                    &  & X_2                   
\end{tikzcd}
\quad 即$\pi\circ T_1=T_2\circ \pi$\par
    那么称$(T_1,X_1)$与$(T_2,X_2)$\textbf{拓扑半共轭}.$(T_2,X_2)$称为$(T_1,X_1)$的\textbf{因子系统}，$(T_1,X_1)$称为$(T_2,X_2)$的\textbf{扩展系统}.\par
    如果$\pi:X_1\to X_2$同胚，则称$(T_1,X_1)$与$(T_2,X_2)$\textbf{拓扑共轭}.
\end{definition}

\begin{theorem}{}
    设$T_1:X_1\to X_1$和$T_2:X_2\to X_2$是紧致空间上的连续映射，$(T_1,X_1)$与$(T_2,X_2)$拓扑共轭，那么$h(T_1)=h(T_2)$
\end{theorem}
\begin{proof}
    设$\alpha$是$X_2$的开覆盖，那么$\pi^{-1}(\alpha)$是$X$的开覆盖.\par
    由于$\pi$是同胚可知$T_1^{-i}\circ\pi^{-1}=\pi^{-1}\circ T_2^{-i}$\par
    则$\setjot[-2]\begin{aligned}[t]
        h \left(T_1,\pi^{-1}(\alpha)\right) &=\lim_{n\to\infty}{\frac{1}{n}H\left(\bigvee_{i=0}^{n-1}T_1^{-i}\left(\pi^{-1}(\alpha)\right)\right)}=\lim_{n\to\infty}{\frac{1}{n}H\left(\bigvee_{i=0}^{n-1}\pi^{-1}\left(T_2^{-i}(\alpha)\right)\right)} \\
        &= \lim_{n\to\infty}{\frac{1}{n}H\left(\bigvee_{i=0}^{n-1}T_2^{-i}(\alpha)\right)}=h(T_2,\alpha)
    \end{aligned}$\par
    因此$\sup_{\alpha}{h(T_2,\alpha)}=\sup_{\alpha}{h \left(T_1,\pi^{-1}(\alpha)\right)}$
    $\Rightarrow h(T_2)\leq h(T_1)$\par
    同理$h(T_1)\leq h(T_2)$，则$h(T_1)=h(T_2)$.
\end{proof}

\begin{corollary}{}
    如果$T_2:X_2\to X_2$是$T_1:X_1\to X_1$的因子系统，那么$h(T_2)\leq h(T_1)$.
\end{corollary}

\section{拓扑熵的性质}

\begin{definition}{$\bm{(n,\varepsilon)-}$生成集}
    设$(X,d)$是紧致度量空间，$T:X\to X$连续映射，设$\varepsilon>0,n\in\mb{N}$，如果$\forall\, y\in X,\exists\, x\in F$，使得$d(x,y)<\varepsilon,d(Tx,Ty)<\varepsilon,\cdots,d \left(T^{n-1}x,T^{n-1}y\right)<\varepsilon$，则称$F$是\textbf{$\bm{(n,\varepsilon)-}$生成集}.\par
    记$r_n(\varepsilon)=\min\left\{\#F|F\subset X,F\,\text{为}\,(n,\varepsilon)-\text{生成集}\right\}$
\end{definition}

\begin{definition}{$\bm{(n,\varepsilon)-}$分离集}
    设$\varepsilon>0,n\in\mb{N},E\subset X$，如果$\forall\, x,y\in E(x\neq y)$，$\exists\, i\in\{0,1,\cdots,n-1\}$，使得$d \left(T^ix,T^iy\right)>\varepsilon$，那么称$E$为$X$的\textbf{$\bm{(n,\varepsilon)-}$分离集}.\par
    记$s_n(\varepsilon)=\max\left\{\#E|E\subset X,E\,\text{为}\,(n,\varepsilon)-\text{分离集}\right\}$
\end{definition}

关于$(n,\varepsilon)-$生成集和分离集，有以下性质：
\begin{proposition}{}
    \begin{enumerate}[(1)]
        \item $r_n(\varepsilon),s_n(\varepsilon)$至于$\varepsilon$单调.
        \item $r_n(\varepsilon),s_n(\varepsilon)$有限
    \end{enumerate}
\end{proposition}
\begin{proof}
    \begin{enumerate}[(1)]
        \item $\forall\,\varepsilon_1<\varepsilon_2$，$(n,\varepsilon_1)-$生成集一定是$(n,\varepsilon_2)-$生成集，故$r_n(\varepsilon_1)\geq r_n(\varepsilon_2)$.\par
        同理可证$s_n(\varepsilon_1)\geq s_n(\varepsilon_2)$.
        \item 先证明$r_n(\varepsilon)$有限，$\forall\, x\in X,\forall\, \varepsilon>0$，可定义$B_n(X,\varepsilon)=\{y\in X|d(x,y)<\varepsilon,d(Tx,Ty)<\varepsilon,\cdots,d(T^{n-1}x,T^{n-1}y)M<\varepsilon\}$\par
        那么$B_n(x,\varepsilon)$为开集且$X=\bigcup_{x\in X}B_n(x,\varepsilon)$.\par
        由$X$是紧致度量空间可知$\exists\, x_1,x_2,\cdots,x_k\in X$使得$X=B_n(x_1,\varepsilon)\cup\cdots\cup B_n(x_k,\varepsilon)$，则$F=\{x_1,x_2,\cdots,x_k\}$为$(n,\varepsilon)-$生成集.\par
        对于$s_n(\varepsilon)$，如果$E$是$(n,\varepsilon)-$分离集，那么$\forall\,x,y\in E$，$B_n \left(x,\frac{\varepsilon}{2}\right)\cap B_n \left(y,\frac{\varepsilon}{2}\right)=\varnothing$.\par
        否则，一定存在$w\in B_n \left(x,\frac{\varepsilon}{2}\right)\cap B_n \left(y,\frac{\varepsilon}{2}\right)$，那么$d(x,w)<\frac{\varepsilon}{2},d(y,w)<\frac{\varepsilon}{2},d(Tx,Tw)<\frac{\varepsilon}{2},d(Ty,Tw)<\frac{\varepsilon}{2},\cdots$，由此可得$d(x,y)<\varepsilon,d(Tx,T)<\varepsilon,\cdots$，与$E$为$(n,\varepsilon)-$分离集矛盾.\par
        因此$E$是有限集.(否则令$\{x_k\}\subseteq E,x_k\to x$，那么$B_n \left(x_k,\frac{\varepsilon}{2}\right)\cap B_n(x_l,\frac{\varepsilon}{2})\neq\varnothing$)
    \end{enumerate}
\end{proof}
这说明上面的定义是合理的.

\begin{definition}{}
    $h_1(T)=\lim_{\varepsilon\to 0}\left(\limsup_{n\to\infty}\frac{1}{n}\ln{r_n(\varepsilon)}\right),h_2(T)=\lim_{\varepsilon\to 0}\left(\limsup_{n\to\infty}\frac{1}{n}\ln{s_n(\varepsilon)}\right)$
\end{definition}

\begin{theorem}{}{4.2}
    $h(T)=h_1(T)=h_2(T)$
\end{theorem}

在证明该定理之前，首先需要证明若干引理.

\begin{lemma}{}
    设$\varepsilon>0,n\in\mb{N}$，那么$r_n(1.1\varepsilon)\leq s_n(\varepsilon)\leq r_n \left(\frac{\varepsilon}{2}\right)$.
\end{lemma}
\begin{proof}
    设$E$是$(n,\varepsilon)-$分离集，使得$|E|=s_n(\varepsilon)$.\par
    由于$E$是最大的$(n,\varepsilon)-$分离集，因此$\forall\, x\in X(x\notin E)$，$E\cup\{x\}$不是$(n,\varepsilon)-$分离集.\par
    则$\exists\, y\in X$，使得$d(x,y)\leq\varepsilon <1.1\varepsilon,d(Tx,Ty)\leq\varepsilon<1.1\varepsilon,\cdots,d \left(T^{n-1}x,T^{n-1}y\right)\leq\varepsilon<1.1\varepsilon$.\par
    因此$E$为$(n,1.1\varepsilon)-$生成集，故$r_n(1.1\varepsilon)\leq s_n(\varepsilon)=|E|$.\par
    设$F$是$\left(n,\frac{\varepsilon}{2}\right)-$生成集，考虑$B_n \left(x,\frac{\varepsilon}{2}\right)(x\in F)$.\par
    $B_n \left(x,\frac{\varepsilon}{2}\right)=\left\{y\in X\left|d(x,y)<\frac{\varepsilon}{2},d(Tx,Ty)<\frac{\varepsilon}{2},\cdots,d \left(T^{n-1}x,T^{n-1}y\right)<\frac{\varepsilon}{2}\right.\right\}$\par
    \begin{claim}{$\forall\, x\in F,\left|B_n \left(x,\dfrac{\varepsilon}{2}\right)\cap E\right|\leq 1$}
        如果$y_1,y_2\in B_n \left(x,\frac{\varepsilon}{2}\right)\cap E$，那么$d(y_1,y_2)\leq d(y_1,x)+d(x,y_2)<\varepsilon,d(Ty_1,Ty_2)\leq d(Ty_1,Tx)+d(Tx,Ty_2)<\varepsilon,\cdots$.\par
        与$E$是$(n,\varepsilon)-$分离集矛盾.
    \end{claim}\par
    又$F$是$(n,\varepsilon)-$生成集，$X=\bigcup_{x\in F}{B_n \left(x,\frac{\varepsilon}{2}\right)}$，因此$|F|\geq |E|,r_n \left(\frac{\varepsilon}{2}\right)\geq s_n(\varepsilon)$.
\end{proof}
可以引理可以帮助我们证明$h_1(T)=h_2(T)$.\par

\begin{lemma}{}
    设$\alpha$是$X$的开覆盖，$\delta$是$\alpha$的Lebesgue数，那么$N \left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)\leq r_n \left(\frac{\varepsilon}{2}\right)$
\end{lemma}
\begin{proof}
    设$F$是$\left(n,\frac{\delta}{2}\right)-$生成集，使得$|F|=r_n \left(\frac{\delta}{2}\right),X=\bigcup_{x\in F}B_n\left(x,\frac{\delta}{2}\right)$.\par
    $\setjot[-2]\begin{aligned}[t]
        B_n \left(x,\frac{\delta}{2}\right)&=\left\{y\in X\left|d(x,y)<\frac{\delta}{2},d(Tx,Ty)<\frac{\delta}{2},\cdots,d \left(T^{n-1}x,T^{n-1}y\right)<\frac{\delta}{2}\right.\right\} \\
        &=B\left(x,\frac{\delta}{2}\right)\cap T^{-1}B\left(Tx,\frac{\delta}{2}\right)\cap\cdots\cap T^{-(n-1)}B\left(T^{n-1}x,\frac{\delta}{2}\right)
    \end{aligned}$\par
    则$\exists\, A_0,A_1,\cdots,A_n\in\alpha$，使得$B\left(x,\frac{\delta}{2}\right)\in A_0,B\left(Tx,\frac{\delta}{2}\right)\in A_1,\cdots,B\left(T^{n-1}x,\frac{\delta}{2}\right)\in A_{n-1}$.\par
    故$B_n\left(x,\frac{\delta}{2}\right)\subseteq A_0\cap T^{-1}A_1\cap\cdots\cap T^{-(n-1)}A_{n-1}\in\bigvee_{i=0}^{n-1}T^{-i}\alpha$.\par
    因此$\bigcup_{x\in F}B_n\left(x,\frac{\delta}{2}\right)$是$\bigvee_{i=0}^{n-1}T^{-i}\alpha$的一个子覆盖，故$N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)\leq r_n \left(\frac{\delta}{2}\right)$.
\end{proof}

\begin{lemma}{}
    设$\alpha$是$X$的开覆盖，满足$\mathrm{diam}\,\alpha<\varepsilon$，那么$s_n(\varepsilon)\leq N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)$
\end{lemma}
\begin{proof}
    设$\gamma\subseteq \bigvee_{i=0}^{n-1}T^{-i}\alpha$，使得$|\gamma|=N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)$.\par
    设$A_i\in\alpha,0\leq i\leq n-1$且$A_0\cap T^{-1}A_1\cap \cdots\cap T^{-(n-1)}A_{n-1}\in\gamma$.\par
    那么对于任意$(n,\varepsilon)-$分离集$E$，$|A_0\cap T^{-1}A\cap\cdots\cap T^{-(n-1)A_{n-1}}|\leq 1$.\par
    否则$\exists\, x,y\in A_0\cap T^{-1}A_1\cap\cdots\cap T^{-(n-1)}A_{n-1}\cap E$，由$\mathrm{diam}\, \alpha<\varepsilon$可知$d(x,y)<\varepsilon,\cdots,d(T^{-(n-1)}x,T^{-(n-1)}y)<\varepsilon$，矛盾.\par
    因此$s_n(\varepsilon)\leq |E|\leq |\gamma|=N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)$.
\end{proof}

最后，我们就可以证明定理\ref{thm:4.2}了.
\begin{theorem*}
    $h(T)=h_1(T)=h_2(T)$
\end{theorem*}
\begin{proof}
    \begin{itemize}
        \item $h_1(T)=h_2(T)$\par
        $r_n(\varepsilon)\leq s_n(\varepsilon)\leq r_n\left(\frac{\varepsilon}{2}\right)$\par
        令$n\to\infty,\varepsilon\to 0$可得$h_1(T)\leq h_2(T)\leq h_1(T)$，故$h_1(T)=h_2(T)$.\par
        \item $h_1(T)\geq h(T)$\par
        设$\alpha$是$X$的开覆盖，$\delta$为$\alpha$的Lebesgue数，则$N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)\leq r_n\left(\frac{\delta}{2}\right)$\par
        $\frac{1}{n}\ln{N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)}\leq \frac{1}{n}\ln{r_n\left(\frac{\delta}{2}\right)}$\par
        取上极限，$h(T,\alpha)=\limsup_{n\to\infty}\frac{1}{n}\ln{N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)}\leq \limsup_{n\to\infty}\frac{1}{n}\ln{r_n\left(\frac{\delta}{2}\right)}$\par
        再令$\delta\to 0$，则$h(T,\alpha)\leq h_1(T)$.对$\alpha$取上极限，则$h(T)\leq h_1(T)$.
        \item $h(T)\geq h_2(T)$.\par
        设$\alpha$是$X$的开覆盖，$\mathrm{diam}\,\alpha<\varepsilon$，那么$s_n(\varepsilon)\leq N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)$.\par
        取上极限可得$\limsup_{n\to\infty}{\frac{1}{n}\ln{s_n(\varepsilon)}}\leq \limsup_{n\to\infty}{\frac{1}{n}\ln{N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)}}=h(T,\alpha)\leq h(T)$.\par
        再令$\varepsilon\to 0$，则$h_2(T)\leq h(T)$.
    \end{itemize}\par
    综合以上三点可知$h(T)=h_1(T)=h_2(T)$.\par
\end{proof}

下面介绍拓扑熵的一些性质.\par

\begin{theorem}{}
    设$X$是紧致度量空间，$T:X\to X$连续映射，那么$\forall\, \ell\in\mb{N}$，$h(T^\ell)=\ell h(T)$
\end{theorem}
\begin{proof}
    设$\alpha$是$X$的开覆盖，$h(T,\alpha)=\lim_{n\to\infty}{\frac{1}{n\ell}H\left(\bigvee_{i=0}^{\ell n-1}T^{-i}\alpha\right)}$.\par
    令$\beta=\alpha\vee T^{-1}\alpha\vee\cdots\vee T^{-(\ell-1)}\alpha$，则$h(T,\alpha)=\lim_{n\to\infty}{\frac{1}{n\ell}H\left(\bigvee_{i=0}^{n-1}\left(T^{-\ell}\right)^i\beta\right)}=\frac{1}{\ell}h(T^\ell,\beta)$.\par
    因此$h(T,\alpha)=\frac{1}{\ell}h(T^\ell,\beta)\leq\frac{1}{\ell}h(T^\ell)$.\par
    对$\alpha$取上极限，$h(T)\leq \frac{1}{\ell}h(T^\ell)$.\par
    另一方面，由于$\beta$是$\alpha$的加细，故$h(T)\geq h(T,\alpha)=\frac{1}{\ell}h(T^\ell,\beta)\geq\frac{1}{\ell}h(T^\ell,\alpha)$.\par
    $\Rightarrow h(T)\geq\frac{1}{\ell}h(T^\ell)$.\par
    因此$h(T^\ell)=\ell h(T)$.
\end{proof}

\begin{theorem}{}
    设$(X_1,d_1),(X_2,d_2)$是紧致度量空间，$T_1:X_1\to X_1$，$T_2:X_2\to X_2$为连续映射，那么$h(T_1\times T_2)=h(T_1)+h(T_2)$.
\end{theorem}
\begin{proof}
    设$X=X_1\times X_2,T=T_1\times T_2$，定义$X$上的度量$d:\forall\, p=(x_1,x_2),q=(y_1,y_2)\in X,d(p,q)=\max\{d_1(x_1,y_1),d_2(x_2,y_2)\}$，则$(X,d)$是紧致度量空间.\par
    \begin{itemize}
        \item $h(T)\leq h(T_1)+h(T_2)$\par
        设$\varepsilon>0,n\in\mb{N}$，$F_1,F_2$分别是$X_1$和$X_2$的$(n,\varepsilon)-$生成集，设$F=F_1\times F_2$.
        \begin{claim}{$F$是$X$的$(n,\varepsilon)-$生成集}
            $\forall\, p\in X$，$p=(x_1,x_2)\in X_1\times X_2$.\par
            则$\exists\, y_1\in F_1$，使得$d_1(x_1,y_1)<\varepsilon,d(T_1x_1,T_1y_1)<\varepsilon,\cdots$\par
            $\exists\, y_2\in F_2$，使得$d_2(x_2,y_2)<\varepsilon,d(T_2x_2,T_2y_2)<\varepsilon,\cdots$\par
            记$q=(y_1,y_2)\in F_1\times F_2$，则$d(p,q)<\varepsilon,d(Tp,Tq)<\varepsilon,\cdots,d(T^{n-1}p,T^{n-1}q)<\varepsilon$
        \end{claim}\par
        因此，取$F_1$和$F_2$使得$|F_1|=r_n(\varepsilon,T_1),|F_2|=r_n(\varepsilon,T_2)$，则$F_1\times F_2$是$(n,\varepsilon)-$生成集.\par
        则$r_n(\varepsilon,T_1)\cdot r_n(\varepsilon,T_2)=|F_1|\times|F_2|=|F_1\times F_2|\geq r_n(\varepsilon,T_1\times T_2)$，即$h(T)\leq h(T_1)+h(T_2)$.
        \item $h(T)\geq h(T_1)+h(T_2)$\par
        设$E_1,E_2$是$X_1,X_2$的$(n,\varepsilon)-$分离集.\par
        \begin{claim}{$E=E_1\times E_2$是$X_1\times X_2$的$(n,\varepsilon)-$分离集}
            $\forall\, x,y\in E,x=(x_1,x_2),y=(y_1,y_2),x\neq y$.\par
            不妨设$x_1\neq y_2$，则$\exists\, i\in\{0,1,2,\cdots,n-1\}$，使得$d_1(T^ix,T^iy)>\varepsilon$\par
            $\Rightarrow d(T^ix,T^iy)>\varepsilon$.
        \end{claim}\par
        因此，取$E_1$和$E_2$使得$|E_1|=s_n(\varepsilon,T_1),|E_2|=s_n(\varepsilon,T_2)$，则$E_1\times E_2$是$(n,\varepsilon)-$分离集.\par
        则$s_n(\varepsilon,T_1)\cdot s_n(\varepsilon,T_2)=|E_1|\times|E_2|=|E_1\times E_2|\leq s_n(\varepsilon,T_1\times T_2)$，即$h(T)\geq h(T_1)+h(T_2)$.
    \end{itemize}  
\end{proof}

\section{拓扑熵的计算}

下面我们的目标是寻找类似于测度熵中Kolmogorov-Sinai定理的工具来计算拓扑熵.\par
\begin{definition}{可扩映射}
    设$X$是一个紧致度量空间，$T:X\to X$同胚，如果存在$e>0$，使得$\forall\, x,y\in X(x\neq y)$，$\exists\, n\in\mb{Z}$，$d(T^nx,T^ny)>e$，那么称$T:X\to X$是一个\textbf{可扩映射}，$e$称为\textbf{可扩常数}.
\end{definition}
\begin{theorem}{}
    设$X$是一个紧致度量空间，$T:X\to X$同胚，那么以下结论等价：
    \begin{enumerate}[(1)]
        \item $T$是可扩映射
        \item 存在$e>0$，使得$\forall\, x,y\in X,d\left(T^nx,T^ny\right)<\varepsilon(\forall\, n\in\mb{Z})\Rightarrow x=y$
        \item $T$有一个生成子覆盖，即$\alpha$是一个有限开覆盖，满足$\forall\, A_i\in\alpha(-\infty<i<+\infty)$，有$\left|\bigcap_{i=-\infty}^\infty{T^{-i}\overline{A_i}}\right|\leq 1$
        \item $T$有一个弱生成子覆盖，即$\alpha$是一个有限开覆盖，满足$\forall\, A_i\in\alpha(-\infty<i<+\infty)$，有$\left|\bigcap_{i=-\infty}^\infty{T^{-i}{A_i}}\right|\leq 1$
    \end{enumerate}
\end{theorem}
\begin{proof}
    $(1)\Leftrightarrow (2)$：定义可得 \par
    $(3)\Rightarrow (4)$：定义可得 \par
    $(4)\Rightarrow (3)$：如果$T$有一个弱生成子覆盖$\alpha$，设$\delta$是$\alpha$的Lebesgue数，取$X$的一个有限子覆盖$\beta$，使得$\mathrm{diam}\,\beta<\frac{\delta}{2}$.\par
    \begin{claim}{$\beta$是$X$的一个生成子覆盖}
        设$B_i\in\beta(-\infty<i<+\infty)$，考虑$\bigcap_{i=-\infty}^{+\infty}{T^{-i}\overline{B_i}}$.\par
        如果$x,y\in\bigcap_{i=-\infty}^{+\infty}T^{-i}\overline{B_i}$，那么$T^ix,T^iy\in\overline{B_i}$.\par
        由于$\mathrm{diam}\,\overline{B_i}\leq\frac{\delta}{2}<\delta\Rightarrow \exists\, A_i\in\alpha,\overline{B_i}\subseteq A_i$.\par
        则$x,y\in\bigcap_{i=-\infty}^{+\infty}{T^{-i}A_i}$，则$x=y$，从而$\left|\bigcap_{i=-\infty}^{+\infty}{T^{-i}\overline{B_i}}\right|\leq 1$.
    \end{claim}\par
    $(1)\Rightarrow (3)$：假设$T$是可扩映射，$e$是$T$的可扩常数，设$\alpha$是$X$的一个有限开覆盖，$\mathrm{diam}\,\alpha<\frac{e}{2}$.\par
    则$\forall\, A_i\in\alpha$，$x,y\in\bigcap_{i=-\infty}^{+\infty}{T^{-i}\overline{A_i}}\Rightarrow T^ix,T^iy\in\overline{A_i}\Rightarrow d\left(T^ix,T^iy\right)\leq\frac{e}{2}(\forall\, i\in\mb{Z})$\par
    $\Rightarrow x=y$，从而$\left|\bigcap_{i=-\infty}^\infty{T^{-i}\overline{A_i}}\right|\leq 1$.\par
    $(3)\Rightarrow (1)$：设$\alpha$是$T$的生成子覆盖，即有限开覆盖满足$\forall\,\{A_i\}_{i\in\mb{Z}}(A_i\in\alpha),\left|\bigcap_{i=-\infty}^{+\infty}{T^{-i}}\overline{A}\right|\leq 1$.\par
    记$\delta$为$\alpha$的Lebesgue数，现证$\frac{\delta}{3}$是$T$的可扩常数.\par
    $\forall\,x,y\in X(x\neq y)$，如果$\forall\, n\in\mb{Z}$，$d\left(T^nx,T^ny\right)\leq\frac{\delta}{3}$，那么$\exists\, B_n\in\alpha(n\in\mb{Z})$，使得$T^nx,T^ny\in B_n$.\par
    $\Rightarrow x,y\in\bigcap_{n=-\infty}^{+\infty}{T^{-n}B_n}$.由$\alpha$是生成子覆盖可知$\left|\bigcap_{n=-\infty}^{+\infty}B_n\right|\leq 1$.\par
    因此$x=y$，矛盾!故$T$是可扩映射，$\frac{\delta}{3}$为可扩常数.
\end{proof}

\begin{lemma}{}
    设$(X,d)$是紧致度量空间，$T:X\to X$同胚，设$\alpha$是$T$的一个生成子覆盖，那么$\forall\,\varepsilon>0$，$\exists\, N\in\mb{N}$，使得$\mathrm{diam}\left(\bigvee_{i=-N}^NT^{-i}\alpha\right)<\varepsilon$.
\end{lemma}
\begin{proof}
    设$\varepsilon_0>0$，使得$\forall\, N\in\mb{N}$，$\mathrm{diam}\left(\bigvee_{i=-N}^N{T^{-1}\alpha}\right)\geq\varepsilon_0$.\par
    那么$\forall\, j\in\mb{N}$，$\exists\,A^j_i\in\alpha(-j\leq i\leq j)$，使得$\mathrm{diam}\left(\bigcap_{i=-j}^{j}T^{-i}A_i^j\right)\geq\varepsilon_0$
    任取其中两点$x_j,y_j$使得$d(x_j,y_j)\geq\frac{\varepsilon_0}{2}$\par
    则可得到$\{x_j\}_{j\in\mb{N}}$和$\{y_j\}_{j\in\mb{N}}$两个序列，由$X$的紧致性可知其有收敛子列，不妨仍记作$\{x_j\},\{y_j\}$\par
    记$x_0=\lim_{j\to\infty}{x_j},y_0=\lim_{j\to\infty}{y_j}$，有$d(x_0,y_0)\geq\frac{\varepsilon_0}{2}$.\par
    $x_j,y_j\in\bigcap_{i=-j}^jT^{-i}A^j_i$，取$i=0$，则$x_j,y_j\in A_0^j$.\par
    由于$\alpha$是有限开覆盖，$A_0^j$的选择只有有限多种.\par
    因此$\exists\, A_0\in\alpha$和$\{x_j\},\{y_j\}$的子列$\left\{x_{n_j}\right\},\left\{y_{n_j}\right\}$使得$x_{n_j},y_{n_j}\in A_0\Rightarrow x_0,y_0\in\overline{A}$.\par
    对于其他位置，固定$i$，则$x_j,y_j\in T^{-i}A_i^j$，同理可得$x_0,y_0\in T^{-i}\overline{A}$.\par
    故$x_0,y_0\in\bigcap_{j=-\infty}^{+\infty}T^{-i}\overline{A}$，由于$\alpha$是生成子覆盖，因此$x_0=y_0$，与$d(x_0,y_0)\geq\frac{\varepsilon_0}{2}$矛盾.
\end{proof}

下面就可以正式证明使用生成子覆盖计算拓扑熵的定理了.
\begin{theorem}{}
    设$(X,d)$是紧致度量空间，$T:X\to X$同胚，如果$\alpha$是$T$的一个生成子覆盖，那么$h(T)=h(T,\alpha)$.
\end{theorem}
\begin{proof}
    对于$X$的任意开覆盖$\beta$，需证$h(T,\beta)\leq h(T,\alpha)$.\par
    设$\delta$为$\beta$的Lebesgue数，则$\exists\, N\in\mb{N}$，使得$\mathrm{diam}\left(\bigvee_{i=-N}^NT^{-i}\alpha\right)<\delta$.\par
    由Lebesgue数定义可知$\bigvee_{i=-N}^NT^{-i}\alpha\geq\beta$.\par
    $\setjot[-2]\begin{aligned}[t]
        h(T,\beta) &\leq h\left(T,\bigvee_{i=-N}^NT^{-i}\alpha\right)=\lim_{n\to\infty}{\frac{1}{n}H\left(\bigvee_{j=0}^{n-1}T^{-j}\left(\bigvee_{i=-N}^NT^{-i}\alpha\right)\right)} \\
        &=\lim_{n\to\infty}{\frac{1}{n}H\left(\bigvee_{i=-N}^{N+n-1}T^{-i}\alpha\right)}=\lim_{n\to\infty}{\frac{1}{n}H\left(\bigvee_{i=0}^{2N+n-1}T^{-i}\alpha\right)}\\
        &=\lim_{m\to\infty}{\frac{1}{m-2N}H\left(\bigvee_{i=0}^{m-1}T^{-i}\alpha\right)} \\
        & =\lim_{m\to\infty}{\frac{1}{m}H\left(\bigvee_{i=0}^{m-1}T^{-i}\alpha\right)}=h(T,\alpha)   
    \end{aligned}$
\end{proof}

\begin{instance}
    设$X=\prod_{i=-\infty}^\infty{X_i},X_i=\{1,2,\cdots,k\}(\forall\,i\in\mb{Z})$.\par
    $T:X\to X$是左平移映射，其为紧致度量空间上的同胚，求其拓扑熵.\par
    \begin{itemize}
        \item $T$是可扩映射.\par
        $\forall\, x=(\cdots,x_{-1},x_0,x_1,\cdots),y=(\cdots,y_{-1},y_0,y_1,\cdots)$，$d(x,y)=\sum_{i=-\infty}^\infty{\frac{|x_i-y_i|}{k^{|i|}}}$.\par
        如果$x\neq y$，那么$\exists\, i_0\in\mb{Z}$使得$x_{i_0}\neq y{i_0}$.\par
        则$T^{-i_0}x$和$T^{-i_0}y$在第$0$个位置相同，因此$d\left(T^{i_0}x,T^{i_0}y\right)\geq 1>\frac{1}{2}$，因此$\frac{1}{2}$是一个可扩常数.
        \item 设$A_i=\left\{(\cdots,x_{-1},x_0,x_1,\cdots)\right|x_0=i\}(1\leq i\leq k)$，则$\alpha=\left\{A_1,A_2,\cdots,A_k\right\}$是$X$的一个开覆盖.\par
        \begin{enumerate}[(1)]
            \item $\forall\, i\in\{1,2,\cdots,k\}$，$A_i$为闭集(如果$\{y_j\}\subseteq A_i$，$y_j$收敛于$y_0$，那么$y_0\in A_i$)
            \item $A_1\cup A_2\cup\cdots\cup A_k=X,A_i\cap A_j=\varnothing(i\neq j)$
            \item 由$(1)(2)$可知$A_i$是开集
        \end{enumerate}
        现证$\alpha$是$T$的一个生成子覆盖.\par
        取$A_i\in\alpha(\forall\, i\in\mb{Z})$，$\bigcap_{i=-\infty}^\infty T^{-i}\overline{A_{n_i}}\xlongequal{\text{闭集}}\bigcap_{i=-\infty}^\infty T^{-i}A_{n_i}=\left\{(\cdots,n_{-1},n_0,n_1,\cdots,n_i,\cdots)\right\}$\par
    \end{itemize}
    $h(T,\alpha)=\lim_{n\to\infty}\frac{1}{n}H\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)=\lim_{n\to\infty}{\frac{1}{n}\ln{N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)}}$\par
        其中$\bigvee_{i=0}^{n-1}T^{-i}\alpha=\left\{A_{i_0}\cap T^{-1}A_{i_1}\cap\cdots\cap T^{-(n-1)}A_{i_{n-1}}\Big|1\leq i_0,i_1,\cdots,1_{n-1}\geq k\right\}$为两两不相交的集族.\par
        故$N\left(\bigvee_{i=0}^{n-1}T^{-i}\alpha\right)=k^n\Rightarrow h(T,\alpha)=\lim_{n\to\infty}\frac{1}{n}\ln{k^n}=\ln{k}=h(T)$.
\end{instance}

\begin{instance}
    设$S^1$是周长为1的圆周，$T:S^1\to S^1$同胚，证明：$h(T)=0$.
\end{instance}
\begin{proof}
    $\forall\,\varepsilon>0,n\in\mb{N}$，构造$S^1$中的$(n,\varepsilon)-$生成集(不妨设$\varepsilon>0$充分小).\par
    $(0,\varepsilon)-$生成集$F_0$，即为将圆周等分并使得每一段弧长小于$\varepsilon$的点集.\par
    设已经构造了$(n-1,\varepsilon)-$生成集，现构造$(n,\varepsilon)-$生成集$F_n=F_{n-1}\cup T^{-(n-1)}F_0$.\par
    由于$T$同胚，$T$保顺序，不妨设$T$保持顺时针方向.\par
    $\forall\, x\in S^1,\exists\, w\in F_{n-1}$，使得$d(x,w)<\varepsilon,d(Tx,Tw)<\varepsilon,\cdots,d(T^{n-2}x,T^{n-2}2)<\varepsilon$(不妨设$w$是$F_{n-1}$中离$x$最近的点).\par
    则$T^{n-1}x$在$T^{n-1}w$相邻的两个区间中.\par
    $\Rightarrow \exists\, T^{n-1}F_n=T^{n-1}F_{n-1}\cup F_0$中相邻的两个点$\{p_1,p_2\}$，使得$T^{n-1}x\in [p_1,p_2],\left|[p_1,p_2]\right|<\varepsilon$.\par
    令$y=T^{-(n-2)}y\Rightarrow d\left(T^{n-2}x,T^{n-2}y\right)<\varepsilon$.\par
    $d\left(T^{n-2}p,T^{n-3}x\right)<\varepsilon\Rightarrow d\left(T^{n-3}x,T^{n-3}y\right)<\varepsilon$，以此类推 \par
    $\Rightarrow F_n$是$(n,\varepsilon)-$生成集.\par
    $|F_N|\geq(n+1)\left(\left[\frac{1}{\varepsilon}\right]+1\right)\Rightarrow r_n(\varepsilon)\geq (n+1)\left(\left[\frac{1}{\varepsilon}\right]+1\right)$\par
    $h(T)=\lim_{\varepsilon\to 0}\limsup_{n\to\infty}\frac{1}{n}\ln{r_n(\varepsilon)}=0$
\end{proof}
