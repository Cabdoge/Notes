\chapter{测度熵}


\section{测度熵的概念}


本章介绍的对象主要为保测映射的熵，它可以帮助我们判断两个保测映射是否是同构的，因此，先介绍同构的概念.

\begin{definition}{可逆保测映射}
    如果一个映射$\Phi$满足：
    \begin{enumerate}[(1)]
        \item $\Phi$是保测映射，即$\forall\, E\subseteq Y_2,m_2(E)=m_1\left(\Phi^{-1}(E)\right)$
        \item $\Phi$可逆，且$\Phi^{-1}$也是保测映射
    \end{enumerate}
    则称$\Phi$是一个\textbf{可逆保测映射}
\end{definition}

\begin{definition}{同构}
    设$(X_1,\mcal{B}_1,m_1)$和$(X_2,\mcal{B}_2,m_2)$是概率测度空间，$T_1:X_1\to X_1$和$T_2:X_2\to X_2$是保测映射.如果存在$Y_1\subseteq X_1,Y_2\subseteq X_2$，且$m_i(Y_i)=1,T(Y_i)\subseteq Y_i(i=1,2)$，以及一个可逆保测映射$\Phi:Y_1\to Y_2$，使得交换图成立： \par
    \begin{tikzcd}
Y_1 \arrow[dd, "\Phi"'] \arrow[rr, "T_1"] &  & Y_1 \arrow[dd, "\Phi"] \\
                                          &  &                        \\
Y_2 \arrow[rr, "T_2"']                    &  & Y_2                   
\end{tikzcd}
\quad 即$\Phi\circ T_1=T_2\circ \Phi$\par
那么称$T_1$与$T_2$\textbf{同构}.
\end{definition}

\begin{question}
    我们要解决的问题是：如何判断两个保测映射是否同构？\par
\end{question}

\begin{instance}
    $T_2:\prod_{i=-\infty}^\infty\{1,2\}\to\prod_{i=-\infty}^\infty\{1,2\},T_3:\prod_{i=-\infty}^\infty\{1,2,3\}\to\prod_{i=-\infty}^\infty\{1,2,3\}$是否同构?
\end{instance}

\begin{definition}{有限可测分解}
    设$(X,\mcal{B},m)$是一个概率测度空间，$\eta=\{A_1,A_2,\cdots,A_n\}(A_i\in\mcal{B},i=1,2,\cdots,n)$，如果$\eta$满足：
    \begin{enumerate}[(1)]
        \item $\forall\, i,j\in\{1,2,\cdots,n\},i\neq j,m\left(A_i\cap A_j\right)=0$
        \item $m\left(\bigcup_{i=1}^n A_i\right)=1$
    \end{enumerate}
    则称$\eta$是一个\textbf{有限可测分解}.
\end{definition}

对于任意一个有限可测分解，可以类似于信息熵的定义方式，定义其测度熵：
\begin{definition}{有限可测分解的测度熵}
    设$\eta=\{A_1,A_2,\cdots,A_n\}$是$X$的一个有限可测分解，定义$H(\eta)=-\sum_{i=1}^n m(A_i)\log m(A_i)$，称为$\eta$的\textbf{测度熵}.
\end{definition}

上述是对一个有限可测分解的研究，如果$X$有多个有限可测分解，可以定义它们之间的运算.
\begin{definition}{有限可测分解的交}
    设$\xi=\{A_1,A_2,\cdots,A_n\},\eta=\{B_1,B_2,\cdots,B_m\}$，定义
    $$\xi\vee\eta=\left\{A_i\cap B_j\left|1\leq i\leq n,1\leq j\leq m\right.\right\}$$
    则$\xi\vee\eta$是$X$的有限可测分解，称为$\xi$和$\eta$的\textbf{交}.
\end{definition}

特别地，如果$T:(X,\mcal{B},m)\to(X,\mcal{B},m)$是一个保测映射，$\eta=\{A_1,A_2,\cdots,A_n\}$是$X$的有限可测分解，那么$T^{-1}\eta=\{T^{-1}A_1,T^{-1}A_2,\cdots,T^{-1}A_n\}$也有有限可测分解.进而$\forall\, k\in\mb{N}$，$T^{-k}\eta=\{T^{-k}A_1,T^{-k}A_2,\cdots,T^{-k}A_n\}$都是$X$的有限可测分解.\par
那么就可以定义$\bigvee_{i=0}^1T^{-i}\eta=T^{-1}\eta\vee\eta,\bigvee_{i=0}^2T^{i}\eta=\eta\vee T^{-1}\eta\vee T^{-2}\eta,\cdots,\bigvee_{i=0}^{n-1}T^{-i}\eta=\eta\vee T^{-1}\eta\vee\cdots\vee T^{-(n-1)}\eta$.\par
利用上述一列有限可测分解的交会将$X$分解得到更细的程度，因此，就把上述交的测度熵的极限定义为$T$和$\eta$共同的测度熵(因为结果依赖于$\eta$的最初选取)：
\begin{definition}{保测映射关于有限可测分解的测度熵}
    定义$h(T,\eta)=\lim_{n\to\infty}\frac{1}{n}H\left(\bigvee_{i=0}^{n-1}T^{-i}\eta\right)$，如果极限存在，那么称$h(T,\eta)$为\textbf{$\bm{T}$关于$\bm{\eta}$的测度熵}.
\end{definition}
按照上述的想法，如果想要单独定义$T$的测度熵，而不依赖于$\eta$的选取，一个最简单的方式就是取遍所有的有限可测分解，并将$T$关于其中每个有限可测分解的测度熵的上确界定义为$T$的测度熵，这一思想在定义函数的全变差时亦有所体现.\par
\begin{definition}{保测映射的测度熵}
    $h(T)=\sup_{\eta\,\text{有限可测分解}} h(T,\eta)$称为保测映射$T$的\textbf{测度熵}.
\end{definition}

现在的遗留问题是在定义保测映射关于有限可测分解的测度熵时，所使用的极限是否存在、是否有限.回答问题的核心是$\phi(x)=x\ln x$在$[0,1]$上是一个凸函数，并且需要证明下面的两个引理：
\begin{lemma}{}{3.1}
    设$\xi=\{A_1,A_2,\cdots,A_n\},\eta=\{B_1,B_2,\cdots,B_m\}$是$X$的2个有限可测分解，那么$H(\xi\vee\eta)\leq H(\xi)+H(\eta)$.
\end{lemma}
\begin{proof}
    $\setjot[-2]\begin{aligned}[t]
        H(\xi\vee\eta) &=-\sum_{\substack{1\leq i\leq n\\ 1\leq j\leq m}}{m(A_i\cap B_j)\ln m(A_i\cap B_j)} \\
        &=-\sum_{i=1}^n\sum_{j=1}^m{\left(m(A_i\cap B_j)\ln{m(A_i)}+m(A_i\cap B_j)\ln{\frac{m(A_i\cap B_j)}{m(A_i)}}\right)} \\
        &=-\sum_{i=1}^n m(A_i) \ln m(A_i) - \sum_{i=1}^n \sum_{j=1}^m m(A_i)\frac{m(A_i \cap B_j)}{m(A_i)}\ln \frac{m(A_i \cap B_j)}{m(A_i)} \\
        &=H(\xi) - \sum_{i=1}^n \sum_{j=1}^m m(A_i)\frac{m(A_i \cap B_j)}{m(A_i)}\ln \frac{m(A_i \cap B_j)}{m(A_i)} \\
    \end{aligned}$\par
    $\setjot[-2]\begin{aligned}[t]
        \hspace{63.8pt}&\leq H(\xi)-\sum_{j=1}^m\phi \left(\sum_{i=1}^n m(A_i)\frac{m(A_i\cap B_j)}{m(A_i)}\right) &\left(\sum_{i=1}^n m(A_i)=1\,\text{应用琴生不等式}\,\right) \\
        &=H(\xi)-\sum_{j=1}^m\phi \left(m(B_j)\right)  \\
        &=H(\xi)-\sum_{j=1}^m m(B_j)\ln m(B_j)=H(\xi)+H(\eta) \\
    \end{aligned}$\par
\end{proof}

\begin{lemma}{}{3.2}
    设$\forall\,n\geq 0,a_n\geq 0$，且满足$a_{m+n}\leq a_m+a_n\;(\forall\, m,n\geq 0)$，那么$\lim_{n\to\infty}{\frac{1}{n}a_n}$存在.
\end{lemma}
\begin{proof}
    $\forall\,m,n\in\mb{N}$，设$n=qm+r\;(0\leq r<m,q\in\mb{N})$.\par
    则$a_n=a_{qm+r}\leq a_{qm}+a_r\leq a_{(q-1)m}+a_m+a_r\leq\cdots\leq qa_m+a_r$.\par
    对不等式两边同除$n$，则有$\frac{a_n}{n}\leq\frac{q a_m+a_r}{qm+r}\leq\frac{a_m}{m}+\frac{a_r}{qm}$.\par
    固定$m$，令$q\to\infty$，则$n\to\infty$，因而$\varlimsup_{n\to\infty}\frac{a_n}{n}\leq\frac{a_m}{m}\;(\forall\, m\in\mb{N})$.\par
    再令$m\to\infty$，则$\varlimsup_{n\to\infty}{\frac{a_n}{n}}\leq\varliminf_{n\to\infty}\frac{a_m}{m}\Rightarrow$极限存在.\par
    又$0\leq\frac{a_n}{n}\leq a_1$，故极限有界.
\end{proof}

借助这两个引理就可以证明定义测度熵时的极限存在.\par
\begin{corollary}
    $\lim_{n\to\infty}\frac{1}{n}H\left(\bigvee_{i=0}^{n-1}T^{-i}\eta\right)$存在.
\end{corollary}
\begin{proof}
    令$a_n=H\left(\bigvee_{i=0}^{n-1}T^{-i}\eta\right),a_0=0$，那么$\forall\,n\geq 0,a_n\geq 0$.\par
    $\setjot[-2]\begin{aligned}[t]
        a_{m+n}&=H\left(\bigvee_{i=0}^{m+n-1}{T^{-i}\eta}\right)=H\left(\bigvee_{i=0}^{n-1}{T^{-i}\eta}+\bigvee_{i=m}^{n+m-1}{T^{-i}\eta}\right) \\
        &\leq H\left(\bigvee_{i=0}^{n-1}{T^{-i}\eta}\right)+H\left(\bigvee_{i=m}^{n+m-1}{T^{-i}\eta}\right) \\
        &=a_m+ H\left(T^{-m}\left(\bigvee_{i=0}^{n-1}{T^{-i}\eta}\right)\right) \\
        &\xlongequal{T\,\text{保测}}a_m+H\left(\bigvee_{i=0}^{n-1}{T^{-i}\eta}\right)=a_m+a_n \\
    \end{aligned} $\par
    应用引理\ref{lem:3.2}即知极限存在.
\end{proof}

\section{条件熵与测度熵}
在证明引理\ref{lem:3.1}时，过程中出现了一串奇怪的乘积式，这一串式子被我们称为条件熵.
\begin{definition}{条件熵}
    设$\xi=\{A_1,A_2,\cdots,A_n\},\eta=\{B_1,B_2,\cdots,B_m\}$是$X$的两个有限可测分解，定义
    $$H(\eta|\xi)=\sum_{i=1}^n{m(A_i)\left(-\sum_{j=1}^m{\frac{m(A_i\cap B_j)}{m(A_i)}\ln{\frac{m(A_i\cap B_j)}{m(A_i)}}}\right)}$$
    称为$\eta$关于$\xi$的\textbf{条件熵}.
\end{definition}
将条件熵的定义引入引理\ref{lem:3.1}中，就可以得知条件熵的两个重要性质：
\begin{lemma}{}
    $H(\xi\vee\eta)=H(\xi)+H(\eta|\xi)\quad H(\eta|\xi)\leq H(\eta)$
\end{lemma}

\begin{instance}
    设$\eta=\{A_1,A_2,\cdots,A_n\}$是$X$的一个有限可测分解，$T:X\to X$是保测映射，那么就有$H(\eta)\leq\ln{n},h(T,\eta)\leq\ln{n}$
    \begin{solution}
        $\setjot[-2]\begin{aligned}[t]
            H(\eta) &=-\sum_{i=1}{m(A_i)\ln{m(A_i)}}=-n\sum_{i=1}^n{\frac{1}{n}\phi(m(A_i))} \\
            &\leq (-n)\phi \left(\sum_{i=1}^n\frac{1}{n}m(A_i)\right) \\
            &=(-n)\phi\left(\frac{1}{n}\right)=\ln{n}
        \end{aligned}$\par
        $h(T,\eta)=\lim_{k\to\infty}{\frac{1}{k}H\left(\bigvee_{i=0}^{k-1}T^{-i}\eta\right)}\leq\varlimsup_{k\to\infty}{\frac{1}{k}\ln{n^k}}=\ln n$.
    \end{solution}
\end{instance}
上面的例子说明一个有限可测分解的测度熵具有上界.

关于条件熵，还有以下等式成立
\begin{lemma}{}
    设$(X,\mcal{B},m)$ 是概率测度空间，$\xi_1,\xi_2,\eta$ 是$X$ 的有限可测分解，则$H \left(\xi_1\vee\xi_2|\eta\right)=H(\xi_1|\eta)+H(\xi_2|\xi_1\vee\eta)$
\end{lemma}
\begin{proof}
    对于$X$的任意有限可测分解$\a$，有$H(\alpha\vee\eta)=H(\eta)+H(\alpha|\eta)$.\par
    $\Rightarrow \setjot[-2]\begin{aligned}[t]
        &H\left(\xi_1\vee\xi_2|\eta\right)=H\left(\xi_1\vee\xi_2\vee\eta\right)-H(\eta)\\
        &H(\xi_1|\eta)=H\left(\xi_1\vee\eta\right) - H(\eta) \\
        &H\left(\xi_2|\xi_1\vee\eta\right)=H\left(\xi_1\vee\xi_2\vee\eta\right)-H\left(\xi_1\vee\eta\right) \\
    \end{aligned}$
\end{proof}

\begin{definition}{加细}
    设$\xi$ 和$\eta$ 是$X$ 的有限可测分解，如果$\xi$ 中的任何一个元素都是$\eta$ 中若干个元素的并，那么称$\eta$ 是$\xi$ 的\textbf{加细}，记为$\xi\leq\eta$
\end{definition}

\begin{lemma}{}
    设$\xi_1\leq\xi_2$，那么$H(\eta|\xi_1)\geq H(\eta|\xi_2)$
\end{lemma}
\begin{proof}
    设$\xi_1=\{A_1,A_2,\cdots,A_n\},\xi_2=\{B_1,B_2,\cdots,B_m\},\eta=\{C_1,C_2,\cdots,C_{\ell}\}$.\par
    $\setjot[-2]\begin{aligned}[t]
        H(\eta|\xi_2)&=-\sum_{i=1}^n{m(B_j)\sum_{j=1}^\ell{\frac{m\left(C_i\cap B_j\right)}{m(B_j)}\ln{\frac{m\left(C_i\cap B_j\right)}{m(B_j)}}}} =-\sum_{j=1}^\ell{\sum_{i=1}^n{m\left(C_i\cap B_j\right)\ln{\frac{m\left(C_i\cap B_j\right)}{m(B_j)}}}} \\
        &=-\sum_{j=1}^\ell\left(\sum_{k=1}^m\sum_{B_i\in A_k}{m\left(C_i\cap B_j\right)\ln{\frac{m\left(C_i\cap B_j\right)}{m(B_j)}}}\right) \\
        &=-\sum_{j=1}^\ell\left(\sum_{k=1}^m m(A_k) \sum_{B_i\in A_k}{\frac{m(B_i)}{m(A_k)}\frac{m\left(C_i\cap B_j\right)}{m(B_i)}\ln{\frac{m\left(C_i\cap B_j\right)}{m(B_j)}}}\right) \\
        &=-\sum_{j=1}^\ell\left(\sum_{k=1}^m m(A_k) \sum_{B_i\in A_k}{\frac{m(B_i)}{m(A_k)}\phi \left(\frac{m\left(C_i\cap B_j\right)}{m(B_i)}\right)}\right) \\
        &\xlongequal{\text{琴生不等式}} -\sum_{j=1}^\ell\left(\sum_{k=1}^m m(A_k) \phi \left(\sum_{B_i\in A_k}{\frac{m(B_i)}{m(A_k)}\frac{m\left(C_i\cap B_j\right)}{m(B_i)}}\right)\right) \\
        &=-\sum_{j=1}^\ell\left(\sum_{k=1}^m m(A_k) \phi \left(\frac{m\left(C_i\cap A_k\right)}{m(A_k)}\right)\right)\\
        &=-\sum_{j=1}^\ell\sum_{k=1}^m{m(A_k)\frac{m(C_j\cap A_k)}{m(A_k)}}\ln\frac{m(C_j\cap A_k)}{m(A_k)}=H(\eta|\xi_1) \\
    \end{aligned}$\par
\end{proof}
从信息论的角度来看，上述不等式意味着拥有的信息越多，同一个事件的混乱程度就越低.\par
基于以上，还有下述推论：
\begin{corollary}{}
    $H(\xi_1\vee\xi_2|\eta)\leq H(\xi_1|\eta)+H(\xi_2|\eta)$.\par
    如果$\eta=\{X\}$，则$H(\alpha|\eta)=H(\alpha)$.
\end{corollary}


\section{测度熵是同构不变量}
下面我们要说明的是，同构的保测映射具有相同的测度熵.首先介绍一个定理：
\begin{theorem}{}
    如果$(X,\mcal{B},m)$ 是概率测度空间，$T:X\to X$ 是保测映射，那么$\forall\, n\in\mb{N},h(T^\ell)=\ell h(T)$.\par
    如果$T$ 是可逆保测映射，那么$h(T^{-1})=h(T)$
\end{theorem}
\begin{proof}
    设$\eta$ 是$X$ 的一个有限可测分解，令$\xi=\eta\vee T^{-1}\eta\vee \cdots\vee T^{-(\ell-1)}\eta$.\par
    那么$\bigvee_{i=0}^{n-1}\left(T^\ell\right)^{-i}\xi=\eta\vee T^{-1}\eta\vee\cdots T^{-(\ell-1)}\eta\vee\cdots\vee T^{-(n\ell-1)}\eta=\bigvee_{i=0}^{n\ell-1}{T^{-i}\eta}$.\par
    $\Rightarrow \setjot[-2]\begin{aligned}[t]
        h \left(T^{\ell},\xi\right)&=\lim_{n\to\infty}{\frac{1}{n}H \left(\bigvee_{i=0}^{n-1}\left(T^\ell\right)^{-i}\xi\right)} =\lim_{n\to\infty}{\frac{1}{n}H \left(\bigvee_{i=0}^{n\ell-1}T^{-i}\eta\right)}\\
        &=\ell\cdot\lim_{n\to\infty}{\frac{1}{nl}H \left(\bigvee_{i=0}^{n\ell-1}{T^{-i}\eta}\right)} \\
        &=\ell h(T,\eta)
    \end{aligned} $\par
    因此$h \left(T^\ell\right)\geq\sup_{\xi}{h \left(T^\ell,\xi\right)}=\ell \sup_{\eta}{h(T,\eta)}=\ell h(T)$.\par
    
    如果$\alpha\geq\beta$，$H(\alpha)=H(\alpha\vee\beta)=H(\beta)+H(\alpha|\beta)\geq H(\beta)$.\par
    因此由$\xi\geq\eta$ ，可得$h \left(T^\ell,\xi\right)\geq h(T^\ell,\eta)$.\par
    $\Rightarrow h \left(T^\ell,\eta\right)\leq \ell h \left(T,\eta\right)$.\par
    取上确界，可得$\sup_{\eta}{h \left(T^\ell,\eta\right)}\leq \ell\sup_{\eta}{h(T,\eta)}$ ，即$h \left(T^\ell\right)\leq \ell h(T)$.\par
    $\Rightarrow h \left(T^\ell\right)=\ell h(T)$.\par
    若$T$ 是可逆保测映射，$\eta$ 是$X$ 是有限可测分解，\par
    则$\setjot[-2]\begin{aligned}[t]
        h(T,\eta) &=\lim_{n\to\infty}{\frac{1}{n}H \left(\eta\vee T^{-1}\eta\vee\cdots\vee T^{-(n-1)}\eta\right)} \\
        &\xlongequal{T\,\text{可逆保测}}\lim_{n\to\infty}{\frac{1}{n}H\left(T^{n-1}\left(\eta\vee T^{-1}\eta\vee\cdots\vee T^{-(n-1)}\eta\right)\right)} \\
        &=\lim_{n\to\infty}{\frac{1}{n}H \left(\eta\vee T \eta\vee\cdots\vee T^{n-1}\eta\right)}\\
        &=h(T^{-1},\eta) \\
    \end{aligned}$\par
    取上确界可得，$h(T)=h\left(T^{-1}\right)$.
\end{proof}

\begin{theorem}{同构保持测度熵}
    设$(X_1,\mcal{B}_1,m_1)$和$(X_2,\mcal{B}_2,m_2)$ 是概率测度空间，$T_1:X_1\to X_1$和$T_2:X_2\to X_2$是保测映射.如果$T_1$和$T_2$ 同构，那么$h_{m_1}\left(T_1\right)=h_{m_2}\left(T_2\right)$.
\end{theorem}
\begin{proof}
    由于$T_1$与$T_2$同构，存在 $M_1\subseteq X_1,M_2\subseteq X_2,m_1(M_1)=m_2(M_2)=1$，$T_1(M_1)\subseteq M_1,T_2(M_2)\subseteq T_2(M_2)$以及$\Phi:M_1\to M_2$是可逆保测变换，使得$\Phi\circ T_1=T_2\circ\Phi$.\par
    可以推得$\Phi\circ T_1^i=T_2^i\circ\Phi,T^{-1}\circ\Phi^{-1}=\Phi^{-1}\circ T_2^{-1}$.\par
    设$\eta$ 是$M_2$ 的有限可测分解(可视作$X_2$的有限可测分解).\par
    那么$\Phi^{-1}(\eta)$ 是$M$ 的有限可测分解(视作$X_1$ 的有限可测分解).\par
    $\setjot[-2]\begin{aligned}[t]
        h_{m_1}\left(T_1,\Phi^{-1}(\eta)\right) &=\lim_{n\to\infty}{\frac{1}{n}H \left(\bigvee_{i=0}^{n-1}T_1^{-i}\left(\Phi^{-1}(\eta)\right)\right)} \\
        &=\lim_{n\to\infty}{\frac{1}{n}H \left(\bigvee_{i=0}^{n-1}\Phi^{-1}\left(T_2^{-i}(\eta)\right)\right)} \\
        &\xlongequal{\Phi\,\text{保测}} \lim_{n\to\infty}{\frac{1}{n}H \left(\bigvee_{i=0}^{n-1}T_2^{-i}(\eta)\right)} \\
        &=h_{m_2}\left(T_2,\eta\right) 
    \end{aligned}$\par
    设$\xi$ 是$M_1$的有限可测分解，同理可得$h_{m_2}\left(T_2,\Phi(\xi)\right)=h_{m_1}\left(T_1,\xi\right)$.\par
    对上述两式取上确界，得$h_{m_1}(T_1)=h_{m_2}(T_2)$.\par
\end{proof}



\section{测度熵的计算}
上一节的结论告诉我们两个同构的保测映射必须有相同的测度熵，因此测度熵可以作为保测映射同构与否的一个判据.接下来我们介绍如何计算测度熵.\par
为了计算测度熵，我们引入一个特殊的有限可测分解：
\begin{definition}{$\sigmaalg$上的等价关系}
    设$(X,\mcal{B},m)$是一个概率测度空间，$\mcal{B}_1,\mcal{B}_2\subseteq \mcal{B}$是2个子$\sigmaalg$，如果：
    \begin{enumerate}[(1)]
        \item $\forall\, E\in\mcal{B}_1,\exists\, F\in\mcal{B}_2$，使得$m \left(E\triangle F\right)=0$
        \item $\forall\, F\in\mcal{B}_2,\exists\, E\in\mcal{B}_1$，使得$m \left(F\triangle E\right)=0$
    \end{enumerate}
    那么记$\mcal{B}_1\circeq \mcal{B}_2$.
\end{definition}
\begin{definition}{生成子}
    设$T:X\to X$是可逆保测变换，$\eta$是$X$的一个有限可测分解，$\bigvee_{i=-n}^n{T^{-i}\eta}$生成的代数记为$\mcal{A}_n$.\par
    令$\mcal{A}=\bigcup_{n=1}^\infty{\mcal{A}_n}$，$\mcal{A}$是包含有限可测分解$T^{-i}\eta\,(i\in\mb{N})$的最小代数$(\mcal{A}_1\subseteq\mcal{A}_2\subseteq\cdots\subseteq\mcal{A}_n\subseteq\cdots)$.\par
    记$\bigvee_{i=-\infty}^{\infty}{T^{-i}\eta}=\mcal{B}(\mcal{A})$\par
    如果$\bigvee_{i=-\infty}^{\infty}{T^{-i}\eta}=\mcal{B}(\mcal{A})\circeq \mcal{B}$，那么称$\eta$为$T$的一个\textbf{生成子}.
\end{definition}
下面的定理指出保测映射的测度熵等于其关于生成子的测度熵，因此计算测度熵时，只需计算其关于生成子的测度熵即可.
\begin{theorem}{Kolmogorov-Sinai 定理}
    设$(X,\mcal{B},m)$是概率测度空间，$T:X\to X$是可逆保测映射，如果$\eta$是$T$的一个生成子，那么$h_m(T)=h_m(T,\eta)$.
\end{theorem}
\begin{proof}
    设$\xi$ 是$X$ 的一个有限可测分解，需证$h(T,\xi)\leq h(T,\eta)$.\par
    设$m,n\in\mb{N}$，$\setjot[-2]\begin{aligned}[t]
        H \left(\bigvee_{i=0}^{n-1}T^{-i}\xi\right) &\leq H \left(\bigvee_{i=0}^{n-1}T^{-i}\xi\vee\bigvee_{j=-m}^{m+n}T^{-j}\eta\right)\\
        &=H \left(\bigvee_{j=-m}^{m+n}T^{-j}\eta\right)+H \left(\bigvee_{i=0}^{n-1}T^{-i}\xi\,\Bigg|\bigvee_{j=-m}^{m+n}T^{-j}\eta\right) \\
        &\leq H \left(\bigvee_{j=-m}^{m+n}T^{-j}\eta\right)+\sum_{i=0}^{n-1}H \left(T^{-i}\xi\,\Bigg|\bigvee_{j=-m}^{m+n}T^{-j}\eta\right)\\
        &\xlongequal{T\,\text{保测}} H \left(\bigvee_{j=-m}^{m+n}T^{-j}\eta\right)+\sum_{i=0}^{n-1}H \left(\xi\,\Bigg|\bigvee_{j=-m}^{m+n}T^{i-j}\eta\right) \\
        &\leq H \left(\bigvee_{j=-m}^{m+n}T^{-j}\eta\right)+\sum_{i=0}^{n-1}H \left(\xi\,\Bigg|\bigvee_{j=-m}^{m}T^{-j}\eta\right)
    \end{aligned} $\par
    $\Rightarrow \setjot[-2]\begin{aligned}[t]
        \frac{1}{n}H \left(\bigvee_{i=0}^{n-1}T^{-i}\eta\right) &\leq \frac{1}{n}H\left(\bigvee_{j=-m}^{m+n}T^{-j}\eta\right)+H \left(\xi\,\Bigg|\bigvee_{j=-m}^{m}T^{-j}\eta\right)\\
        &=\frac{1}{n}H \left(\bigvee_{j=0}^{n+2m}T^{-j}\eta\right)+H \left(\xi\,\Bigg|\bigvee_{j=-m}^{m}T^{j}\eta\right)
    \end{aligned} $ \par
    $\Rightarrow H(T,\xi)\leq H(T,\eta)+H \left(\xi\,\Bigg|\bigvee_{j=-m}^{m}T^{-j}\eta\right)$.\par
    因此，只需证明$\forall\,\varepsilon>0,\exists\, m\in\mb{N}$，使得$H \left(\xi\,\Bigg|\bigvee_{j=-m}^{m}T^{j}\eta\right)<\varepsilon$.\par
    由$\eta$ 为生成子可知$\bigvee_{i=-\infty}^\infty{T^{-i}\eta}\circeq \mcal{B}$\par
    $\bigvee_{i=-\infty}^\infty{T^{-i}\eta}$是由代数$\mcal{A}=\bigcup_{n=1}^\infty{\mcal{A}_n},\mcal{A}_n=\bigvee_{i=-n}^n{T^{-i}\eta}$生成的($\mcal{A}_1\subseteq \mcal{A}_2\subseteq\cdots\subseteq \mcal{A}_n\subseteq\cdots$).\par
    则$\mcal{B}\circeq\mcal{B}(\mcal{A})$，故$\forall\, B\in\mcal{B},\forall\,\varepsilon>0,\exists\, A\in\mcal{A}$，使得$m \left(B\triangle A\right)<\varepsilon$.\par
    设$\xi=\{\mcal{B}_1,\mcal{B}_2,\cdots,\mcal{B}_k\}$，则$\exists\, n\in\mb{N},A_1,A_2,\cdots,A_k\in\bigvee_{i=-m}^m{T^{-i}\eta}$生成的代数$\mcal{A}_m$，使得$m \left(B_i\triangleq A_i\right)<\delta\;(1\leq i\leq k)$.\par
    $H \left(\xi\,\Bigg|\bigvee_{\ell=-m}^m{T^{-\ell}\eta}\right)=-\sum_{\ell}{m(C_i)\sum_{j=1}^k\frac{m(B_j\cap C_i)}{m(C_i)}\ln{\frac{m(B_j\cap C_i)}{m(C_i)}}}\; \left(\bigvee_{\ell=-m}^{m}T^{-\ell}\eta=\{C_i\}_i\right)$\par
    $\forall\, j\in\{1,2,\cdots,k\}$，$A_j=\bigcup_{i}{C_{i,j}}\in\mcal{A}_m,C_{i,j}\in\{C_i\}_i$.\par
    调整$A_j'$，令$S_j=\left\{C_{i,j}\Bigl|m(B_j|C_{i,j})\geq 1-\sqrt{\delta},\mathrm{i.e.}\, \frac{m(B_j\cap C_{i,j})}{m(C_{i,j})}\geq 1-\sqrt{\delta}\right\}$\par
    定义$A_j'=\bigcup_{C_{i,j}\in S_j}{C_{i,j}},A_j'\subseteq A_j\;(\forall\, j\in\{1,2,\cdots,k\})$
    \begin{claim}{$m \left(A_j\setminus A_j'\right)\leq \sqrt{\delta}$}
        $A_j\setminus A_j'=\bigcup_{C_{i,j}\notin S_j}{C_{i,j}}$\par
        $C_{i,j}\notin S_j\Rightarrow m(B_j|C_{i,j})<1-\sqrt{\delta}\Rightarrow m \left(B_j^c|C_{i,j}\right)\geq \sqrt{\delta}$.\par
        则$m(A_i\setminus A_i')=\sum_{C_{i,j}\notin S_j}{m(C_{i,j})}\setjot[-2]\begin{aligned}[t]
            &\leq \frac{1}{\sqrt{\delta}}\sum_{C_{i,j}\notin S_j}m(C_{i,j})m(B_j^c|C_{i,j}) \\
            &=\frac{1}{\sqrt{\delta}}\sum_{C_{i,j}\notin S_j}{m(B_j^c\cap C_{i,j})} \\
            &\leq\frac{1}{\sqrt{\delta}}m(B_j^c\cap A_j) \\
            &\leq \frac{1}{\sqrt{\delta}}m(A_j\triangle B_j)<\sqrt{\delta} \\
        \end{aligned}$\par
    \end{claim}\par
    对$H \left(\xi\,\Bigg|\bigvee_{\ell=-m}^m{T^{-\ell}\eta}\right)$分组求和，则 \par
    $\setjot[-2]\begin{aligned}[t]
        H \left(\xi\,\Bigg|\bigvee_{\ell=-m}^m{T^{-\ell}\eta}\right)&=-\sum_{i}{m(C_i)\sum_{j=1}^k\frac{m(B_j\cap C_i)}{m(C_i)}\ln{\frac{m(B_j\cap C_i)}{m(C_i)}}}\\
        &=-\sum_{C_i\subseteq A_j'}+\left(-\sum_{C_i\cap\bigcup_{j=1}^{k}A_j'=\varnothing}\right)
    \end{aligned} $\par
    \begin{itemize}
        \item $-\sum_{C_{i}\subseteq A_j'}{{m(C_i)\sum_{j=1}^k\frac{m(B_j\cap C_i)}{m(C_i)}\ln{\frac{m(B_j\cap C_i)}{m(C_i)}}}}$.\par
        如果$C_i\subseteq A_s'$，那么$\frac{m(B_s\cap C_i)}{m(C_i)}\geq 1-\sqrt{\delta},\frac{m(B_j\cap C_i)}{m(C_i)}\leq \sqrt{\delta}\;(\forall\, j\neq s)$\par
        因此$-\sum_{C_{i}\subseteq A_j'}{{m(C_i)\sum_{j=1}^k\frac{m(B_j\cap C_i)}{m(C_i)}\ln{\frac{m(B_j\cap C_i)}{m(C_i)}}}}\leq k \left((1-\sqrt{\delta})\ln(1-\sqrt{\delta})+\sqrt{\delta}\ln\sqrt{\delta}\right)$
        \item $-\sum_{C_i\cap\bigcup_{j=1}^{k}A_j'=\varnothing}{{m(C_i)\sum_{j=1}^k\frac{m(B_j\cap C_i)}{m(C_i)}\ln{\frac{m(B_j\cap C_i)}{m(C_i)}}}}$\par
        \begin{claim}{$m \left(X\setminus \displaystyle\bigcup_{j=1}^k{A_j'}\right)\leq 2k\sqrt{\delta}$}
            $\setjot[-2]\begin{aligned}[t]
                m \left(X\setminus \displaystyle\bigcup_{j=1}^k{A_j'}\right) &= m \left(\bigcup_{j=1}^k{B_j}\setminus \bigcup_{j=1}^kA_j'\right)\\
                &\leq m \left(\bigcup_{j=1}^k{B_j}\setminus \bigcup_{j=1}^k{A_j}\right)+m \left(\bigcup_{j=1}^k{A_j}\setminus \bigcup_{j=1}^{k}{A_j'}\right)\\
                &\leq \sum_{j=1}^k{m(B_j\setminus A_j)}+\sum_{j=1}^k{m(A_j\setminus A_j')} \\
                &\leq k\delta+k\sqrt{\delta}\leq 2k\sqrt{\delta}
            \end{aligned}$
        \end{claim}
        $-\sum_{C_i\cap\bigcup_{j=1}^{k}A_j'=\varnothing}{{m(C_i)\sum_{j=1}^k\frac{m(B_j\cap C_i)}{m(C_i)}\ln{\frac{m(B_j\cap C_i)}{m(C_i)}}}}\leq \sum_{C_i\cap\bigcup_{j=1}^{k}A_j'=\varnothing}{m(C_i)\ln k}\leq 2k\ln{k}\sqrt{\delta}$
    \end{itemize}\par
    适当选取$\delta>0$，使得$-k\left((1-\sqrt{\delta})\ln(1-\sqrt{\delta})+\sqrt{\delta}\ln\sqrt{\delta}\right)+2k\ln{k}\sqrt{\delta}<\varepsilon$.\par
    那么$H \left(\xi\,\Bigg|\bigvee_{\ell=-m}^m{T^{-\ell}\eta}\right)<\varepsilon$.
\end{proof}
对于不可逆的保测变换，也有类似的结论：
\begin{theorem}{}
    设$(X,\mcal{B},m)$为概率测度空间，$T:X\to X$是保测变换，如果有限可测分解$\eta$满足$\bigvee_{i=0}^\infty{T^{-i}\eta}\circeq \mcal{B}$，那么$h_m(T)=h(T,\eta)$
\end{theorem}

\begin{instance}
    $\setjot[-6]\begin{aligned}[t]
        R_\alpha :S^1& \to S^1 \\
        x&\mapsto x+\alpha\pmod{1} 
    \end{aligned}$，其中$m$为Lebesgue测度，计算$h_m(R_\alpha)$.\par
    \begin{solution}
    \begin{enumerate}[(1)]
        \item 设$\alpha\in\mb{Q},\exists\, k\in\mb{N}$，使得$R_\alpha^k=\mathrm{Id}$.\par
        $h_m(R_\alpha^k)=h_m(\mathrm{Id})=\sup_\eta{\lim_{n\to\infty}{\frac{1}{n}H \left(\bigvee_{i=0}^{n-1}\left(\mathrm{Id}\right)^{-i}\eta\right)}}=0$\par
        $\Rightarrow kh_m(R_\alpha)=0\Rightarrow h_m(R_\alpha)=0$
        \item 设$\alpha\in\mb{R}\setminus\mb{Q}$，设$A_1=\left[0,\frac{1}{2}\right),A_2=\left[\frac{1}{2},1\right)$，则$\eta=\{A_1,A_2,\}$是$R_\alpha$的生成子.\par
        考虑$R_\alpha^{-i}\eta(-\infty<i<+\infty)$.\par
        $\left|\bigvee_{i=-m}^m{R_\alpha}^{-i}\eta\right|=(2m+1)\times 2$，故$\alpha\in\mb{R}\setminus\mb{Q}$，$\{n\alpha\}_{n\in\mb{N}}$在$S^1$中稠密.\par
        则$\bigvee_{i=-\infty}^{\infty}{R_\alpha^{-i}}\eta=\mcal{B}\left(\bigcup_{n\in\mb{N}}{\mcal{A}\left(\bigvee_{i=-m}^m R_\alpha^{-i}\eta\right)}\right)\circeq S^1$上的$\sigmaalg$.\par
        $h_m(R_\alpha)=h(R_\alpha,\eta)\setjot[-2]\begin{aligned}[t]
            &=\lim_{n\to\infty}{\frac{1}{n}H \left(\bigvee_{i=0}^{n-1}R_\alpha^{-i}\eta\right)} \\
            &\leq \limsup_{n\to\infty}{\frac{1}{n}\ln \left|\bigvee_{i=0}^{n-1}{R_\alpha^{-i}\eta}\right|}=\limsup_{n\to\infty}{\frac{1}{n}\ln{2n}}=0
        \end{aligned}$
    \end{enumerate}
    \end{solution}
\end{instance}

\begin{instance}
    设$X=\{1,2,\cdots,k\},Y=\prod_{i=-\infty}^{\infty}{X},\setjot[-6]\begin{aligned}[t]
        T:Y&\to Y \\
        (\cdots,x_{-1},x_0,x_1,\cdots) &\mapsto (\cdots,x_0,x_1,x_2,\cdots)
    \end{aligned}$\par
    $T$是$Y$上的左平移映射，设$m_X$是$X$上的概率测度$m_X(\{i\})=\frac{1}{k}\;(\forall\, i\in X)$，由此可得$Y$上的概率测度$m_Y$，计算$h_{m_Y}(T)$.\par
    \begin{solution}
        设$A_i=\{(\cdots,x_{-1},x_0,x_1,\cdots)\in Y|x_0=i\}\;(\forall\, i\in X)$.\par
        则$\eta=\{A_1,A_2,\cdots,A_k\}$是$Y$的有限可测分解，且$\eta$是$T$的生成子.\par
        $\bigvee_{i=-m}^m{T^{-i}\eta}$中的每一个元素$S=\bigcap_{i=-m}^m{T^{-i}A_{k_i}}=\{(\cdots,x_{-1},x_0,x_1,\cdots)|x_i=k_i,-m\leq i\leq m\}\; (\forall\, k_i\in X)$.\par
        $\Rightarrow \bigvee_{i=-\infty}^\infty{T^{-i}\eta}$包含$Y$中的所有矩体，故$\bigvee_{i=-\infty}^\infty{T^{-i\eta}}\circeq\mcal{B}$.\par
        $\setjot[-2]\begin{aligned}[t]
            h_{m_Y}(T) &= h(T,\eta) =\lim_{n\to\infty}{\frac{1}{n}H \left(\bigvee_{i=0}^{n-1}T^{-i}\eta\right)} \\
            &=\lim_{n\to\infty}{\frac{1}{n}\sum_{A_j\in\bigvee_{i=0}^{n-1}T^{-i}\eta}{-m(A_j)\ln{m(A_j)}}} \\
            &=\lim_{n\to\infty}{\frac{1}{n}\sum_{A_j\in\bigvee_{i=0}^{n-1}T^{-i}\eta}{-\frac{1}{k^n}\ln{\frac{1}{k^n}}}}\; \left(m(A_j)=\frac{1}{k^n}\right) \\
            &=\lim_{n\to\infty}{\frac{1}{n}\left(k^n\cdot \left(-\frac{1}{k^n}\ln\frac{1}{k^n}\right)\right)}=\ln{k}
        \end{aligned}$
    \end{solution}
    因此$\setjot[-2]\begin{aligned}[t]
        & T_2:\prod_{i=-\infty}^\infty\{1,2\}\to \prod_{i=-\infty}^\infty\{1,2\} \\
        &T_3:\prod_{i=-\infty}^\infty\{1,2,3\}\to\prod_{i=-\infty}^\infty\{1,2,3\}
    \end{aligned}$不同构.\par

    对于单边的情形$\setjot[-2]\begin{aligned}[t]
        T:\prod_{i=0}^\infty{X}&\to \prod_{i=0}^\infty{X} \\
        (x_0,x_1,x_2,\cdots) &\mapsto (x_1,x_2,\cdots)
    \end{aligned}$\par
    $\eta={A_1,A_2,\cdots,A_k}$同样为$T$的生成子.\par

\end{instance}
